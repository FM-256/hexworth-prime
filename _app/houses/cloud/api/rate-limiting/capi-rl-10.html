2
Token Bucket Rate Limiter

Implement the token bucket algorithm from RL-02. This middleware checks the client's token bucket before allowing the request through.

// src/middleware/rateLimiter.js const buckets = new Map(); function createBucket(capacity, refillRate) { return { tokens: capacity, capacity, refillRate, lastRefill: Date.now() }; } function refill(bucket) { const now = Date.now(); const elapsed = (now - bucket.lastRefill) / 1000; bucket.tokens = Math.min(bucket.capacity, bucket.tokens + elapsed * bucket.refillRate); bucket.lastRefill = now; } function rateLimiter(options = {}) { const capacity = options.capacity || 10; const refillRate = options.refillRate || 1; return (req, res, next) => { const clientId = req.headers['x-api-key'] || req.ip; if (!buckets.has(clientId)) { buckets.set(clientId, createBucket(capacity, refillRate)); } const bucket = buckets.get(clientId); refill(bucket); // Set rate limit headers on every response res.set('X-RateLimit-Limit', capacity); res.set('X-RateLimit-Remaining', Math.floor(bucket.tokens)); if (bucket.tokens >= 1) { bucket.tokens -= 1; next(); } else { const retryAfter = Math.ceil(1 / refillRate); res.set('Retry-After', retryAfter); res.status(429).json({ error: 'rate_limit_exceeded', message: `Too many requests. Retry after ${retryAfter}s`, retry_after: retryAfter }); } }; } module.exports = rateLimiter;
3
Response Caching Layer

Add an in-memory cache that stores API responses with TTL. Only GET requests are cached. Cache keys include the full URL and query parameters.

// src/middleware/cacheLayer.js const NodeCache = require('node-cache'); const cache = new NodeCache({ stdTTL: 60, checkperiod: 30 }); function cacheLayer(ttl = 60) { return (req, res, next) => { if (req.method !== 'GET') return next(); const key = `cache:${req.originalUrl}`; const cached = cache.get(key); if (cached) { res.set('X-Cache', 'HIT'); return res.json(cached); } // Intercept res.json to capture the response const originalJson = res.json.bind(res); res.json = (data) => { if (res.statusCode === 200) { cache.set(key, data, ttl); } res.set('X-Cache', 'MISS'); originalJson(data); }; next(); }; } cacheLayer.invalidate = (pattern) => { const keys = cache.keys().filter(k => k.includes(pattern)); keys.forEach(k => cache.del(k)); return keys.length; }; module.exports = cacheLayer;
4
Circuit Breaker

Implement the circuit breaker pattern from RL-04. This protects downstream services from cascading failures.

// src/middleware/circuitBreaker.js class CircuitBreaker { constructor(name, options = {}) { this.name = name; this.state = 'CLOSED'; this.failureCount = 0; this.threshold = options.threshold || 5; this.timeout = options.timeout || 30000; this.lastFailure = 0; } async call(fn) { if (this.state === 'OPEN') { if (Date.now() - this.lastFailure > this.timeout) { this.state = 'HALF_OPEN'; console.log(`[${this.name}] Circuit HALF_OPEN - testing`); } else { throw { status: 503, message: 'Service temporarily unavailable' }; } } try { const result = await fn(); this.failureCount = 0; this.state = 'CLOSED'; return result; } catch (err) { this.failureCount++; this.lastFailure = Date.now(); if (this.failureCount >= this.threshold) { this.state = 'OPEN'; console.log(`[${this.name}] Circuit OPEN after ${this.failureCount} failures`); } throw err; } } status() { return { name: this.name, state: this.state, failures: this.failureCount }; } } module.exports = CircuitBreaker;
5
Wire It All Together

Create the Express server that combines rate limiting, caching, and circuit breaking into a complete API gateway.

// src/server.js const express = require('express'); const rateLimiter = require('./middleware/rateLimiter'); const cacheLayer = require('./middleware/cacheLayer'); const CircuitBreaker = require('./middleware/circuitBreaker'); const app = express(); const breaker = new CircuitBreaker('backend', { threshold: 3, timeout: 10000 }); // Global rate limit: 10 req burst, 2/sec refill app.use(rateLimiter({ capacity: 10, refillRate: 2 })); // Cached product listing app.get('/api/products', cacheLayer(30), async (req, res) => { try { const data = await breaker.call(() => fetchProducts()); res.json(data); } catch (err) { res.status(err.status || 500).json({ error: err.message }); } }); // Health + circuit status app.get('/api/health', (req, res) => { res.json({ status: 'ok', circuit: breaker.status() }); }); app.listen(3000, () => console.log('Gateway on :3000'));

Test Under Load

Use curl or a load testing tool to observe the gateway behavior under different conditions.

Lab Takeaways

  • Token bucket rate limiting provides burst tolerance with sustained rate control
  • Response caching with X-Cache headers makes cache behavior observable
  • Circuit breakers prevent cascading failures to downstream services
  • All three patterns compose cleanly as Express middleware
  • Load testing reveals the interaction between rate limiting, caching, and circuit breaking
