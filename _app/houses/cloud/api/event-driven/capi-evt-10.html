WebSocket Connection Manager

Build the connection manager that tracks connected users, handles authentication, and routes messages to specific users or channels.

// src/wsManager.js class WSManager { constructor() { this.connections = new Map(); // userId -> Set of ws this.channels = new Map(); // channelId -> Set of userId } addConnection(userId, ws) { if (!this.connections.has(userId)) { this.connections.set(userId, new Set()); } this.connections.get(userId).add(ws); ws.userId = userId; ws.isAlive = true; console.log(`User ${userId} connected (${this.connections.get(userId).size} sessions)`); } removeConnection(ws) { const userId = ws.userId; if (userId && this.connections.has(userId)) { this.connections.get(userId).delete(ws); if (this.connections.get(userId).size === 0) { this.connections.delete(userId); } } } sendToUser(userId, event) { const sockets = this.connections.get(userId); if (!sockets) return false; const msg = JSON.stringify(event); sockets.forEach(ws => { if (ws.readyState === 1) ws.send(msg); }); return true; } sendToChannel(channelId, event) { const subscribers = this.channels.get(channelId) || new Set(); subscribers.forEach(userId => this.sendToUser(userId, event)); } subscribe(userId, channelId) { if (!this.channels.has(channelId)) { this.channels.set(channelId, new Set()); } this.channels.get(channelId).add(userId); } getStats() { return { users: this.connections.size, channels: this.channels.size, totalConnections: [...this.connections.values()] .reduce((sum, set) => sum + set.size, 0) }; } } module.exports = new WSManager();
3
WebSocket Server with Auth and Heartbeat

Wire up the WebSocket server with token-based authentication and ping/pong heartbeats.

// src/server.js const express = require('express'); const { WebSocketServer } = require('ws'); const http = require('http'); const manager = require('./wsManager'); const app = express(); app.use(express.json()); const server = http.createServer(app); const wss = new WebSocketServer({ server }); // WebSocket connection handler wss.on('connection', (ws, req) => { const token = new URL(req.url, 'http://x').searchParams.get('token'); const userId = validateToken(token); if (!userId) { ws.close(4001, 'Unauthorized'); return; } manager.addConnection(userId, ws); ws.on('message', (data) => { try { const msg = JSON.parse(data); if (msg.type === 'subscribe') { manager.subscribe(userId, msg.channel); ws.send(JSON.stringify({ type: 'subscribed', channel: msg.channel })); } } catch (e) { ws.send(JSON.stringify({ error: 'invalid_message' })); } }); ws.on('pong', () => { ws.isAlive = true; }); ws.on('close', () => manager.removeConnection(ws)); }); // Heartbeat every 30 seconds setInterval(() => { wss.clients.forEach(ws => { if (!ws.isAlive) return ws.terminate(); ws.isAlive = false; ws.ping(); }); }, 30000); // REST API: Send notification app.post('/api/notify', (req, res) => { const { userId, channel, type, data } = req.body; const event = { type: type || 'notification', data, timestamp: Date.now() }; if (userId) { manager.sendToUser(userId, event); } else if (channel) { manager.sendToChannel(channel, event); } res.json({ sent: true, stats: manager.getStats() }); }); // Webhook receiver app.post('/webhooks/external', (req, res) => { const event = req.body; // Route external events to relevant channels if (event.type === 'order.shipped') { manager.sendToUser(event.data.customerId, { type: 'notification', data: { title: 'Order Shipped!', body: event.data.trackingNumber } }); } res.status(200).json({ received: true }); }); // Health endpoint app.get('/health', (req, res) => res.json(manager.getStats())); server.listen(3000, () => console.log('Notification server on :3000'));

Lab Takeaways

  • WebSocket connection managers track users, sessions, and channels
  • REST APIs and webhooks can both trigger real-time notifications
  • Channel-based routing enables targeted message delivery
  • Heartbeats detect and clean up dead connections
  • Token-based auth during WebSocket handshake secures connections
