MODULE 3 — Container Escape Vectors

Identify configurations vulnerable to container escape +10 per correct ID / +15 per MC
Review four docker run commands. Check every configuration that creates a container escape risk — where an attacker who compromises the container could break out to the host.
CONFIG A: Internal reporting service
docker run -d \
  --name reporter \
  -p 8080:8080 \
  -v /var/log:/var/log:ro \
  -e REPORT_KEY=abc123 \
  --memory=512m \
  --cpus=0.5 \
  myapp/reporter:v1.2
CONFIG B: CI/CD build agent
docker run -d \
  --name build-agent \
  --privileged \
  -v /var/run/docker.sock:/var/run/docker.sock \
  -v /home/ci:/workspace \
  -e CI_TOKEN=ghp_xxxxxxxxxxxx \
  myapp/builder:latest
CONFIG C: System diagnostics container
docker run -d \
  --name diagnostics \
  --pid=host \
  --cap-add=SYS_ADMIN \
  --cap-add=SYS_PTRACE \
  -v /proc:/host-proc:rw \
  myapp/diag:v3
CONFIG D: Web frontend service
docker run -d \
  --name frontend \
  --read-only \
  --tmpfs /tmp \
  -p 443:443 \
  --cap-drop=ALL \
  --cap-add=NET_BIND_SERVICE \
  --security-opt=no-new-privileges \

  name: app-reader

  resources: ["pods", "services", "endpoints"]
  verbs: ["get", "list", "watch"]
- apiGroups: [""]

  name: deploy-manager
  namespace: staging

- apiGroups: ["*"]
  resources: ["*"]
  verbs: ["*"]

  name: debug-access

- apiGroups: [""]
