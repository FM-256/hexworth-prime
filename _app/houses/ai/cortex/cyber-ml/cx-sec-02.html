import torch
import torch.nn as nn
import numpy as np
from sklearn import metrics

class Pipeline:
    def __init__(self, config):
        self.config = config
        self.model = None

    def train(self, X_train, y_train):
        # Feature engineering specific to Network Intrusion Detection
        features = self.extract_features(X_train)
        # Model training with appropriate hyperparameters
        self.model.fit(features, y_train)

    def evaluate(self, X_test, y_test):
        # Compute domain-appropriate metrics
        preds = self.model.predict(self.extract_features(X_test))
        return {
            'accuracy': metrics.accuracy_score(y_test, preds),
            'f1': metrics.f1_score(y_test, preds, average='weighted'),
            'auc': metrics.roc_auc_score(y_test, preds)
        }

Class Imbalance

Security datasets are almost always severely imbalanced -- malicious events are rare compared to benign ones. Always evaluate with precision/recall and AUC-ROC rather than accuracy. Use SMOTE, class weights, or threshold calibration to handle imbalance.
