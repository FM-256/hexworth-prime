Hands-On: DQN Agent

A complete DQN pipeline from scratch: environment setup, network architecture, replay buffer, training loop, evaluation protocol, and learning curve visualization. The target environment is LunarLander-v2.

PHASE 01 — ENVIRONMENT SETUP

Environment and Dependencies

# Installation # pip install gymnasium[box2d] torch numpy matplotlib tqdm import gymnasium as gym import torch import torch.nn as nn import torch.nn.functional as F import numpy as np import matplotlib.pyplot as plt from collections import deque import random from tqdm import tqdm # Environment: LunarLander-v2 # Observation: 8-dimensional vector (position, velocity, angle, angular vel, leg contacts) # Actions: 4 discrete (do nothing, left engine, main engine, right engine) # Reward: +100-140 for landing, -100 for crash, -0.3/frame for main engine # Solved: mean reward >= 200 over 100 consecutive episodes env = gym.make('LunarLander-v2') OBS_DIM = env.observation_space.shape[0] # 8 N_ACTIONS = env.action_space.n # 4 print(f"Observation space: {env.observation_space}") print(f"Action space: {env.action_space}") print(f"Reward range: {env.reward_range}") # seed for reproducibility SEED = 42 torch.manual_seed(SEED) np.random.seed(SEED) random.seed(SEED) DEVICE = torch.device('cuda' if torch.cuda.is_available() else 'cpu') print(f"Device: {DEVICE}")
PHASE 02 — NETWORK ARCHITECTURE

DQN Network

For the LunarLander vector observation, we use a Dueling DQN architecture. The shared feature extractor feeds two heads: a value stream V(s) and an advantage stream A(s,a). This separates "how good is this state" from "which action is best here".

class DuelingDQN(nn.Module): """ Dueling DQN for LunarLander-v2. Shared backbone + separate V(s) and A(s,a) heads. Q(s,a) = V(s) + A(s,a) - mean(A(s,·)) """ def __init__(self, obs_dim=8, n_actions=4, hidden=256): super().__init__() # shared feature extractor self.backbone = nn.Sequential( nn.Linear(obs_dim, hidden), nn.ReLU(), nn.Linear(hidden, hidden), nn.ReLU(), ) # value stream: V(s) — how good is this state overall? self.value = nn.Sequential( nn.Linear(hidden, hidden // 2), nn.ReLU(), nn.Linear(hidden // 2, 1), ) # advantage stream: A(s,a) — how much better is each action? self.advantage = nn.Sequential( nn.Linear(hidden, hidden // 2), nn.ReLU(), nn.Linear(hidden // 2, n_actions), ) # initialize weights for stable early training self._init_weights() def _init_weights(self): for m in self.modules(): if isinstance(m, nn.Linear): nn.init.orthogonal_(m.weight, gain=np.sqrt(2)) nn.init.constant_(m.bias, 0.0) def forward(self, x): features = self.backbone(x) V = self.value(features) # (batch, 1) A = self.advantage(features) # (batch, n_actions) Q = V + (A - A.mean(dim=1, keepdim=True)) # (batch, n_actions) return Q # sanity check: output shape model = DuelingDQN(OBS_DIM, N_ACTIONS).to(DEVICE) dummy = torch.zeros(1, OBS_DIM).to(DEVICE) print(f"Output shape: {model(dummy).shape}") # torch.Size([1, 4]) n_params = sum(p.numel() for p in model.parameters()) print(f"Parameters: {n_params:,}") # ~200k
PHASE 03 — REPLAY BUFFER

Experience Replay Buffer

class ReplayBuffer: """Ring buffer for (state, action, reward, next_state, done) transitions.""" def __init__(self, capacity=100_000): self.buffer = deque(maxlen=capacity) def push(self, s, a, r, ns, d): self.buffer.append((np.array(s, dtype=np.float32), int(a), float(r), np.array(ns, dtype=np.float32), float(d))) def sample(self, n): batch = random.sample(self.buffer, n) s, a, r, ns, d = zip(*batch) return (torch.tensor(np.array(s), device=DEVICE), torch.tensor(a, device=DEVICE, dtype=torch.long), torch.tensor(r, device=DEVICE, dtype=torch.float32), torch.tensor(np.array(ns), device=DEVICE), torch.tensor(d, device=DEVICE, dtype=torch.float32)) def __len__(self): return len(self.buffer)
PHASE 04 — AGENT

DQN Agent

class DDQNAgent: """ Double DQN agent with Dueling network. Combines: Dueling architecture + Double Q-target + experience replay. """ # hyperparameters — tuned for LunarLander-v2 BATCH_SIZE = 64 GAMMA = 0.99 LR = 5e-4 BUFFER_SIZE = 100_000 TARGET_UPDATE = 500 # sync target every N gradient steps EPS_START = 1.0 EPS_END = 0.01 EPS_DECAY = 30_000 # linear decay over this many steps MIN_BUFFER = 1_000 # start training after this many samples def __init__(self, obs_dim, n_actions): self.n_actions = n_actions self.online = DuelingDQN(obs_dim, n_actions).to(DEVICE) self.target = DuelingDQN(obs_dim, n_actions).to(DEVICE) self.target.load_state_dict(self.online.state_dict()) self.target.eval() self.optimizer = torch.optim.Adam(self.online.parameters(), lr=self.LR) self.buffer = ReplayBuffer(self.BUFFER_SIZE) self.steps = 0 self.losses = [] def epsilon(self): frac = min(1.0, self.steps / self.EPS_DECAY) return self.EPS_START + frac * (self.EPS_END - self.EPS_START) def act(self, state): if np.random.rand() < self.epsilon(): return np.random.randint(self.n_actions) s = torch.tensor(state, dtype=torch.float32, device=DEVICE).unsqueeze(0) with torch.no_grad(): return int(self.online(s).argmax().item()) def update(self): if len(self.buffer) < self.MIN_BUFFER: return s, a, r, ns, d = self.buffer.sample(self.BATCH_SIZE) # Double DQN: online selects next action, target evaluates it with torch.no_grad(): next_actions = self.online(ns).argmax(dim=1, keepdim=True) next_q = self.target(ns).gather(1, next_actions).squeeze(1) td_target = r + self.GAMMA * next_q * (1 - d) q_pred = self.online(s).gather(1, a.unsqueeze(1)).squeeze(1) loss = F.smooth_l1_loss(q_pred, td_target) # Huber loss self.optimizer.zero_grad() loss.backward() nn.utils.clip_grad_norm_(self.online.parameters(), 10.0) self.optimizer.step() self.losses.append(loss.item()) self.steps += 1 if self.steps % self.TARGET_UPDATE == 0: self.target.load_state_dict(self.online.state_dict())
PHASE 05 — TRAINING LOOP

Training Loop

def train(n_episodes=800, eval_every=50, eval_episodes=20): agent = DDQNAgent(OBS_DIM, N_ACTIONS) rewards_history = [] eval_history = [] solved_at = None for ep in tqdm(range(n_episodes), desc='Training'): state, _ = env.reset(seed=ep) done = False ep_reward = 0.0 while not done: action = agent.act(state) next_state, reward, term, trunc, _ = env.step(action) done = term or trunc agent.buffer.push(state, action, reward, next_state, done) agent.update() ep_reward += reward state = next_state rewards_history.append(ep_reward) # periodic evaluation using greedy policy (no exploration) if (ep + 1) % eval_every == 0: avg_eval = evaluate(agent, eval_episodes) eval_history.append((ep + 1, avg_eval)) eps = agent.epsilon() avg_train = np.mean(rewards_history[-eval_every:]) print(f"\nEp {ep+1:4d} | Train: {avg_train:7.1f} | " f"Eval: {avg_eval:7.1f} | Eps: {eps:.3f}") if avg_eval >= 200 and solved_at is None: solved_at = ep + 1 print(f" ** SOLVED at episode {solved_at} **") return agent, rewards_history, eval_history, solved_at def evaluate(agent, n_episodes=20): """Evaluate agent with greedy policy (epsilon=0).""" eval_env = gym.make('LunarLander-v2') orig_eps = agent.EPS_END total_rewards = [] for ep in range(n_episodes): state, _ = eval_env.reset(seed=1000 + ep) done = False ep_reward = 0.0 while not done: s = torch.tensor(state, dtype=torch.float32, device=DEVICE).unsqueeze(0) with torch.no_grad(): action = int(agent.online(s).argmax().item()) state, reward, term, trunc, _ = eval_env.step(action) done = term or trunc ep_reward += reward total_rewards.append(ep_reward) eval_env.close() return np.mean(total_rewards)
PHASE 06 — VISUALIZATION

Learning Curve Visualization

def plot_training(rewards_history, eval_history, solved_at=None): """ Plot smoothed training rewards and evaluation checkpoints. Saves figure to 'dqn_training.png'. """ fig, axes = plt.subplots(1, 2, figsize=(14, 5)) fig.patch.set_facecolor('#0f0f13') for ax in axes: ax.set_facecolor('#1a1a24') ax.tick_params(colors='#94a3b8') for spine in ax.spines.values(): spine.set_edgecolor('#2e2e48') # ── left: training curve with rolling average ────────────────── episodes = np.arange(len(rewards_history)) raw = np.array(rewards_history) window = 50 smooth = np.convolve(raw, np.ones(window)/window, mode='valid') axes[0].plot(episodes, raw, alpha=0.25, color='#8b5cf6', linewidth=0.8) axes[0].plot(np.arange(window-1, len(raw)), smooth, color='#8b5cf6', linewidth=2, label=f'Rolling-{window}') axes[0].axhline(200, color='#10b981', linestyle='--', linewidth=1.2, label='Solved threshold') if solved_at: axes[0].axvline(solved_at, color='#f59e0b', linestyle=':', linewidth=1.5, label=f'Solved @ ep {solved_at}') axes[0].set_xlabel('Episode', color='#94a3b8') axes[0].set_ylabel('Reward', color='#94a3b8') axes[0].set_title('Training Rewards', color='#e2e8f0') axes[0].legend(facecolor='#22223a', edgecolor='#2e2e48', labelcolor='#e2e8f0') # ── right: evaluation scores ─────────────────────────────────── eval_eps, eval_scores = zip(*eval_history) axes[1].plot(eval_eps, eval_scores, color='#34d399', marker='o', markersize=4, linewidth=2, label='Eval (greedy)') axes[1].axhline(200, color='#10b981', linestyle='--', linewidth=1.2) axes[1].set_xlabel('Episode', color='#94a3b8') axes[1].set_ylabel('Mean Reward (20 eps)', color='#94a3b8') axes[1].set_title('Evaluation Rewards', color='#e2e8f0') axes[1].legend(facecolor='#22223a', edgecolor='#2e2e48', labelcolor='#e2e8f0') plt.tight_layout() plt.savefig('dqn_training.png', dpi=150, facecolor=fig.get_facecolor()) plt.show() print("Saved: dqn_training.png")
PHASE 07 — CHECKPOINTING

Model Checkpointing

def save_checkpoint(agent, path='dqn_lunarlander.pt'): """Save online network weights and training state.""" torch.save({ 'online_state_dict': agent.online.state_dict(), 'target_state_dict': agent.target.state_dict(), 'optimizer_state': agent.optimizer.state_dict(), 'steps': agent.steps, 'losses': agent.losses[-1000:], # keep last 1000 losses }, path) print(f"Checkpoint saved: {path}") def load_checkpoint(path='dqn_lunarlander.pt'): """Load and resume a saved agent.""" agent = DDQNAgent(OBS_DIM, N_ACTIONS) ckpt = torch.load(path, map_location=DEVICE) agent.online.load_state_dict(ckpt['online_state_dict']) agent.target.load_state_dict(ckpt['target_state_dict']) agent.optimizer.load_state_dict(ckpt['optimizer_state']) agent.steps = ckpt['steps'] agent.losses = ckpt['losses'] print(f"Loaded checkpoint: step {agent.steps}") return agent
PHASE 08 — EVALUATION AND ANALYSIS

Detailed Evaluation

def detailed_evaluation(agent, n_episodes=100): """ Run 100 evaluation episodes and report detailed statistics. Tracks: mean/std/min/max reward, crash rate, landing rate, timeout rate. """ eval_env = gym.make('LunarLander-v2') rewards, outcomes = [], [] for ep in range(n_episodes): state, _ = eval_env.reset(seed=2000 + ep) done = False ep_reward = 0.0 steps = 0 terminated_crashed = False while not done: s = torch.tensor(state, dtype=torch.float32, device=DEVICE).unsqueeze(0) with torch.no_grad(): action = int(agent.online(s).argmax().item()) state, reward, term, trunc, info = eval_env.step(action) done = term or trunc ep_reward += reward steps += 1 if term and ep_reward < -50: terminated_crashed = True rewards.append(ep_reward) if terminated_crashed: outcomes.append('crash') elif ep_reward >= 200: outcomes.append('success') elif trunc: outcomes.append('timeout') else: outcomes.append('other') rewards = np.array(rewards) from collections import Counter outcome_counts = Counter(outcomes) print("\n=== Evaluation Results (100 episodes) ===") print(f"Mean Reward: {rewards.mean():.1f} +/- {rewards.std():.1f}") print(f"Min / Max: {rewards.min():.1f} / {rewards.max():.1f}") print(f"Solved >=200: {(rewards >= 200).mean()*100:.1f}%") print(f"Success: {outcome_counts['success']}") print(f"Crashes: {outcome_counts['crash']}") print(f"Timeouts: {outcome_counts['timeout']}") eval_env.close() return rewards
