Phase 2 — Tokenization

Phase 2
from transformers import AutoTokenizer

MODEL_NAME = 'bert-base-uncased'
MAX_LEN    = 512

tokenizer = AutoTokenizer.from_pretrained(MODEL_NAME)

def tokenize_batch(batch):
    """Tokenize a batch of texts. Returns input_ids, attention_mask, token_type_ids."""
    encoding = tokenizer(
        batch['text'],
        truncation=True,
        padding='max_length',   # pad to MAX_LEN in batch map; DataCollator handles dynamic padding
        max_length=MAX_LEN,
        return_tensors=None      # return lists for datasets map()
    )
    encoding['labels'] = batch['label']
    return encoding

# Apply in parallel across all splits
tokenized_train = train_ds.map(tokenize_batch, batched=True, batch_size=256, remove_columns=['text'])
tokenized_val   = val_ds.map(tokenize_batch,   batched=True, batch_size=256, remove_columns=['text'])
tokenized_test  = test_data.map(tokenize_batch, batched=True, batch_size=256, remove_columns=['text'])

tokenized_train.set_format('torch')
tokenized_val.set_format('torch')
tokenized_test.set_format('torch')

# Examine one tokenized example
sample = tokenized_train[0]
print(f"input_ids shape:    {sample['input_ids'].shape}")
print(f"attention_mask sum: {sample['attention_mask'].sum().item()}  (non-padding tokens)")

Phase 3 — Data Loading

Phase 3
from torch.utils.data import DataLoader
from transformers import DataCollatorWithPadding

# Dynamic padding: each batch padded to its longest sequence (more efficient than max_len)
collator = DataCollatorWithPadding(tokenizer=tokenizer)

BATCH_SIZE = 16   # fits on 12GB GPU; reduce to 8 if OOM; use gradient accumulation

train_loader = DataLoader(tokenized_train, batch_size=BATCH_SIZE, shuffle=True,  collate_fn=collator, num_workers=4, pin_memory=True)
val_loader   = DataLoader(tokenized_val,   batch_size=BATCH_SIZE*2, shuffle=False, collate_fn=collator, num_workers=4)
test_loader  = DataLoader(tokenized_test,  batch_size=BATCH_SIZE*2, shuffle=False, collate_fn=collator, num_workers=4)

Phase 4 — Model and Optimizer

Phase 4
import torch
from transformers import AutoModelForSequenceClassification, get_linear_schedule_with_warmup
from torch.optim import AdamW

device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')

model = AutoModelForSequenceClassification.from_pretrained(MODEL_NAME, num_labels=2)
model = model.to(device)

# Differential learning rates: smaller LR for pre-trained layers, larger for head
optimizer_grouped = [
    {'params': [p for n, p in model.bert.named_parameters()], 'lr': 2e-5},
    {'params': model.classifier.parameters(), 'lr': 5e-5}
]
optimizer = AdamW(optimizer_grouped, weight_decay=0.01, eps=1e-8)

NUM_EPOCHS   = 3
WARMUP_RATIO = 0.06
TOTAL_STEPS  = NUM_EPOCHS * len(train_loader)
WARMUP_STEPS = int(TOTAL_STEPS * WARMUP_RATIO)

scheduler = get_linear_schedule_with_warmup(
    optimizer,
    num_warmup_steps=WARMUP_STEPS,
    num_training_steps=TOTAL_STEPS
)

print(f"Total steps: {TOTAL_STEPS} | Warmup: {WARMUP_STEPS}")

Phase 5 — Training Loop

Phase 5
from torch.cuda.amp import autocast, GradScaler
import time

scaler = GradScaler()   # mixed precision: FP16 forward, FP32 gradients

def train_epoch(model, loader, optimizer, scheduler, scaler, device):
    model.train()
    total_loss, total_correct, total_samples = 0, 0, 0

    for step, batch in enumerate(loader):
        batch = {k: v.to(device) for k, v in batch.items()}

        with autocast():                    # FP16 forward pass
            outputs = model(**batch)
            loss    = outputs.loss

        scaler.scale(loss).backward()

        if (step + 1) % 1 == 0:            # update every step (no accumulation here)
            scaler.unscale_(optimizer)
            torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)
            scaler.step(optimizer)
            scaler.update()
            scheduler.step()
            optimizer.zero_grad()

        preds = outputs.logits.argmax(dim=-1)
        total_correct += (preds == batch['labels']).sum().item()
        total_samples += len(batch['labels'])
        total_loss    += loss.item()

        if step % 200 == 0:
            print(f"  step {step}/{len(loader)} | loss {loss.item():.4f}")

    return total_loss / len(loader), total_correct / total_samples


def evaluate(model, loader, device):
    model.eval()
    all_preds, all_labels, all_logits = [], [], []

    with torch.no_grad():
        for batch in loader:
            batch = {k: v.to(device) for k, v in batch.items()}
            outputs = model(**batch)
            preds   = outputs.logits.argmax(dim=-1)
            all_preds.extend(preds.cpu().tolist())
            all_labels.extend(batch['labels'].cpu().tolist())
            all_logits.extend(outputs.logits.softmax(dim=-1)[:, 1].cpu().tolist())

    return np.array(all_preds), np.array(all_labels), np.array(all_logits)


# Main training loop
best_val_f1 = 0
history = {'train_loss': [], 'val_acc': [], 'val_f1': []}

from sklearn.metrics import f1_score, accuracy_score

for epoch in range(NUM_EPOCHS):
    t0 = time.time()
    train_loss, train_acc = train_epoch(model, train_loader, optimizer, scheduler, scaler, device)

    val_preds, val_labels, val_probs = evaluate(model, val_loader, device)
    val_acc = accuracy_score(val_labels, val_preds)
    val_f1  = f1_score(val_labels, val_preds)

    history['train_loss'].append(train_loss)
    history['val_acc'].append(val_acc)
    history['val_f1'].append(val_f1)

    print(f"\nEpoch {epoch+1}/{NUM_EPOCHS} | {time.time()-t0:.0f}s")
    print(f"  train loss: {train_loss:.4f} | train acc: {train_acc:.4f}")
    print(f"  val   acc:  {val_acc:.4f} | val F1: {val_f1:.4f}")

    if val_f1 > best_val_f1:
        best_val_f1 = val_f1
        model.save_pretrained('best_bert_sentiment')
        tokenizer.save_pretrained('best_bert_sentiment')
        print(f"  Saved new best model (F1={best_val_f1:.4f})")

# Identify systematic error patterns
test_texts  = test_data['text']
errors_idx  = np.where(test_preds != test_labels)[0]

# Sort errors by confidence (highest first = most egregious mistakes)
error_confs = test_probs[errors_idx]
top_errors  = errors_idx[np.argsort(error_confs)[::-1][:10]]

print("=== Top 10 High-Confidence Errors ===\n")
for idx in top_errors:
    pred_label  = "positive" if test_preds[idx] else "negative"
    true_label  = "positive" if test_labels[idx] else "negative"
    print(f"[conf={test_probs[idx]:.3f}] Predicted: {pred_label} | True: {true_label}")
    print(f"  {test_texts[idx][:200]}...\n")

# Categorize errors by length (short reviews harder to classify?)
lengths = np.array([len(t.split()) for t in test_texts])
print(f"\nError rate by length:")
for lo, hi in [(0,50), (50,150), (150,300), (300,10000)]:
    mask  = (lengths >= lo) & (lengths < hi)
    wrong = (test_preds[mask] != test_labels[mask]).sum()
    total = mask.sum()
    print(f"  {lo}-{hi} words: {wrong}/{total} errors ({100*wrong/total:.1f}%)")

Phase 8 — Production Inference

Phase 8
class SentimentClassifier:
    """
    Production-ready BERT sentiment wrapper.
    Handles single strings and batches, returns label + confidence.
    """
    def __init__(self, model_path: str, device: str = 'cpu'):
        self.tokenizer = AutoTokenizer.from_pretrained(model_path)
        self.model     = AutoModelForSequenceClassification.from_pretrained(model_path)
        self.model.to(device).eval()
        self.device    = device
        self.labels    = ['negative', 'positive']

    def predict(self, texts, batch_size=32, threshold=0.5):
        """
        texts: str or list[str]
        returns: list of {'label': str, 'score': float, 'positive_prob': float}
        """
        if isinstance(texts, str):
            texts = [texts]

        results = []
        for i in range(0, len(texts), batch_size):
            batch_texts = texts[i:i+batch_size]
            encoding    = self.tokenizer(
                batch_texts, truncation=True, padding=True,
                max_length=512, return_tensors='pt'
            ).to(self.device)

            with torch.no_grad():
                logits = self.model(**encoding).logits
                probs  = logits.softmax(dim=-1).cpu()

            for prob in probs:
                pos_prob = prob[1].item()
                results.append({
                    'label':        'positive' if pos_prob >= threshold else 'negative',
                    'score':         max(pos_prob, 1 - pos_prob),
                    'positive_prob': pos_prob
                })
        return results

Threshold Tuning The default 0.5 threshold maximizes accuracy but not F1 or precision/recall trade-offs. For high-stakes applications (e.g., flagging malicious content), use the ROC curve to pick a threshold that sets the false positive rate you can tolerate, then re-evaluate precision and recall at that operating point.

Common Pitfalls and Fixes

ProblemSymptomFix
Catastrophic forgetting Training loss drops but validation loss spikes early Lower LR (1e-5), warmup steps, early stopping on val_f1
OOM on GPU CUDA out of memory error Reduce batch size; use gradient accumulation; FP16
Gradient explosion Loss becomes NaN Clip gradients at 1.0; reduce LR; check label encoding
Slow tokenization DataLoader bottleneck Pre-tokenize with .map(batched=True), cache to disk, use num_workers
Long reviews truncated Poor accuracy on verbose reviews Try beginning + end concatenation strategy; use Longformer for 4096+ tokens

Key Takeaways

  • Pre-tokenize the dataset with .map(batched=True) and cache — this eliminates the data loading bottleneck
  • Use DataCollatorWithPadding for dynamic padding — much more memory efficient than padding all sequences to max_length
  • Differential learning rates (smaller for BERT backbone, larger for head) preserve pre-trained representations
  • Linear warm-up + decay is the standard BERT scheduling recipe; 6% warm-up steps is a common starting point
  • Mixed precision (FP16) roughly halves GPU memory usage and speeds up training 1.5-2x with no accuracy loss
  • Always evaluate with F1, not just accuracy, especially on tasks with potential class imbalance
  • Error analysis on high-confidence misclassifications surfaces systematic model failures (sarcasm, negation, long-range dependencies)
