Hands-On: Security NLP Pipeline

End-to-end threat intelligence pipeline: ingest CVE text, preprocess, extract entities, classify severity, summarize, and serve via RAG search.

Learning Objectives

  • Chain preprocessing, NER, classification, summarization, and retrieval into a single pipeline
  • Persist a FAISS index and load it for low-latency query serving
  • Build a structured threat intel record from unstructured CVE text
  • Evaluate each pipeline stage independently and identify failure modes
  • Expose the pipeline as a Flask API endpoint
Full Pipeline Architecture
Input: Raw CVE Description Text (NVD JSON feed) | Stage 1 — Preprocessing: normalize, tokenize, clean HTML/URLs | Stage 2 — NER: extract CVE IDs, products, vendors, attack types | Stage 3 — Classification: severity (Critical/High/Medium/Low), attack vector | Stage 4 — Summarization: 1-2 sentence BART summary for analyst digest | Stage 5 — Embedding: encode for RAG index (sentence-transformers) | Output: Structured Intel Record + FAISS Index Entry {cve_id, products, attack_type, severity, summary, embedding}

Phase 1 — Data Ingestion and Preprocessing

Phase 1
import requests
import json
import re
import html
from datetime import datetime

def fetch_nvd_cves(year: int = 2023, limit: int = 500) -> list:
    """
    Fetch CVE records from NVD 2.0 API.
    Returns list of dicts: {cve_id, description, published, cvss_score}.
    """
    base_url = "https://services.nvd.nist.gov/rest/json/cves/2.0"
    params = {
        'pubStartDate': f'{year}-01-01T00:00:00.000',
        'pubEndDate':   f'{year}-12-31T23:59:59.999',
        'resultsPerPage': limit
    }
    r = requests.get(base_url, params=params, timeout=30)
    r.raise_for_status()
    data = r.json()

    records = []
    for item in data.get('vulnerabilities', []):
        cve = item['cve']
        desc = next(
            (d['value'] for d in cve.get('descriptions', []) if d['lang'] == 'en'),
            ''
        )
        metrics = cve.get('metrics', {})
        cvss_score = None
        if 'cvssMetricV31' in metrics:
            cvss_score = metrics['cvssMetricV31'][0]['cvssData']['baseScore']

        records.append({
            'cve_id':      cve['id'],
            'description': html.unescape(desc),
            'published':   cve['published'],
            'cvss_score':  cvss_score
        })
    return records


def preprocess_cve(text: str) -> str:
    """Normalize CVE description text for downstream NLP."""
    text = html.unescape(text)
    text = re.sub(r'<[^>]+>', ' ', text)
    text = re.sub(r'\s+', ' ', text).strip()
    return text

import spacy
from spacy.pipeline import EntityRuler

nlp = spacy.load('en_core_web_sm')
ruler = nlp.add_pipe('entity_ruler', before='ner')
ruler.add_patterns([
    {'label': 'CVE',       'pattern': [{'TEXT': {'REGEX': r'CVE-\d{4}-\d{4,}'}}]},
    {'label': 'CWE',       'pattern': [{'TEXT': {'REGEX': r'CWE-\d+'}}]},
    {'label': 'ATTACK_TYPE', 'pattern': [{'LOWER': 'sql'}, {'LOWER': 'injection'}]},
    {'label': 'ATTACK_TYPE', 'pattern': [{'LOWER': 'buffer'}, {'LOWER': 'overflow'}]},
    {'label': 'ATTACK_TYPE', 'pattern': [{'LOWER': 'use-after-free'}]},
    {'label': 'ATTACK_TYPE', 'pattern': [{'LOWER': 'cross-site'}, {'LOWER': 'scripting'}]},
    {'label': 'ATTACK_TYPE', 'pattern': [{'LOWER': 'path'}, {'LOWER': 'traversal'}]},
])

def extract_entities(text: str) -> dict:
    doc = nlp(text)
    entities = {'CVE': [], 'CWE': [], 'ATTACK_TYPE': [], 'ORG': [], 'PRODUCT': []}
    for ent in doc.ents:
        if ent.label_ in entities:
            entities[ent.label_].append(ent.text)
    return entities

Phase 3 — Severity Classification

Phase 3
def classify_severity(cvss_score: float) -> str:
    """Map CVSS 3.x base score to severity label per NVD convention."""
    if cvss_score is None:
        return 'Unknown'
    if cvss_score >= 9.0: return 'Critical'
    if cvss_score >= 7.0: return 'High'
    if cvss_score >= 4.0: return 'Medium'
    if cvss_score >  0.0: return 'Low'
    return 'None'


# Text-based severity classification when CVSS is unavailable
from transformers import pipeline

zsc = pipeline('zero-shot-classification', model='facebook/bart-large-mnli')

def classify_attack_vector(description: str) -> str:
    """Classify the primary attack vector using zero-shot classification."""
    labels = ['remote code execution', 'denial of service', 'information disclosure',
              'privilege escalation', 'authentication bypass', 'data tampering']
    result = zsc(description[:400], candidate_labels=labels)
    return result['labels'][0]   # top label

Phase 4 — Summarization and Embedding

Phase 4 / 5
from transformers import pipeline
from sentence_transformers import SentenceTransformer
import faiss, numpy as np, pickle, os

summarizer = pipeline('summarization', model='facebook/bart-large-cnn')
encoder    = SentenceTransformer('all-MiniLM-L6-v2')

def process_record(record: dict) -> dict:
    """Run full NLP pipeline on a single CVE record."""
    text = preprocess_cve(record['description'])

    # Summarize (skip very short descriptions)
    if len(text.split()) > 40:
        summary_result = summarizer(text, max_length=60, min_length=20, do_sample=False)
        summary = summary_result[0]['summary_text']
    else:
        summary = text

    return {
        'cve_id':       record['cve_id'],
        'description':  text,
        'summary':      summary,
        'entities':     extract_entities(text),
        'severity':     classify_severity(record['cvss_score']),
        'attack_vector': classify_attack_vector(text),
        'cvss_score':   record['cvss_score'],
        'published':    record['published'],
    }


def build_intel_index(records: list, index_path: str = 'cve_index.faiss',
                       meta_path: str = 'cve_meta.pkl'):
    """Process all records, build FAISS index, persist to disk."""
    processed, embeddings = [], []

    for i, rec in enumerate(records):
        result = process_record(rec)
        processed.append(result)
        # Embed the summary for retrieval (shorter than full description)
        embeddings.append(result['summary'])
        if (i + 1) % 50 == 0:
            print(f"Processed {i+1}/{len(records)}")

    # Encode all summaries
    emb_matrix = encoder.encode(embeddings, batch_size=64, normalize_embeddings=True)

    # Build FAISS index
    index = faiss.IndexFlatIP(emb_matrix.shape[1])
    index.add(emb_matrix.astype(np.float32))

    # Persist
    faiss.write_index(index, index_path)
    with open(meta_path, 'wb') as f:
        pickle.dump(processed, f)

    print(f"Index built: {len(processed)} CVEs | {index_path}")
    return index, processed

Phase 5 — Flask API Serving

Phase 5
from flask import Flask, request, jsonify
import faiss, pickle

app = Flask(__name__)

# Load persisted index and metadata at startup
INDEX  = faiss.read_index('cve_index.faiss')
with open('cve_meta.pkl', 'rb') as f:
    META = pickle.load(f)


@app.route('/search', methods=['GET'])
def search():
    """
    Query: GET /search?q=buffer+overflow+linux+kernel&k=5&severity=High
    Returns top-k CVE records matching the query.
    """
    query    = request.args.get('q', '')
    k        = int(request.args.get('k', 5))
    severity = request.args.get('severity')   # optional filter

    if not query:
        return jsonify({'error': 'query required'}), 400

    q_emb   = encoder.encode([query], normalize_embeddings=True)
    scores, indices = INDEX.search(q_emb.astype(np.float32), k * 3)   # over-fetch for filter

    results = []
    for score, idx in zip(scores[0], indices[0]):
        rec = META[idx]
        if severity and rec['severity'] != severity:
            continue
        results.append({
            'cve_id':      rec['cve_id'],
            'severity':    rec['severity'],
            'cvss_score':  rec['cvss_score'],
            'attack_vector': rec['attack_vector'],
            'summary':     rec['summary'],
            'similarity':  float(score)
        })
        if len(results) >= k:
            break

    return jsonify({'query': query, 'results': results})


if __name__ == '__main__':
    app.run(debug=True, port=5000)

Key Takeaways

  • End-to-end pipelines chain preprocessing, NER, classification, summarization, and retrieval into a reusable system
  • CVSS-based severity classification is deterministic; text-based classification needs zero-shot NLI or fine-tuned models
  • FAISS enables sub-millisecond similarity search over tens of thousands of documents — persist the index for fast API serving
  • The bottleneck in CPU inference is typically the summarization model — batching and model quantization reduce it
  • Flask + FAISS provides a lightweight, productionizable serving layer — add auth, rate limiting, and HTTPS before public deployment
  • Each pipeline stage should be independently evaluated and benchmarked — failure modes compound across stages
