1. NER as Sequence Labeling

Named Entity Recognition identifies spans in text and classifies them into predefined categories: PERSON, ORGANIZATION, LOCATION, DATE, CVE, MALWARE_FAMILY, etc. The standard formulation treats NER as a token-level classification problem using IOB tagging.

IOB Tagging Scheme
Text: "Lazarus Group exploited CVE-2023-44487 via HTTP/2" IOB2: Lazarus B-ORG Group I-ORG exploited O CVE-2023-44487 B-CVE via O HTTP/2 B-PROTOCOL B = Beginning of entity span I = Inside entity span (continuation) O = Outside any entity BIOES extends this: S = Single-token entity E = End of multi-token entity -- richer encoding, slightly better for CRF

2. spaCy: Rule-Based and Statistical NER

import spacy
from spacy.pipeline import EntityRuler

nlp = spacy.load('en_core_web_sm')

# Add custom entity types with EntityRuler
ruler = nlp.add_pipe('entity_ruler', before='ner')
patterns = [
    # CVE pattern
    {'label': 'CVE',     'pattern': [{'TEXT': {'REGEX': r'CVE-\d{4}-\d{4,}'}}]},
    # CWE pattern
    {'label': 'CWE',     'pattern': [{'TEXT': {'REGEX': r'CWE-\d+'}}]},
    # Known malware families (phrase patterns)
    {'label': 'MALWARE', 'pattern': [{'LOWER': 'wannacry'}]},
    {'label': 'MALWARE', 'pattern': [{'LOWER': 'emotet'}]},
    {'label': 'MALWARE', 'pattern': [{'LOWER': 'cobalt'}, {'LOWER': 'strike'}]},
    # APT groups
    {'label': 'APT',     'pattern': [{'LOWER': 'lazarus'}, {'LOWER': 'group'}]},
    {'label': 'APT',     'pattern': [{'TEXT': {'REGEX': r'APT\d+'}}]},
]
ruler.add_patterns(patterns)

text = "Lazarus Group exploited CVE-2023-44487 to deploy WannaCry on 192.168.1.100"
doc  = nlp(text)
for ent in doc.ents:
    print(f"[{ent.label_:10s}] {ent.text}")

3. BERT for Token Classification (NER)

from transformers import AutoTokenizer, AutoModelForTokenClassification
import torch

# Define label set (IOB2 scheme)
LABELS = ['O', 'B-CVE', 'I-CVE', 'B-MALWARE', 'I-MALWARE',
          'B-APT', 'I-APT', 'B-TECHNIQUE', 'I-TECHNIQUE']
label2id = {l: i for i, l in enumerate(LABELS)}
id2label = {i: l for i, l in enumerate(LABELS)}

MODEL_NAME = 'bert-base-cased'   # cased: case matters for NER!
tokenizer  = AutoTokenizer.from_pretrained(MODEL_NAME)
model      = AutoModelForTokenClassification.from_pretrained(
    MODEL_NAME,
    num_labels=len(LABELS),
    id2label=id2label,
    label2id=label2id
)

def tokenize_and_align_labels(examples, tokenizer, label2id, max_len=128):
    """
    BERT uses WordPiece subword tokenization.
    A single word may split into multiple subword tokens.
    We only assign labels to the FIRST subword of each word;
    subsequent subwords get label -100 (ignored in loss).
    """
    tokenized = tokenizer(
        examples['tokens'],
        is_split_into_words=True,   # input is already tokenized by word
        truncation=True,
        max_length=max_len,
        padding='max_length'
    )
    aligned_labels = []
    for i, word_ids in enumerate(tokenized.word_ids(batch_index=0)):
        # word_ids maps subword token index -> original word index (None for special tokens)
        label_ids = []
        prev_word_id = None
        for word_id in word_ids:
            if word_id is None:
                label_ids.append(-100)          # special token
            elif word_id != prev_word_id:
                label_ids.append(label2id[examples['labels'][word_id]])
            else:
                label_ids.append(-100)          # continuation subword
            prev_word_id = word_id
        aligned_labels.append(label_ids)
    tokenized['labels'] = aligned_labels
    return tokenized

4. Evaluation: Entity-Level F1

NER evaluation is done at the entity span level, not token level. A prediction is correct only if both the span boundaries and the entity type exactly match the ground truth. The seqeval library implements this correctly. 

from seqeval.metrics import classification_report, f1_score
# pip install seqeval

# seqeval expects list of label sequences
true_labels = [
    ['O', 'B-APT', 'I-APT', 'O', 'B-CVE', 'O'],
    ['B-MALWARE', 'O', 'O']
]
pred_labels = [
    ['O', 'B-APT', 'I-APT', 'O', 'O', 'O'],  # missed the CVE
    ['B-MALWARE', 'O', 'O']                      # correct
]

print(classification_report(true_labels, pred_labels))
print(f"Overall F1: {f1_score(true_labels, pred_labels):.4f}")

# Using HuggingFace Trainer with seqeval compute_metrics
def compute_metrics(eval_pred):
    logits, labels = eval_pred
    predictions = logits.argmax(axis=-1)

    # Convert ids back to label strings, skip -100 padding
    true_seqs = [[id2label[l] for l in label_row if l != -100] for label_row in labels]
    pred_seqs = [[id2label[p] for p, l in zip(pred_row, label_row) if l != -100]
                 for pred_row, label_row in zip(predictions, labels)]

    results = classification_report(true_seqs, pred_seqs, output_dict=True)
    return {'f1': results['weighted avg']['f1-score'],
            'precision': results['weighted avg']['precision'],
            'recall': results['weighted avg']['recall']}

# Full inference pipeline using HuggingFace pipeline API
from transformers import pipeline

# Use a pre-trained security NER model (or your fine-tuned checkpoint)
ner_pipeline = pipeline(
    'token-classification',
    model='dslim/bert-base-NER',   # standard CONLL-2003 model for baseline

threat_report = """

Key Takeaways

  • NER is sequence labeling with IOB2 tags — B- marks entity start, I- marks continuation, O marks non-entity
  • Use cased BERT for NER — case information (Lazarus vs lazarus) is a critical feature
  • WordPiece subword tokenization requires label alignment: only the first subword of each word gets a real label
  • Evaluate with seqeval at entity span level — a partially correct span boundary counts as a miss
  • spaCy's EntityRuler provides fast, rule-based entity extraction as a complement to statistical models
  • Security NER requires domain-specific fine-tuning — general CONLL models miss CVEs, APT names, and malware families
