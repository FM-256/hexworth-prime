1. Classical Approach: TF-IDF + Linear Model

from sklearn.pipeline import Pipeline
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.linear_model import LogisticRegression
from sklearn.naive_bayes import MultinomialNB
from sklearn.metrics import classification_report, f1_score
from sklearn.model_selection import cross_val_score
import numpy as np

# TF-IDF + Logistic Regression (strong baseline for most classification tasks)
lr_pipeline = Pipeline([
    ('tfidf', TfidfVectorizer(
        ngram_range=(1, 2),
        max_features=50000,
        sublinear_tf=True,
        min_df=3,
        stop_words='english'
    )),
    ('clf', LogisticRegression(
        C=1.0,
        max_iter=1000,
        class_weight='balanced',  # handle imbalance
        solver='saga',            # fast for large feature sets
        n_jobs=-1
    ))
])

# 5-fold cross-validation

print(f"CV F1: {cv_scores.mean():.4f} +/- {cv_scores.std():.4f}")

2. Handling Class Imbalance

Class-Weighted Loss

Pass class_weight='balanced' to sklearn or compute torch.tensor(weights) for nn.CrossEntropyLoss. Penalizes errors on minority class more heavily.

Oversampling (SMOTE)

Synthetic Minority Over-sampling generates new samples interpolated between existing minority examples. Use imblearn.over_sampling.SMOTE on TF-IDF features.

Threshold Moving

Default threshold 0.5 optimizes accuracy. For imbalanced classes, lower the positive threshold to improve recall at the cost of precision. Tune on val set using ROC curve.

from sklearn.utils.class_weight import compute_class_weight
import torch

# Compute class weights for PyTorch loss
classes    = np.unique(y_train)
weights    = compute_class_weight('balanced', classes=classes, y=y_train)
class_w    = torch.tensor(weights, dtype=torch.float)
criterion  = torch.nn.CrossEntropyLoss(weight=class_w)

# Threshold optimization
from sklearn.metrics import roc_curve, f1_score

probs  = model.predict_proba(X_val)[:, 1]   # P(positive class)
fpr, tpr, thresholds = roc_curve(y_val, probs)

# Find threshold that maximizes F1 on validation set
best_thresh, best_f1 = 0.5, 0
for t in thresholds:
    preds_t = (probs >= t).astype(int)
    f = f1_score(y_val, preds_t)
    if f > best_f1:
        best_f1, best_thresh = f, t
print(f"Best threshold: {best_thresh:.3f} | Val F1: {best_f1:.4f}")

3. Multi-Label Classification

Some texts belong to multiple categories simultaneously. A CVE description might be tagged as both "injection" and "authentication bypass". Multi-label classification replaces the softmax output with independent sigmoid outputs — one per label. 

from transformers import AutoModelForSequenceClassification

# Multi-label BERT: output is num_labels independent sigmoid logits
model = AutoModelForSequenceClassification.from_pretrained(
    'bert-base-uncased',
    num_labels=10,               # 10 possible categories
    problem_type='multi_label_classification'
)
# HuggingFace automatically uses BCE loss when problem_type is set

# Inference: apply threshold to each label independently
def predict_multilabel(text: str, model, tokenizer,
                       label_names: list, threshold: float = 0.5) -> dict:
    encoding = tokenizer(text, return_tensors='pt', truncation=True, max_length=256)
    with torch.no_grad():
        logits = model(**encoding).logits
    probs   = logits.sigmoid()[0]
    results = {}
    for name, prob in zip(label_names, probs):
        if prob >= threshold:
            results[name] = prob.item()
    return results

# Multi-label evaluation: use sklearn with zero_division=0
from sklearn.metrics import multilabel_confusion_matrix, classification_report
# y_true, y_pred: [n_samples, n_labels] binary arrays
print(classification_report(y_true, y_pred, target_names=label_names, zero_division=0))

4. Zero-Shot Classification

Zero-shot classification uses NLI (Natural Language Inference) models to classify text into arbitrary categories without any task-specific fine-tuning. The model is asked: "Does this text entail that it is about [category]?" If the entailment score is high, the text belongs to that category. 

from transformers import pipeline

# facebook/bart-large-mnli is the standard zero-shot classifier
zsc = pipeline('zero-shot-classification', model='facebook/bart-large-mnli')

text       = "The attacker used spear-phishing emails to gain initial access to the network."
categories = ['initial access', 'exfiltration', 'persistence',
              'lateral movement', 'defense evasion']

result = zsc(text, candidate_labels=categories, multi_label=True)
for label, score in zip(result['labels'], result['scores']):
    print(f"{label:25s}  {score:.3f}")

# Dynamic categories: no retraining needed when you add new MITRE ATT&CK techniques
mitre_tactics = [
    'reconnaissance', 'resource development', 'initial access',
    'execution', 'persistence', 'privilege escalation',
    'defense evasion', 'credential access', 'discovery',
    'lateral movement', 'collection', 'command and control',
    'exfiltration', 'impact'
]
result = zsc(text, candidate_labels=mitre_tactics)

Key Takeaways

  • TF-IDF + logistic regression is a strong and fast baseline that should always be tried before BERT
  • Naive Bayes assumes feature independence but works surprisingly well for bag-of-words text features
  • Class imbalance: handle with class-weighted loss, threshold tuning, or oversampling — rarely with undersampling
  • Multi-label classification uses sigmoid outputs + binary cross-entropy — each label is an independent binary decision
  • Zero-shot classification via NLI requires no labeled data and supports dynamic label sets — ideal for MITRE ATT&CK mapping
  • BERT fine-tuned on even 200-500 labeled examples typically outperforms all classical approaches on structured NLP tasks
