Project Overview

This module walks through building a complete image classifier on CIFAR-10 (10 classes: airplane, automobile, bird, cat, deer, dog, frog, horse, ship, truck) from scratch using PyTorch. The target is achieving >85% test accuracy. We will then swap to a pretrained ResNet-18 and demonstrate transfer learning achieving >93%.

Step 1: Environment and Data

Python -- Setup and data loading
import torch
import torch.nn as nn
import torch.optim as optim
import torch.nn.functional as F
from torchvision import datasets, transforms, models
from torch.utils.data import DataLoader
import numpy as np

device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
print(f"Training on: {device}")

# CIFAR-10 statistics (precomputed over training set)
MEAN = [0.4914, 0.4822, 0.4465]
STD  = [0.2023, 0.1994, 0.2010]
CLASSES = ('plane', 'car', 'bird', 'cat', 'deer',
           'dog', 'frog', 'horse', 'ship', 'truck')

train_transform = transforms.Compose([
    transforms.RandomCrop(32, padding=4),
    transforms.RandomHorizontalFlip(),
    transforms.ColorJitter(brightness=0.1, contrast=0.1),
    transforms.ToTensor(),
    transforms.Normalize(MEAN, STD)
])

test_transform = transforms.Compose([
    transforms.ToTensor(),
    transforms.Normalize(MEAN, STD)
])

train_set = datasets.CIFAR10('./data', train=True,  download=True, transform=train_transform)
test_set  = datasets.CIFAR10('./data', train=False, download=True, transform=test_transform)

train_loader = DataLoader(train_set, batch_size=128, shuffle=True,  num_workers=4, pin_memory=True)
test_loader  = DataLoader(test_set,  batch_size=256, shuffle=False, num_workers=4, pin_memory=True)
print(f"Train: {len(train_set)} | Test: {len(test_set)}")
# Train: 50000 | Test: 10000

Step 2: Model Architecture

Python -- Custom CNN for CIFAR-10
class CIFAR10CNN(nn.Module):
    """
    3-block CNN for CIFAR-10 (32x32 input).
    Target: ~85% test accuracy from scratch.
    """
    def __init__(self):
        super().__init__()
        self.features = nn.Sequential(
            # Block 1: 32x32 -> 32x32 -> 16x16
            nn.Conv2d(3, 64, 3, padding=1),   nn.BatchNorm2d(64),  nn.ReLU(True),
            nn.Conv2d(64, 64, 3, padding=1),  nn.BatchNorm2d(64),  nn.ReLU(True),
            nn.MaxPool2d(2, 2), nn.Dropout2d(0.1),

            # Block 2: 16x16 -> 16x16 -> 8x8
            nn.Conv2d(64, 128, 3, padding=1),  nn.BatchNorm2d(128), nn.ReLU(True),
            nn.Conv2d(128, 128, 3, padding=1), nn.BatchNorm2d(128), nn.ReLU(True),
            nn.MaxPool2d(2, 2), nn.Dropout2d(0.15),

            # Block 3: 8x8 -> 8x8 -> 4x4
            nn.Conv2d(128, 256, 3, padding=1),  nn.BatchNorm2d(256), nn.ReLU(True),
            nn.Conv2d(256, 256, 3, padding=1), nn.BatchNorm2d(256), nn.ReLU(True),
            nn.MaxPool2d(2, 2),
        )
        self.classifier = nn.Sequential(
            nn.Flatten(),                    # 4*4*256 = 4096
            nn.Linear(4096, 512), nn.ReLU(True), nn.Dropout(0.5),
            nn.Linear(512, 10)
        )

    def forward(self, x):
        return self.classifier(self.features(x))

model = CIFAR10CNN().to(device)
print(f"Parameters: {sum(p.numel() for p in model.parameters()):,}")
# Parameters: 4,369,674

Step 3: Training Loop

Python -- training and evaluation functions
optimizer = optim.AdamW(model.parameters(), lr=1e-3, weight_decay=1e-4)
scheduler = optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=100)
criterion = nn.CrossEntropyLoss(label_smoothing=0.1)

def train_epoch(model, loader, optimizer, criterion):
    model.train()
    total_loss, correct, total = 0, 0, 0
    for images, labels in loader:
        images, labels = images.to(device), labels.to(device)
        optimizer.zero_grad()
        outputs = model(images)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        total_loss += loss.item() * images.size(0)
        correct += (outputs.argmax(1) == labels).sum().item()
        total += images.size(0)
    return total_loss / total, correct / total

def evaluate(model, loader, criterion):
    model.eval()
    total_loss, correct, total = 0, 0, 0
    with torch.no_grad():
        for images, labels in loader:
            images, labels = images.to(device), labels.to(device)
            outputs = model(images)
            total_loss += criterion(outputs, labels).item() * images.size(0)
            correct += (outputs.argmax(1) == labels).sum().item()
            total += images.size(0)
    return total_loss / total, correct / total

# Training loop
best_acc = 0.0
for epoch in range(1, 101):
    train_loss, train_acc = train_epoch(model, train_loader, optimizer, criterion)
    test_loss,  test_acc  = evaluate(model, test_loader, criterion)
    scheduler.step()
    if test_acc > best_acc:
        best_acc = test_acc
        torch.save(model.state_dict(), 'best_cifar10_cnn.pth')
    if epoch % 10 == 0:
        print(f"Epoch {epoch:3d} | Train: {train_acc:.3f} | Test: {test_acc:.3f} | Best: {best_acc:.3f}")
# Epoch  10 | Train: 0.734 | Test: 0.723 | Best: 0.724
# Epoch  50 | Train: 0.871 | Test: 0.841 | Best: 0.843
# Epoch 100 | Train: 0.913 | Test: 0.862 | Best: 0.866

Step 4: Transfer Learning with ResNet-18

Python -- fine-tune pretrained ResNet-18 on CIFAR-10
# Load pretrained ResNet-18
resnet = models.resnet18(weights=models.ResNet18_Weights.IMAGENET1K_V1)

# Adapt for CIFAR-10: replace first conv (designed for 224x224) and FC
resnet.conv1 = nn.Conv2d(3, 64, kernel_size=3, stride=1, padding=1, bias=False)
resnet.maxpool = nn.Identity()    # remove early maxpool (would shrink 32x32 too fast)
resnet.fc = nn.Linear(512, 10)  # 10 CIFAR-10 classes
resnet = resnet.to(device)

# Fine-tune all layers (full fine-tuning, not feature extraction)
optimizer = optim.SGD(resnet.parameters(), lr=0.05,
                      momentum=0.9, weight_decay=5e-4, nesterov=True)
scheduler = optim.lr_scheduler.OneCycleLR(
    optimizer, max_lr=0.1, epochs=50, steps_per_epoch=len(train_loader)
)

# 50 epochs with OneCycleLR typically achieves ~93% test accuracy
# vs ~86% from scratch with the custom CNN above
# Epoch  50 | Train: 0.967 | Test: 0.934 | Best: 0.936

Step 5: Evaluation and Diagnostics

Python -- confusion matrix and per-class accuracy
from sklearn.metrics import confusion_matrix
import matplotlib.pyplot as plt
import seaborn as sns

model.eval()
all_preds, all_labels = [], []
with torch.no_grad():
    for images, labels in test_loader:
        outputs = model(images.to(device))
        preds = outputs.argmax(1).cpu()
        all_preds.extend(preds.tolist())
        all_labels.extend(labels.tolist())

cm = confusion_matrix(all_labels, all_preds)

# Per-class accuracy
for i, cls in enumerate(CLASSES):
    acc = cm[i, i] / cm[i].sum()

Interpreting the Confusion Matrix

Cat has the lowest per-class accuracy (83.1%). The confusion matrix typically shows cats confused with dogs (similar body shapes) and deer confused with horses. This reflects that CNNs learn texture and shape features that are similar between visually similar classes. Solutions: more augmentation for hard classes, label-aware sampling, or focal loss to up-weight hard examples.

Extension Challenges

  1. Add MixUp data augmentation (alpha=0.4) to the training pipeline and measure the accuracy change. Does MixUp help more with the hard classes (cat, dog)?
  2. Apply test-time augmentation (TTA): at inference, run 5 augmented versions of each test image and average their predictions. By how many percentage points does this improve accuracy?
  3. Implement learning rate warmup: linearly increase LR from 0 to max_lr over the first 5 epochs before starting cosine decay. Compare final accuracy with and without warmup.
  4. Replace the CrossEntropyLoss with a custom focal loss (FL(p) = -alpha * (1-p)^gamma * log(p)) and find the best gamma value on CIFAR-10 by grid search over [0.5, 1.0, 2.0].
  5. Package the trained model for inference: convert to TorchScript using torch.jit.trace, then measure inference latency on CPU for batch sizes 1, 16, 64.

Key Takeaways

  • Full training pipeline: data loading with augmentation, model definition, optimizer + scheduler, eval loop
  • AdamW with cosine LR schedule and label smoothing is a strong default for training from scratch
  • Transfer learning (ResNet-18 pretrained) achieves ~93% vs ~86% from scratch with fewer epochs
  • Confusion matrix reveals which class pairs are hardest -- informs augmentation and loss strategies
  • Cosine annealing with warm restarts often outperforms fixed LR or step decay schedules
  • Save best checkpoint by validation accuracy, not final epoch -- training might overfit at the end
