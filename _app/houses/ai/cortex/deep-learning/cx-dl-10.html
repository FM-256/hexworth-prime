Tensors

A PyTorch tensor is an N-dimensional array analogous to NumPy ndarray, but with GPU support and autograd integration. Understanding dtype and device is essential for avoiding silent precision loss and illegal cross-device operations.

tensors.py
import torch

# Creation
a = torch.zeros(3, 4)                          # float32, CPU
b = torch.ones(3, 4, dtype=torch.float16)   # float16
c = torch.arange(0, 10, 2)                   # [0,2,4,6,8]
r = torch.randn(2, 3)                         # standard normal

# Shape operations
print(r.shape)              # torch.Size([2, 3])
r_T = r.T                  # transpose
r_flat = r.view(-1)        # flatten to 1D
r_3d = r.unsqueeze(0)      # add batch dim -> [1, 2, 3]

# Device transfer
device = 'cuda' if torch.cuda.is_available() else 'cpu'
r = r.to(device)

# Math
x = torch.tensor([1.0, 2.0, 3.0])
y = torch.tensor([4.0, 5.0, 6.0])
print(x @ y)                # dot product: tensor(32.)
print(torch.stack([x, y]))  # shape [2, 3]

Autograd

PyTorch builds a dynamic computation graph on every forward pass. When .backward() is called on a scalar loss, gradients are accumulated into .grad for every leaf tensor with requires_grad=True. Call optimizer.zero_grad() before each step to clear stale gradients.

autograd.py
w = torch.tensor(1.0, requires_grad=True)
x = torch.tensor(2.0)

# Graph is built during forward
loss = (w * x - 3) ** 2
loss.backward()
print(w.grad)    # tensor(-2.)  dL/dw = 2*(w*x-3)*x = 2*(2-3)*2 = -4... actually -2

# Disable grad tracking for inference
with torch.no_grad():
    pred = w * x
print(pred.requires_grad)  # False

# Detach from graph (e.g., logging metrics)
loss_val = loss.detach().item()  # float, no grad

Custom Dataset and DataLoader

dataset.py
from torch.utils.data import Dataset, DataLoader
import numpy as np

class TabularDataset(Dataset):
    def __init__(self, X, y):
        self.X = torch.tensor(X, dtype=torch.float32)
        self.y = torch.tensor(y, dtype=torch.long)

    def __len__(self):
        return len(self.y)

    def __getitem__(self, idx):
        return self.X[idx], self.y[idx]

# Usage
X_train = np.random.randn(1000, 20)
y_train = np.random.randint(0, 5, 1000)
train_ds = TabularDataset(X_train, y_train)
train_loader = DataLoader(
    train_ds,
    batch_size=64,
    shuffle=True,
    num_workers=4,      # parallel data loading
    pin_memory=True    # faster CPU->GPU transfer
)

Complete Training Loop

training_loop.py
import torch
import torch.nn as nn

device = 'cuda' if torch.cuda.is_available() else 'cpu'
model = MyModel().to(device)
optimizer = torch.optim.AdamW(model.parameters(), lr=1e-3, weight_decay=0.01)
criterion = nn.CrossEntropyLoss()
best_val_loss = float('inf')

for epoch in range(50):
    # --- Train ---
    model.train()
    train_loss = 0.0
    for X, y in train_loader:
        X, y = X.to(device), y.to(device)
        optimizer.zero_grad()
        loss = criterion(model(X), y)
        loss.backward()
        nn.utils.clip_grad_norm_(model.parameters(), 1.0)
        optimizer.step()
        train_loss += loss.item()

    # --- Validate ---
    model.eval()
    val_loss = 0.0
    correct = 0
    with torch.no_grad():
        for X, y in val_loader:
            X, y = X.to(device), y.to(device)
            logits = model(X)
            val_loss += criterion(logits, y).item()
            correct += (logits.argmax(1) == y).sum().item()

    avg_train = train_loss / len(train_loader)
    avg_val   = val_loss   / len(val_loader)
    acc = correct / len(val_loader.dataset)
    print(f"Epoch {epoch:3d}  train={avg_train:.4f}  val={avg_val:.4f}  acc={acc:.3f}")

    # Save best checkpoint
    if avg_val < best_val_loss:
        best_val_loss = avg_val
        torch.save(model.state_dict(), 'best_model.pt')
Loading a checkpoint

To restore the best model: model.load_state_dict(torch.load('best_model.pt', map_location=device)). Always pass map_location to handle cases where checkpoint was saved on GPU but loaded on CPU.

Key Takeaways

  • Tensors live on CPU or GPU — all operands must be on the same device before math operations
  • Autograd builds the computation graph dynamically; .backward() flows gradients to all requires_grad leaves
  • Custom Dataset requires __len__ and __getitem__; DataLoader handles batching, shuffling, and parallel loading
  • Always call model.train() before training and model.eval() before validation
  • Save the best model by val loss, not the last epoch — the last epoch is often overfit
