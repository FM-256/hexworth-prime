    Task 1 — Basic for loop
    for i in 1 2 3 4 5; do echo $i; done
    Task 2 — Brace expansion loop
    for i in {1..5}; do echo $i; done
    Task 3 — C-style for loop
    for ((i=0; i<5; i++)); do echo $i; done
    Task 4 — While loop
    i=1; while [ $i -le 3 ]; do echo $i; i=$((i+1)); done
    Task 5 — Until loop
    i=1; until [ $i -gt 3 ]; do echo $i; i=$((i+1)); done
    Task 6 — Loop over files
    for f in /etc/*.conf; do echo $f; done
    Task 7 — seq range
    for i in $(seq 1 5); do echo $i; done
    Task 8 — Break in loop
    for i in {1..10}; do if [ $i -eq 5 ]; then break; fi; echo $i; done
