Problems 1-5: Foundational Patterns

PythonProblem 1: Valid Parentheses (LeetCode #20)
# Pattern: Stack. O(n) time, O(n) space.
def is_valid(s):
    stack = []; mapping = {')':'(', '}':'{}'[0], ']':'['}
    for c in s:
        if c in mapping:
            if not stack or stack[-1] != mapping[c]: return False
            stack.pop()
        else: stack.append(c)
    return not stack
PythonProblem 2: Max Profit (LeetCode #121)
# Pattern: Greedy, track min price. O(n) time, O(1) space.
def max_profit(prices):
    min_p = float('inf'); best = 0
    for p in prices:
        min_p = min(min_p, p)
        best = max(best, p - min_p)
    return best
PythonProblem 3: Climbing Stairs (LeetCode #70)
# Pattern: DP (Fibonacci). O(n) time, O(1) space.
def climb_stairs(n):
    a, b = 1, 1
    for _ in range(2, n+1): a, b = b, a+b
    return b
PythonProblem 4: Number of Islands (LeetCode #200)
# Pattern: BFS/DFS component count. O(mn) time and space.
def num_islands(grid):
    count = 0
    def dfs(r,c):
        if r<0 or r>=len(grid) or c<0 or c>=len(grid[0]) or grid[r][c]!='1': return
        grid[r][c]='0'; dfs(r+1,c); dfs(r-1,c); dfs(r,c+1); dfs(r,c-1)
    for r in range(len(grid)):
        for c in range(len(grid[0])):
            if grid[r][c]=='1': dfs(r,c); count+=1
    return count
PythonProblem 5: Product of Array Except Self (LeetCode #238)
# Pattern: Prefix + suffix products. O(n) time, O(1) extra space.
def product_except_self(nums):
    n = len(nums); res = [1]*n
    prefix = 1
    for i in range(n): res[i] = prefix; prefix *= nums[i]
    suffix = 1
    for i in range(n-1,-1,-1): res[i] *= suffix; suffix *= nums[i]
    return res

PythonProblem 6: Word Break (LeetCode #139) — DP
def word_break(s, wordDict):
    words = set(wordDict); n = len(s)
    dp = [False]*(n+1); dp[0] = True
    for i in range(1,n+1):
        for j in range(i):
            if dp[j] and s[j:i] in words: dp[i]=True; break
    return dp[n]
PythonProblem 7: Find Median from Data Stream (LeetCode #295)
import heapq

class MedianFinder:
    def __init__(self): self.lo=[]; self.hi=[]
    def addNum(self, n):
        heapq.heappush(self.lo, -n)
        heapq.heappush(self.hi, -heapq.heappop(self.lo))
        if len(self.hi)>len(self.lo): heapq.heappush(self.lo,-heapq.heappop(self.hi))
    def findMedian(self):
        if len(self.lo)>len(self.hi): return float(-self.lo[0])
        return (-self.lo[0]+self.hi[0])/2
PythonProblem 8: Merge K Sorted Lists (LeetCode #23)
import heapq

def merge_k_lists(lists):
    heap = []
    for i, node in enumerate(lists):
        if node: heapq.heappush(heap, (node.val, i, node))
    dummy = cur = type('N',(),{'val':0,'next':None})()
    while heap:
        _, i, node = heapq.heappop(heap)
        cur.next = node; cur = cur.next
        if node.next: heapq.heappush(heap, (node.next.val, i, node.next))
    return dummy.next
# O(N log k) where N = total nodes, k = number of lists

PythonProblem 10: Trapping Rain Water (LeetCode #42)
def trap(height):
    l, r = 0, len(height)-1
    lmax, rmax = 0, 0; water = 0
    while l < r:
        if height[l] < height[r]:
            lmax = max(lmax, height[l])
            water += lmax - height[l]; l+=1
        else:
            rmax = max(rmax, height[r])
            water += rmax - height[r]; r-=1
    return water
# O(n) two-pointer — no extra space needed

Key Takeaways

  • Pattern recognition is the core skill — identify stack/DP/greedy/graph before coding.
  • Always estimate complexity before coding; reject O(n^2) for n=10^5 inputs immediately.
  • Two-heap pattern (MedianFinder) appears in many streaming/sliding window problems.
  • Two-pointer on sorted data is always O(n) — look for this optimization on sorted inputs.
