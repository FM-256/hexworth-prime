PythonBFS shortest path (LeetCode #994 Rotting Oranges style)
from collections import deque

def bfs_dist(grid, starts):
    rows, cols = len(grid), len(grid[0])
    dist = [[-1]*cols for _ in range(rows)]
    q = deque()
    for r,c in starts:
        dist[r][c] = 0; q.append((r,c))
    dirs = [(0,1),(0,-1),(1,0),(-1,0)]
    while q:
        r,c = q.popleft()
        for dr,dc in dirs:
            nr,nc = r+dr,c+dc
            if 0<=nrand 0<=ncand grid[nr][nc]==1 and dist[nr][nc]==-1:
                dist[nr][nc] = dist[r][c]+1; q.append((nr,nc))
    return dist
PythonDijkstra (LeetCode #743 Network Delay Time)
import heapq

def network_delay(times, n, k):
    graph = [[] for _ in range(n+1)]
    for u,v,w in times: graph[u].append((w,v))
    dist = [float('inf')]*(n+1); dist[k]=0
    pq = [(0,k)]
    while pq:
        d,u = heapq.heappop(pq)
        if d > dist[u]: continue
        for w,v in graph[u]:
            if dist[u]+w < dist[v]:
                dist[v]=dist[u]+w; heapq.heappush(pq,(dist[v],v))
    ans = max(dist[1:])
    return ans if ans < float('inf') else -1

Topological Sort

PythonCourse Schedule (LeetCode #207) — Cycle detection via DFS
def can_finish(n, prerequisites):
    graph = [[] for _ in range(n)]
    for a,b in prerequisites: graph[b].append(a)
    state = [0]*n  # 0=unvisited, 1=visiting, 2=done
    def dfs(u):
        if state[u]==1: return False  # cycle
        if state[u]==2: return True
        state[u]=1
        for v in graph[u]:
            if not dfs(v): return False
        state[u]=2; return True
    return all(dfs(i) for i in range(n) if state[i]==0)

Key Takeaways

  • BFS on unweighted graph gives shortest path in O(V+E).
  • Dijkstra with binary heap: O((V+E) log V) — standard for weighted shortest paths.
  • Topological sort: DFS with 3-state marking detects cycles and produces topo order.
  • Grid graphs: treat cells as nodes, directions as edges — standard BFS/DFS applies.
