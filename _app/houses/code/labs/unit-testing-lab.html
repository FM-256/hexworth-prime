The getMax(arr) function should return the largest number in an array. Write comprehensive tests covering normal cases, edge cases, and error cases.
📦 Function to Test
function getMax(arr) { if (!Array.isArray(arr)) { throw new Error("Input must be an array"); } if (arr.length === 0) { throw new Error("Array cannot be empty"); } return Math.max(...arr); }
✏️ Write Comprehensive Tests
📋 Test Output
Click "Run Test" to see results...
💡 Need a hint? Click to reveal

Cover multiple scenarios:

// Normal
expect(getMax([1, 5, 3])).toBe(5);
// Single element
expect(getMax([42])).toBe(42);
// Negatives
expect(getMax([-5, -1, -10])).toBe(-1);
// Empty - should throw
expectToThrow(() => getMax([]), "Array cannot be empty");
// Not array - should throw
expectToThrow(() => getMax("hello"), "Input must be an array");

📋 Test Output
Click "Run Test" to see results...
💡 Need a hint? Click to reveal

Think about all the edge cases:

// Valid emails
expect(isValidEmail("user@example.com")).toBe(true);
expect(isValidEmail("test.name@domain.org")).toBe(true);

// Invalid - missing parts
expect(isValidEmail("userexample.com")).toBe(false);
expect(isValidEmail("user@")).toBe(false);
expect(isValidEmail("@domain.com")).toBe(false);

// Edge cases
expect(isValidEmail(null)).toBe(false);
expect(isValidEmail(undefined)).toBe(false);
expect(isValidEmail(12345)).toBe(false);
expect(isValidEmail("")).toBe(false);
