Before you can visualize data, you need to load it cleanly. The csv module's DictReader turns CSV rows into dictionaries, making column access intuitive. No pandas required.

Loading CSV with DictReader

import csv

def load_csv(filepath):
    """Load a CSV file into a list of dicts."""
    with open(filepath, "r") as f:
        reader = csv.DictReader(f)
        data = []
        for row in reader:
            # Convert numeric strings to numbers
            cleaned = {}
            for key, value in row.items():
                try:
                    cleaned[key] = float(value)
                except ValueError:
                    cleaned[key] = value
            data.append(cleaned)
    return data

Loading JSON

import json

def load_json(filepath):
    """Load a JSON file (expects list of objects)."""
    with open(filepath, "r") as f:
        return json.load(f)

Auto-Detecting Numeric Columns

def get_numeric_columns(data):
    """Return column names that contain numeric data."""
    if not data:
        return []
    return [k for k, v in data[0].items()
            if isinstance(v, (int, float))]
Tip: Type Coercion

CSV files store everything as strings. Always try to convert values to float or int during loading. The try/except pattern catches non-numeric columns cleanly without crashing.

Statistical Summary

Python's statistics module provides mean, median, mode, and standard deviation without external dependencies. Combined with our data loader, we can compute summaries for any numeric column.

The statistics Module

import statistics

def summarize_column(data, column):
    """Compute summary statistics for a numeric column."""
    values = [row[column] for row in data if isinstance(row.get(column), (int, float))]
    if not values:
        return {}
    return {
        "count": len(values),
        "mean": round(statistics.mean(values), 2),
        "median": round(statistics.median(values), 2),
        "stdev": round(statistics.stdev(values), 2) if len(values) > 1 else 0,
        "min": min(values),
        "max": max(values),
        "sum": round(sum(values), 2),
    }
statistics vs numpy

The statistics module is slower than numpy but requires no installation. For dashboards with under 100K rows, it is more than fast enough. Reserve numpy for heavy numerical work.

Chart Creation

Matplotlib is the most widely used charting library in Python. We will create bar charts, line charts, and pie charts from our loaded data.

Bar Chart

import matplotlib.pyplot as plt

def create_bar_chart(labels, values, title, xlabel, ylabel):
    fig, ax = plt.subplots(figsize=(8, 5))
    ax.bar(labels, values, color="#4b8bbe")
    ax.set_title(title)
    ax.set_xlabel(xlabel)
    ax.set_ylabel(ylabel)
    ax.grid(axis="y", alpha=0.3)
    return fig

Line Chart

def create_line_chart(x, y, title, xlabel, ylabel):
    fig, ax = plt.subplots(figsize=(8, 5))
    ax.plot(x, y, marker="o", color="#4ade80", linewidth=2)
    ax.fill_between(x, y, alpha=0.1, color="#4ade80")
    ax.set_title(title)
    ax.set_xlabel(xlabel)
    ax.set_ylabel(ylabel)
    ax.grid(alpha=0.3)
    return fig

Pie Chart

def create_pie_chart(labels, values, title):
    fig, ax = plt.subplots(figsize=(6, 6))
    ax.pie(values, labels=labels, autopct="%1.1f%%",
           colors=["#4b8bbe", "#4ade80", "#ffd93d", "#ff6b9d"])
    ax.set_title(title)
    return fig

Dashboard Layout

A real dashboard shows multiple charts at once. Matplotlib's GridSpec lets you arrange subplots in a flexible grid, creating a professional multi-panel layout.

GridSpec Layout

from matplotlib.gridspec import GridSpec

def create_dashboard(data):
    fig = plt.figure(figsize=(14, 10))
    fig.suptitle("Sales Dashboard", fontsize=16, fontweight="bold")
    gs = GridSpec(2, 2, figure=fig, hspace=0.3, wspace=0.3)

    # Top-left: Bar chart
    ax1 = fig.add_subplot(gs[0, 0])
    ax1.bar(data["months"], data["sales"], color="#4b8bbe")
    ax1.set_title("Monthly Sales")

    # Top-right: Line chart
    ax2 = fig.add_subplot(gs[0, 1])
    ax2.plot(data["months"], data["sales"], "o-", color="#4ade80")
    ax2.set_title("Sales Trend")

    # Bottom-left: Pie chart
    ax3 = fig.add_subplot(gs[1, 0])
    ax3.pie(data["sales"], labels=data["months"], autopct="%1.0f%%")
    ax3.set_title("Distribution")

    # Bottom-right: Stats table
    ax4 = fig.add_subplot(gs[1, 1])
    ax4.axis("off")
    stats_text = f"Total: ${sum(data['sales']):,.0f}\nAvg: ${statistics.mean(data['sales']):,.0f}"
    ax4.text(0.5, 0.5, stats_text, transform=ax4.transAxes,
             ha="center", va="center", fontsize=14)
    ax4.set_title("Summary")

    return fig

tkinter Integration

To turn a matplotlib figure into a desktop application, embed it in a tkinter window using FigureCanvasTkAgg. This gives you buttons, menus, and interactive controls alongside your charts.

Embedding matplotlib in tkinter

import tkinter as tk
from matplotlib.backends.backend_tkagg import FigureCanvasTkAgg, NavigationToolbar2Tk

def build_app():
    root = tk.Tk()
    root.title("Data Dashboard")
    root.geometry("1200x800")

    # Create the dashboard figure
    fig = create_dashboard(sample_data)

    # Embed in tkinter
    canvas = FigureCanvasTkAgg(fig, master=root)
    canvas.draw()
    canvas.get_tk_widget().pack(fill=tk.BOTH, expand=True)

    # Add navigation toolbar (zoom, pan, save)
    toolbar = NavigationToolbar2Tk(canvas, root)
    toolbar.update()

    root.mainloop()
Tip: FigureCanvasTkAgg

FigureCanvasTkAgg bridges matplotlib and tkinter. Call canvas.draw() after creating or updating the figure. Use canvas.get_tk_widget() to get the tkinter widget for packing into your layout.
