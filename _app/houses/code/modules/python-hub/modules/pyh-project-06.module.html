The Game Loop

def draw_grid(screen):
    """Draw a grid background."""
    for x in range(0, WINDOW_WIDTH, CELL_SIZE):
        pygame.draw.line(screen, GRAY, (x, 0), (x, WINDOW_HEIGHT))
    for y in range(0, WINDOW_HEIGHT, CELL_SIZE):
        pygame.draw.line(screen, GRAY, (0, y), (WINDOW_WIDTH, y))

def game_loop(screen, clock):
    running = True
    while running:
        # 1. Process events
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                running = False

        # 2. Update game state
        # (coming in next sections)

        # 3. Render
        screen.fill(BLACK)
        draw_grid(screen)
        pygame.display.flip()
        clock.tick(FPS)

    pygame.quit()
    sys.exit()
Tip: FPS and Game Speed

clock.tick(FPS) limits the loop to N frames per second. For Snake, 10 FPS gives a classic feel. We will increase FPS as the score rises to increase difficulty. Use CELL_SIZE to align everything to a grid.

Snake Movement

The snake is represented as a deque of (x, y) grid positions. A deque (double-ended queue) is perfect because we add to the front (head moves) and remove from the back (tail follows), both in O(1) time.

Snake Data Structure

from collections import deque

# Direction vectors (dx, dy)
UP = (0, -1)
DOWN = (0, 1)
LEFT = (-1, 0)
RIGHT = (1, 0)

# Initial snake: 3 segments in the center
snake = deque()
start_x = WINDOW_WIDTH // CELL_SIZE // 2
start_y = WINDOW_HEIGHT // CELL_SIZE // 2
snake.append((start_x, start_y))
snake.append((start_x - 1, start_y))
snake.append((start_x - 2, start_y))

direction = RIGHT

Movement Logic

def move_snake(snake, direction):
    """Move the snake one step in the current direction."""
    head_x, head_y = snake[0]
    dx, dy = direction
    new_head = (head_x + dx, head_y + dy)
    snake.appendleft(new_head)  # Add new head
    snake.pop()                 # Remove tail

def handle_input(event, direction):
    """Change direction based on key press (no reversals)."""
    key_map = {
        pygame.K_UP: UP, pygame.K_DOWN: DOWN,
        pygame.K_LEFT: LEFT, pygame.K_RIGHT: RIGHT,
    }
    if event.key in key_map:
        new_dir = key_map[event.key]
        # Prevent 180-degree reversal
        if (new_dir[0] + direction[0] != 0) or (new_dir[1] + direction[1] != 0):
            return new_dir
    return direction
Why deque?

A list would work, but list.insert(0, item) is O(N) because it shifts all elements. deque.appendleft() is O(1). For a snake that could be 100+ segments long, this matters.

Food & Growth

Food appears at random grid positions. When the snake's head lands on food, the snake grows by one segment (skip the pop()) and new food spawns elsewhere.

Food Spawning

import random

def spawn_food(snake, cols, rows):
    """Spawn food at a random position not occupied by the snake."""
    while True:
        pos = (random.randint(0, cols - 1), random.randint(0, rows - 1))
        if pos not in snake:
            return pos

Growth on Collision

def move_snake(snake, direction, food):
    """Move snake; grow if eating food."""
    head_x, head_y = snake[0]
    dx, dy = direction
    new_head = (head_x + dx, head_y + dy)
    snake.appendleft(new_head)

    if new_head == food:
        # Don't pop tail = snake grows by 1
        return True   # Ate food
    else:
        snake.pop()
        return False  # No food
Tip: Avoiding Infinite Loops

If the snake fills the entire grid, spawn_food would loop forever. For a production game, check if len(snake) >= cols * rows first and declare victory. For this project the grid is large enough that this is not a concern.

Collision & Game Over

The game ends when the snake hits a wall or itself. We check both conditions after every move. On game over, we show a screen and offer a restart option.

Collision Detection

def check_collision(snake, cols, rows):
    """Check for wall or self collision. Return True if game over."""
    head_x, head_y = snake[0]

    # Wall collision
    if head_x < 0 or head_x >= cols or head_y < 0 or head_y >= rows:
        return True

    # Self collision (head in body)
    if snake[0] in list(snake)[1:]:
        return True

    return False

Game Over State

def show_game_over(screen, score):
    """Display game over screen."""
    font_big = pygame.font.Font(None, 72)
    font_small = pygame.font.Font(None, 36)

    text = font_big.render("GAME OVER", True, RED)
    score_text = font_small.render(f"Score: {score}", True, GREEN)
    restart_text = font_small.render("Press R to restart", True, GRAY)

    screen.blit(text, (WINDOW_WIDTH // 2 - text.get_width() // 2, 150))
    screen.blit(score_text, (WINDOW_WIDTH // 2 - score_text.get_width() // 2, 240))
    screen.blit(restart_text, (WINDOW_WIDTH // 2 - restart_text.get_width() // 2, 300))

Score & Display

Score tracking, font rendering, and a game over screen turn our mechanics into a complete game experience. Pygame's font system renders text to surfaces that we blit onto the screen.

Rendering the Snake & Food

def draw_snake(screen, snake):
    for i, (x, y) in enumerate(snake):
        color = GREEN if i == 0 else DARK_GREEN
        rect = pygame.Rect(x * CELL_SIZE, y * CELL_SIZE, CELL_SIZE, CELL_SIZE)
        pygame.draw.rect(screen, color, rect)
        pygame.draw.rect(screen, BLACK, rect, 1)  # Border

def draw_food(screen, food):
    rect = pygame.Rect(food[0] * CELL_SIZE, food[1] * CELL_SIZE, CELL_SIZE, CELL_SIZE)
    pygame.draw.rect(screen, RED, rect)

def draw_score(screen, score):
    font = pygame.font.Font(None, 36)
    text = font.render(f"Score: {score}", True, GREEN)
    screen.blit(text, (10, 10))
