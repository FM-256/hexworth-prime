Opening and Inspecting Images

from PIL import Image

def load_image(filepath):
    """Open an image and display its properties."""
    img = Image.open(filepath)
    print(f"Format: {img.format}")
    print(f"Size: {img.size[0]}x{img.size[1]}")
    print(f"Mode: {img.mode}")  # RGB, RGBA, L (grayscale)
    return img

Creating Thumbnails

def create_thumbnail(img, max_size=(200, 200)):
    """Create a thumbnail preserving aspect ratio."""
    thumb = img.copy()
    thumb.thumbnail(max_size)
    return thumb
Tip: Always Copy First

Pillow operations like thumbnail() modify the image in-place. Always call img.copy() before transforming to preserve the original. This lets users undo changes by reverting to the stored copy.

Filter Engine

Pillow's ImageFilter module provides common filters: blur, sharpen, edge detection, emboss. We will build a filter engine that applies any named filter to an image.

Built-in Filters

from PIL import ImageFilter

FILTERS = {
    "blur": ImageFilter.BLUR,
    "sharpen": ImageFilter.SHARPEN,
    "edge": ImageFilter.FIND_EDGES,
    "emboss": ImageFilter.EMBOSS,
    "contour": ImageFilter.CONTOUR,
    "detail": ImageFilter.DETAIL,
    "smooth": ImageFilter.SMOOTH_MORE,
}

def apply_filter(img, filter_name):
    """Apply a named filter to an image."""
    filt = FILTERS.get(filter_name.lower())
    if not filt:
        print(f"Unknown filter: {filter_name}")
        return img
    return img.filter(filt)
How Filters Work

Most filters are convolution operations: a small matrix (kernel) slides across the image, computing weighted sums of neighboring pixels. BLUR averages neighbors (smoothing), SHARPEN amplifies differences (edge enhancement), and FIND_EDGES detects intensity changes.

Color Adjustments

Beyond filters, Pillow's ImageEnhance module adjusts brightness, contrast, color saturation, and sharpness with a simple factor-based API.

ImageEnhance API

from PIL import ImageEnhance

def adjust_brightness(img, factor):
    """Adjust brightness. factor: 0.0=black, 1.0=original, 2.0=double."""
    enhancer = ImageEnhance.Brightness(img)
    return enhancer.enhance(factor)

def adjust_contrast(img, factor):
    """Adjust contrast. factor: 0.0=gray, 1.0=original, 2.0=double."""
    enhancer = ImageEnhance.Contrast(img)
    return enhancer.enhance(factor)

def adjust_color(img, factor):
    """Adjust color saturation. 0.0=grayscale, 1.0=original."""
    enhancer = ImageEnhance.Color(img)
    return enhancer.enhance(factor)

The factor parameter is intuitive: 1.0 = no change, values below reduce the effect, values above amplify it. A brightness factor of 0.0 produces a black image; 2.0 doubles brightness.

Custom Filters

Beyond built-in filters, Pillow lets you define custom convolution kernels. A kernel is a small matrix that defines how neighboring pixels contribute to each output pixel.

ImageFilter.Kernel()

from PIL import ImageFilter

# A 3x3 Gaussian blur kernel
gaussian_blur = ImageFilter.Kernel(
    size=(3, 3),
    kernel=[1, 2, 1,
            2, 4, 2,
            1, 2, 1],
    scale=16,   # Sum of kernel values (normalizer)
    offset=0
)

# A 3x3 edge detection kernel
laplacian = ImageFilter.Kernel(
    size=(3, 3),
    kernel=[ 0, -1,  0,
            -1,  4, -1,
             0, -1,  0],
    scale=1,
    offset=128  # Shift to see negative values as gray
)
Tip: Scale and Offset

scale divides the kernel sum (prevents overflow). Set it to the sum of all positive kernel values. offset adds a constant to the result (useful for edge detection where values can be negative).

tkinter GUI

Now we build the graphical interface. The GUI includes a file browser to open images, a preview panel, and buttons for each filter. We use tkinter.filedialog for file selection and ImageTk.PhotoImage for displaying Pillow images in tkinter.

Application Window

import tkinter as tk
from tkinter import filedialog
from PIL import Image, ImageTk

class FilterStudioApp:
    def __init__(self, root):
        self.root = root
        root.title("Image Filter Studio")
        root.geometry("900x700")

        self.original_img = None
        self.current_img = None

        # Toolbar
        toolbar = tk.Frame(root, bg="#2d2d2d", height=50)
        toolbar.pack(fill=tk.X)

        tk.Button(toolbar, text="Open", command=self.open_file).pack(side=tk.LEFT, padx=5, pady=10)
        for name in ["Blur", "Sharpen", "Edge", "Emboss"]:
            tk.Button(toolbar, text=name,
                      command=lambda n=name: self.apply(n)).pack(side=tk.LEFT, padx=2, pady=10)
        tk.Button(toolbar, text="Reset", command=self.reset).pack(side=tk.RIGHT, padx=5, pady=10)

        # Image preview
        self.canvas = tk.Label(root, bg="#1a1f2e")
        self.canvas.pack(fill=tk.BOTH, expand=True)

    def open_file(self):
        path = filedialog.askopenfilename(
            filetypes=[("Images", "*.png *.jpg *.jpeg *.bmp")]
        )
        if path:
            self.original_img = Image.open(path)
            self.current_img = self.original_img.copy()
            self.update_preview()

    def update_preview(self):
        display = self.current_img.copy()
        display.thumbnail((800, 600))
        photo = ImageTk.PhotoImage(display)
        self.canvas.config(image=photo)
        self.canvas.image = photo  # Keep reference to prevent GC

Save & Export

The final feature lets users save their filtered images. Pillow supports saving to PNG, JPEG, BMP, and TIFF. We also add batch processing to apply a filter to an entire folder of images.

Saving Images

from pathlib import Path

def save_image(img, output_path, format=None):
    """Save image, auto-detecting format from extension."""
    path = Path(output_path)
    if format is None:
        format = path.suffix.lstrip(".").upper()
        if format == "JPG":
            format = "JPEG"
    img.save(output_path, format=format)
    print(f"Saved to {output_path} ({format})")

Batch Processing

def batch_filter(input_dir, output_dir, filter_name):
    """Apply a filter to all images in a directory."""
    input_path = Path(input_dir)
    output_path = Path(output_dir)
    output_path.mkdir(parents=True, exist_ok=True)

    for img_file in input_path.glob("*.jpg"):
        img = Image.open(img_file)
        filtered = apply_filter(img, filter_name)
        out = output_path / img_file.name
        filtered.save(out)
        print(f"Processed: {img_file.name}")
