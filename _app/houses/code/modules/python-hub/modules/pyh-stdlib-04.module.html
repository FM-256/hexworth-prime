Counter

The Counter class from the collections module is a specialized dictionary for counting hashable objects. It is the fastest way to tally frequencies, find most common elements, and do set-like arithmetic on counts.

Creating and Using Counters

from collections import Counter

# Count from a string
letters = Counter("mississippi")
print(letters)
# Counter({'s': 4, 'i': 4, 'p': 2, 'm': 1})

# Count from a list
words = ["error", "info", "error", "warning", "error", "info"]
log_counts = Counter(words)
print(log_counts)
# Counter({'error': 3, 'info': 2, 'warning': 1})

# Most common elements
print(log_counts.most_common(2))
# [('error', 3), ('info', 2)]

Counter Arithmetic

from collections import Counter

today = Counter({"error": 5, "warning": 3, "info": 10})
yesterday = Counter({"error": 2, "warning": 5, "info": 8})

# Addition
total = today + yesterday
print(total)  # Counter({'info': 18, 'error': 7, 'warning': 8})

# Subtraction (drops zero and negative)
diff = today - yesterday
print(diff)   # Counter({'error': 3, 'info': 2})

# Update (add counts in place)
today.update(["error", "error", "critical"])
print(today)  # error: 7, warning: 3, info: 10, critical: 1

# Subtract counts
today.subtract({"info": 5})
print(today["info"])  # 5

# Get all elements (repeated by count)
print(list(Counter({"a": 2, "b": 3}).elements()))
# ['a', 'a', 'b', 'b', 'b']
Tip: Counter as a Bag/Multiset

Counter implements a "bag" or "multiset" data structure. It supports intersection (&) and union (|) operations in addition to + and -. Use c[key] to access counts — missing keys return 0 instead of raising KeyError.

defaultdict & More

A defaultdict automatically creates a default value for missing keys, eliminating the need for "check if key exists, then initialize" patterns. OrderedDict remembers insertion order, and ChainMap searches through multiple dictionaries.

defaultdict

from collections import defaultdict

# defaultdict(list) - auto-creates empty lists for new keys
groups = defaultdict(list)
data = [("admin", "Alice"), ("user", "Bob"), ("admin", "Charlie"),
        ("user", "Diana"), ("guest", "Eve")]

for role, name in data:
    groups[role].append(name)  # No need to check if key exists!

print(dict(groups))
# {'admin': ['Alice', 'Charlie'], 'user': ['Bob', 'Diana'], 'guest': ['Eve']}

# defaultdict(int) - auto-creates 0 for new keys
counts = defaultdict(int)
for word in "hello world hello python".split():
    counts[word] += 1  # No KeyError on first occurrence

# defaultdict(set) - auto-creates empty sets
tags = defaultdict(set)
tags["server1"].add("production")
tags["server1"].add("web")
tags["server2"].add("staging")

ChainMap

from collections import ChainMap

# ChainMap searches through multiple dicts in order
defaults = {'color': 'blue', 'size': 'medium', 'theme': 'dark'}
user_prefs = {'color': 'green'}
env_override = {'theme': 'light'}

config = ChainMap(env_override, user_prefs, defaults)
print(config['color'])  # green (from user_prefs)
print(config['theme'])  # light (from env_override)
print(config['size'])   # medium (from defaults)
Info: Regular dict is Ordered Since Python 3.7

Since Python 3.7, regular dict maintains insertion order as a language guarantee. OrderedDict is still useful for its move_to_end() method and for code that needs to work with Python 3.6 or earlier.

namedtuple & deque

namedtuple creates lightweight, immutable classes for structured data. deque (double-ended queue) provides O(1) operations on both ends, making it perfect for queues, stacks, and sliding windows.

namedtuple

from collections import namedtuple

# Define a named tuple type
Server = namedtuple('Server', ['hostname', 'ip', 'port', 'role'])

# Create instances
web = Server('web-01', '10.0.0.1', 80, 'frontend')
db = Server(hostname='db-01', ip='10.0.0.2', port=5432, role='database')

# Access by name or index
print(web.hostname)  # web-01
print(web[1])         # 10.0.0.1
print(web._fields)   # ('hostname', 'ip', 'port', 'role')

# Convert to dict
print(web._asdict())  # {'hostname': 'web-01', 'ip': '10.0.0.1', ...}

# Create modified copy (namedtuples are immutable)
web_ssl = web._replace(port=443)
print(web_ssl)  # Server(hostname='web-01', ip='10.0.0.1', port=443, ...)

deque

from collections import deque

# Basic deque operations
q = deque(['task1', 'task2', 'task3'])

# Add to right (end) and left (front)
q.append('task4')        # right end
q.appendleft('task0')   # left end

# Remove from right and left
last = q.pop()       # task4
first = q.popleft()  # task0

# Rotate elements
q = deque([1, 2, 3, 4, 5])
q.rotate(2)   # Move 2 from right to left: deque([4, 5, 1, 2, 3])
q.rotate(-1)  # Move 1 from left to right: deque([5, 1, 2, 3, 4])

# Bounded deque (sliding window)
recent = deque(maxlen=3)
for i in range(6):
    recent.append(i)
    print(list(recent))
# [0], [0, 1], [0, 1, 2], [1, 2, 3], [2, 3, 4], [3, 4, 5]
Tip: deque vs list for Queues

list.pop(0) is O(n) because it shifts every element. deque.popleft() is O(1). For any queue or FIFO pattern, always use deque. Lists are fine for stacks (LIFO) since list.pop() is already O(1).

itertools — Infinite & Combinatoric

The itertools module provides a toolkit of fast, memory-efficient iterator building blocks. Infinite iterators produce values forever (until you stop them), while combinatoric iterators generate permutations and combinations.

Infinite Iterators

from itertools import count, cycle, repeat

# count(start, step) - infinite counter
for i in count(10, 2):
    if i > 20: break
    print(i, end=" ")  # 10 12 14 16 18 20

# cycle(iterable) - repeat sequence forever
colors = cycle(["red", "green", "blue"])
for i in range(6):
    print(next(colors), end=" ")  # red green blue red green blue

# repeat(value, times) - repeat a single value
print(list(repeat("hello", 3)))  # ['hello', 'hello', 'hello']

Combinatoric Iterators

from itertools import product, permutations, combinations, combinations_with_replacement

# product - Cartesian product (nested for loops)
print(list(product("AB", "12")))
# [('A','1'), ('A','2'), ('B','1'), ('B','2')]

# permutations - all orderings
print(list(permutations("ABC", 2)))
# [('A','B'), ('A','C'), ('B','A'), ('B','C'), ('C','A'), ('C','B')]

# combinations - unordered selections (no repeats)
print(list(combinations("ABCD", 2)))
# [('A','B'), ('A','C'), ('A','D'), ('B','C'), ('B','D'), ('C','D')]

# combinations_with_replacement - allow repeats
print(list(combinations_with_replacement("AB", 2)))
# [('A','A'), ('A','B'), ('B','B')]
Security: Combinatorial Explosion

Combinatoric iterators can produce enormous outputs. permutations("abcdefghij") generates 3,628,800 items. product(range(100), repeat=4) generates 100 million. Always calculate the output size first and use iterators (not lists) when dealing with large combinatorial spaces.

itertools — Terminating

Terminating iterators consume finite input and produce finite output. They are the workhorses for data processing: chaining multiple iterables, filtering with predicates, grouping consecutive elements, and slicing without loading everything into memory.

Chaining and Slicing

from itertools import chain, islice

# chain - concatenate multiple iterables
servers = ["web1", "web2"]
databases = ["db1", "db2"]
caches = ["redis1"]

all_hosts = list(chain(servers, databases, caches))
print(all_hosts)  # ['web1', 'web2', 'db1', 'db2', 'redis1']

# islice - slice an iterator without converting to list
from itertools import count
first_five = list(islice(count(100), 5))
print(first_five)  # [100, 101, 102, 103, 104]

Filtering and Grouping

from itertools import dropwhile, takewhile, groupby, compress

# takewhile - take items while predicate is true
data = [1, 3, 5, 7, 2, 4, 6]
print(list(takewhile(lambda x: x < 6, data)))
# [1, 3, 5]  (stops at 7, which is >= 6)

# dropwhile - skip items while predicate is true
print(list(dropwhile(lambda x: x < 6, data)))
# [7, 2, 4, 6]  (starts at 7, includes everything after)

# compress - filter by selectors
items = ["a", "b", "c", "d"]
mask = [True, False, True, False]
print(list(compress(items, mask)))  # ['a', 'c']

# groupby - group consecutive equal elements
# IMPORTANT: data must be sorted by the key first!
logs = [("ERROR", "msg1"), ("ERROR", "msg2"), ("INFO", "msg3"), ("INFO", "msg4")]
for level, entries in groupby(logs, key=lambda x: x[0]):
    print(f"{level}: {list(entries)}")

starmap and zip_longest

from itertools import starmap, zip_longest

# starmap - apply function to each tuple of args
pairs = [(2, 3), (4, 5), (6, 7)]
products = list(starmap(lambda a, b: a * b, pairs))
print(products)  # [6, 20, 42]

# zip_longest - zip with fill value for shorter iterables
names = ["Alice", "Bob"]
scores = [95, 87, 91]
print(list(zip_longest(names, scores, fillvalue="N/A")))
# [('Alice', 95), ('Bob', 87), ('N/A', 91)]
Tip: groupby Requires Sorted Input

groupby only groups consecutive elements with the same key. If your data isn't sorted, you'll get multiple groups for the same key. Always sort by the group key first: sorted(data, key=keyfunc).

functools

The functools module provides higher-order functions that act on or return other functions. These tools let you reduce boilerplate, add caching, create partial functions, and write better decorators.

functools.reduce()

from functools import reduce

# Reduce a sequence to a single value
numbers = [1, 2, 3, 4, 5]

# Sum: ((((1+2)+3)+4)+5) = 15
total = reduce(lambda a, b: a + b, numbers)
print(total)  # 15

# Product: ((((1*2)*3)*4)*5) = 120
product = reduce(lambda a, b: a * b, numbers)
print(product)  # 120

# Find max (without using max())
biggest = reduce(lambda a, b: a if a > b else b, numbers)
print(biggest)  # 5

functools.partial()

from functools import partial

# Freeze some function arguments
def power(base, exponent):
    return base ** exponent

square = partial(power, exponent=2)
cube = partial(power, exponent=3)

print(square(5))  # 25
print(cube(3))    # 27

# Practical: create specialized loggers
import logging
error_log = partial(print, "[ERROR]")
info_log = partial(print, "[INFO]")
error_log("Connection failed")  # [ERROR] Connection failed

functools.lru_cache()

from functools import lru_cache
import time

# Cache function results (memoization)
@lru_cache(maxsize=128)
def expensive_lookup(user_id):
    """Simulate a slow database query."""
    time.sleep(1)  # Simulate latency
    return {"id": user_id, "name": f"User_{user_id}"}

# First call: slow (1 second)
result1 = expensive_lookup(42)

# Second call with same argument: instant (cached)
result2 = expensive_lookup(42)

# Check cache stats
print(expensive_lookup.cache_info())
# CacheInfo(hits=1, misses=1, maxsize=128, currsize=1)

functools.wraps()

from functools import wraps

def timer(func):
    """Decorator that times function execution."""
    @wraps(func)  # Preserves original function metadata
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        elapsed = time.time() - start
        print(f"{func.__name__} took {elapsed:.4f}s")
        return result
    return wrapper

@timer
def process_data(n):
    """Process n items."""
    return sum(range(n))

process_data(1000000)
print(process_data.__name__)  # process_data (preserved by @wraps)
print(process_data.__doc__)   # Process n items. (preserved by @wraps)
Tip: functools Selection Guide

Use reduce() when you need to combine a sequence into a single result. Use partial() to create specialized versions of general functions. Use lru_cache() to memoize expensive computations. Use wraps() in every decorator you write.
