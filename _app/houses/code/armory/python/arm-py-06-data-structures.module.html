Learning Objectives

  • Use lists for ordered, mutable collections
  • Use tuples for immutable sequences
  • Use dictionaries for key-value mappings
  • Use sets for unique, unordered collections
  • Master slicing syntax

Data Structures at a Glance

list [ ] Ordered, mutable. Use when you need to add, remove, or change items.
tuple ( ) Ordered, immutable. Use for fixed collections like coordinates or records.
dict { k: v } Key-value pairs, fast lookup. Use for mappings and structured data.
set { } Unordered, unique items only. Use for membership tests and deduplication.

Lists

lists.py
# Create a list
open_ports = [22, 80, 443, 8080]

# Access by index (0-based)
print(open_ports[0])   # 22
print(open_ports[-1])  # 8080 (last item)

# Modify
open_ports.append(3306)        # Add to end
open_ports.insert(1, 53)       # Insert at index 1
open_ports.remove(80)           # Remove by value
popped = open_ports.pop()       # Remove and return last item

# Common methods
print(len(open_ports))          # Length
print(443 in open_ports)       # Membership test: True
open_ports.sort()               # Sort in place
sorted_copy = sorted(open_ports)  # Return sorted copy

# Combine lists
tcp_ports = [22, 80]
udp_ports = [53, 123]
all_ports = tcp_ports + udp_ports  # [22, 80, 53, 123]

Slicing

Slicing extracts a portion of a sequence. The syntax is sequence[start:stop:step]. The stop index is exclusive.

slicing.py
data = [10, 20, 30, 40, 50, 60, 70, 80]

print(data[2:5])     # [30, 40, 50]  -- index 2, 3, 4
print(data[:3])      # [10, 20, 30]  -- first 3
print(data[5:])      # [60, 70, 80]  -- from index 5 to end
print(data[::2])     # [10, 30, 50, 70]  -- every other
print(data[::-1])    # [80, 70, 60, 50, 40, 30, 20, 10]  -- reversed

# Slicing works on strings too
ip = "192.168.1.100"
print(ip[:3])   # "192"
print(ip[-3:])  # "100"

Tuples

Tuples are immutable lists. Once created, you cannot add, remove, or change elements. Use them for data that should not change.

tuples.py
# Create a tuple
server = ("web01", "192.168.1.10", 8080)

# Access by index (same as lists)
print(server[0])  # web01
print(server[1])  # 192.168.1.10

# Unpack into variables
name, ip, port = server
print(f"{name} at {ip}:{port}")

# Tuples as dictionary keys (lists cannot be)
connections = {}
connections[("192.168.1.10", 22)] = "active"
connections[("10.0.0.5", 443)] = "closed"

# Single-element tuple needs a trailing comma
single = (42,)       # This is a tuple
not_tuple = (42)     # This is just an int in parentheses

Dictionaries

dicts.py
# Create a dictionary
host = {
    "hostname": "dc01.corp.local",
    "ip": "10.0.0.1",
    "os": "Windows Server 2022",
    "open_ports": [53, 88, 135, 389, 445],
    "is_dc": True,
}

# Access values
print(host["hostname"])           # dc01.corp.local
print(host.get("os"))              # Windows Server 2022
print(host.get("mac", "unknown")) # unknown (default)

# Modify
host["ip"] = "10.0.0.2"          # Update existing
host["role"] = "domain controller"  # Add new key
del host["is_dc"]                  # Remove key

# Iterate
for key, value in host.items():
    print(f"  {key}: {value}")

# Useful methods
print(list(host.keys()))    # All keys
print(list(host.values()))  # All values
print("ip" in host)          # True

Sets

sets.py
# Create a set (duplicates are automatically removed)
failed_ips = {"10.0.0.5", "192.168.1.50", "10.0.0.5", "10.0.0.23"}
print(failed_ips)  # {'10.0.0.5', '192.168.1.50', '10.0.0.23'}

# Set operations
internal = {"10.0.0.5", "10.0.0.23", "10.0.0.100"}
external = {"8.8.8.8", "10.0.0.5", "1.1.1.1"}

# Intersection: IPs in both sets
overlap = internal & external
print(overlap)  # {'10.0.0.5'}

# Union: all unique IPs
all_ips = internal | external
print(len(all_ips))  # 5

# Difference: IPs only in internal
internal_only = internal - external
print(internal_only)  # {'10.0.0.23', '10.0.0.100'}

# Practical: deduplicate a list
log_ips = ["10.0.0.5", "10.0.0.5", "10.0.0.23", "10.0.0.5"]
unique = list(set(log_ips))
print(unique)  # ['10.0.0.5', '10.0.0.23']

Nested Structures

nested.py
# Real-world: scan results as nested data
scan_results = {
    "target": "192.168.1.0/24",
    "hosts": [
        {
            "ip": "192.168.1.1",
            "ports": [22, 80],
            "os": "Linux"
        },
        {
            "ip": "192.168.1.10",
            "ports": [135, 445, 3389],
            "os": "Windows"
        }
    ]
}

# Navigate the structure
for host in scan_results["hosts"]:
    print(f"{host['ip']} ({host['os']}) - {len(host['ports'])} open ports")
# 192.168.1.1 (Linux) - 2 open ports
# 192.168.1.10 (Windows) - 3 open ports

Key Takeaways

  • Lists are mutable ordered sequences: [1, 2, 3]
  • Tuples are immutable ordered sequences: (1, 2, 3)
  • Dicts map keys to values: {"key": "value"}
  • Sets hold unique items: {1, 2, 3}
  • Slicing: seq[start:stop:step] -- works on lists, tuples, and strings
  • Use .get(key, default) on dicts to avoid KeyError
  • Set operations (&, |, -) are powerful for comparing collections
