String Methods

Strings are immutable in Python. Every method returns a new string -- it never modifies the original.

string_methods.py
hostname = "  Web-Server-01.corp.local  "

# Cleaning
print(hostname.strip())       # "Web-Server-01.corp.local"
print(hostname.lower())       # "  web-server-01.corp.local  "
print(hostname.upper())       # "  WEB-SERVER-01.CORP.LOCAL  "

# Searching
domain = "web-server-01.corp.local"
print(domain.startswith("web"))      # True
print(domain.endswith(".local"))     # True
print(domain.find("corp"))           # 15 (index where found)
print(domain.count("."))             # 2
print("server" in domain)           # True

# Replacing
print(domain.replace("corp", "prod"))  # web-server-01.prod.local

# Checking content
print("abc123".isalnum())    # True
print("12345".isdigit())     # True
print("Hello".isalpha())     # True

Split and Join

split_join.py
# split() breaks a string into a list
log_line = "2026-03-15 14:23:01 ERROR Failed login from 10.0.0.5"
parts = log_line.split()
print(parts)
# ['2026-03-15', '14:23:01', 'ERROR', 'Failed', 'login', 'from', '10.0.0.5']

# Split on a specific delimiter
csv_line = "hostname,ip,port,status"
fields = csv_line.split(",")
print(fields)  # ['hostname', 'ip', 'port', 'status']

# Split with max splits
print(log_line.split(" ", 2))
# ['2026-03-15', '14:23:01', 'ERROR Failed login from 10.0.0.5']

# join() combines a list into a string
words = ["nmap", "-sV", "-p", "22,80,443", "target.com"]
command = " ".join(words)
print(command)  # "nmap -sV -p 22,80,443 target.com"

# Join with different separators
ports = ["22", "80", "443"]
print(",".join(ports))   # "22,80,443"
print("\n".join(ports))  # Each on its own line

Reading Files

Always use the with statement when working with files. It automatically closes the file when you are done, even if an error occurs.

read_files.py
# Read entire file as a single string
with open("config.txt", "r") as f:
    content = f.read()
    print(content)

# Read file as a list of lines
with open("access.log", "r") as f:
    lines = f.readlines()
    print(f"Total lines: {len(lines)}")

# Process line by line (memory efficient for large files)
with open("access.log", "r") as f:
    for line_num, line in enumerate(f, 1):
        line = line.strip()  # Remove trailing newline
        if "ERROR" in line:
            print(f"Line {line_num}: {line}")

Never use f.read() on large files. It loads the entire file into memory. For files over a few MB, iterate line by line with a for loop instead.

Writing Files

write_files.py
# Write (creates or overwrites)
with open("report.txt", "w") as f:
    f.write("Scan Report\n")
    f.write("=" * 40 + "\n")
    f.write(f"Date: 2026-03-15\n")
    f.write(f"Hosts scanned: 254\n")

# Append (adds to existing file)
with open("report.txt", "a") as f:
    f.write("\n--- Additional findings ---\n")
    f.write("Port 3389 open on 5 hosts\n")

# Write multiple lines at once
findings = [
    "192.168.1.10 - SSH open\n",
    "192.168.1.20 - HTTP open\n",
    "192.168.1.30 - RDP open\n",
]
with open("findings.txt", "w") as f:
    f.writelines(findings)

Practical: Log Parser

log_parser.py
def parse_auth_log(filepath):
    """Parse an auth log and extract failed login attempts."""
    failed = {}

    with open(filepath, "r") as f:
        for line in f:
            line = line.strip()
            if "Failed password" not in line:
                continue

            # Extract IP (last word before "port")
            parts = line.split()
            try:
                port_idx = parts.index("port")
                ip = parts[port_idx - 1]
                failed[ip] = failed.get(ip, 0) + 1
            except (ValueError, IndexError):
                continue

    return failed

Key Takeaways

  • Strings are immutable -- methods return new strings
  • split() breaks strings apart; join() puts them back together
  • Always use with open(...) as f: for file operations
  • File modes: "r" read, "w" write (overwrite), "a" append
  • For large files, iterate line by line instead of reading the whole file
  • strip() removes whitespace and newlines from both ends
