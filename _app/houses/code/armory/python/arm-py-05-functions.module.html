Defining Functions

Functions let you package reusable logic under a name. In Python, you define a function with def, and it can return a value with return.

basic_functions.py
def classify_port(port):
    """Classify a network port by its range."""
    if port < 0 or port > 65535:
        return "invalid"
    elif port < 1024:
        return "well-known"
    elif port < 49152:
        return "registered"
    else:
        return "dynamic"

# Call the function
print(classify_port(22))     # well-known
print(classify_port(8080))   # registered
print(classify_port(50000))  # dynamic

The triple-quoted string right after def is called a docstring. It documents what the function does and is accessible via help(classify_port). Always write docstrings for functions others will use.

Arguments and Parameters

arguments.py
# Default arguments
def scan_target(host, port=80, timeout=5.0):
    """Scan a single host:port with optional timeout."""
    print(f"Scanning {host}:{port} (timeout: {timeout}s)")

# All of these work:
scan_target("192.168.1.1")                 # port=80, timeout=5.0
scan_target("192.168.1.1", 443)            # timeout=5.0
scan_target("192.168.1.1", timeout=10)     # port=80, timeout=10
scan_target("192.168.1.1", 22, 2.0)       # all specified

# Keyword arguments can be in any order
scan_target(timeout=1, host="10.0.0.1", port=3389)

*args and **kwargs

Sometimes you want a function to accept any number of arguments. *args collects extra positional arguments as a tuple, and **kwargs collects extra keyword arguments as a dictionary.

flexible_args.py
def log_event(severity, message, *tags, **metadata):
    """Log a security event with tags and metadata."""
    print(f"[{severity.upper()}] {message}")
    if tags:
        print(f"  Tags: {', '.join(tags)}")
    for key, val in metadata.items():
        print(f"  {key}: {val}")

log_event(
    "critical",
    "Unauthorized access detected",
    "intrusion", "auth", "lateral-movement",
    source_ip="10.0.0.5",
    target="DC01",
    protocol="SMB"
)
# Output:
# [CRITICAL] Unauthorized access detected
#   Tags: intrusion, auth, lateral-movement
#   source_ip: 10.0.0.5
#   target: DC01
#   protocol: SMB

Return Values

return_values.py
# Return multiple values (as a tuple)
def analyze_password(password):
    """Return password length, has_upper, has_digit."""
    length = len(password)
    has_upper = any(c.isupper() for c in password)
    has_digit = any(c.isdigit() for c in password)
    return length, has_upper, has_digit

# Unpack the returned tuple
length, upper, digit = analyze_password("Secret42")
print(f"Length: {length}, Uppercase: {upper}, Digits: {digit}")
# Length: 8, Uppercase: True, Digits: True

# Functions without return give back None
def greet(name):
    print(f"Hello, {name}")

result = greet("Alice")
print(result)  # None

Scope

Variables defined inside a function are local -- they do not exist outside. Variables defined at module level are global.

scope.py
threshold = 100  # Global variable

def check_load(current):
    # 'current' and 'status' are local to this function
    status = "overloaded" if current > threshold else "normal"
    return status

print(check_load(150))  # overloaded
# print(status)  # NameError -- status does not exist here

# Modifying a global requires the 'global' keyword
counter = 0

def increment():
    global counter
    counter += 1

increment()
increment()
print(counter)  # 2

Avoid global when possible. Passing values through arguments and return values makes code easier to test and debug.

Lambda Expressions

A lambda is a small anonymous function written in a single expression. It is useful when you need a quick throwaway function, typically as an argument to sorted(), map(), or filter().

lambda.py
# Regular function
def double(x):
    return x * 2

# Same thing as a lambda
double = lambda x: x * 2

# Practical use: sort a list of tuples by second element
hosts = [
    ("web01", 95),
    ("db01", 42),
    ("app01", 78),
    ("cache01", 12),
]

# Sort by CPU load (second element), highest first
by_load = sorted(hosts, key=lambda h: h[1], reverse=True)
for name, load in by_load:
    print(f"  {name}: {load}% CPU")
# web01: 95% CPU
# app01: 78% CPU
# db01: 42% CPU
# cache01: 12% CPU

Key Takeaways

  • Define functions with def name(params): and return values with return
  • Use default arguments for optional parameters
  • *args captures extra positional args; **kwargs captures keyword args
  • Functions can return multiple values via tuple unpacking
  • Local variables are not visible outside their function
  • Lambdas are single-expression anonymous functions: lambda x: x * 2
