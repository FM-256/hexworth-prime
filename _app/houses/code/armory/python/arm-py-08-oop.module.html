Classes and Objects

A class is a blueprint for creating objects. An object is an instance of a class. Think of a class as the template and each object as a concrete thing built from that template.

basic_class.py
class Host:
    """Represents a network host."""

    def __init__(self, ip, hostname, os_type="unknown"):
        """Initialize with IP, hostname, and optional OS."""
        self.ip = ip
        self.hostname = hostname
        self.os_type = os_type
        self.open_ports = []

    def add_port(self, port):
        """Record an open port."""
        if port not in self.open_ports:
            self.open_ports.append(port)

    def summary(self):
        """Return a formatted summary."""
        ports = ", ".join(str(p) for p in sorted(self.open_ports))
        return f"{self.hostname} ({self.ip}) - {self.os_type} - Ports: {ports or 'none'}"

# Create instances
web = Host("192.168.1.10", "web01", "Linux")
web.add_port(80)
web.add_port(443)

dc = Host("10.0.0.1", "dc01", "Windows")
dc.add_port(88)
dc.add_port(445)

print(web.summary())
# web01 (192.168.1.10) - Linux - Ports: 80, 443
print(dc.summary())
# dc01 (10.0.0.1) - Windows - Ports: 88, 445

The self parameter refers to the current instance. It is passed automatically when you call a method -- you never write web.summary(web), just web.summary().

Inheritance

Inheritance lets you create a new class based on an existing one. The child class inherits all attributes and methods, and can add or override them.

inheritance.py
class Host:
    def __init__(self, ip, hostname):
        self.ip = ip
        self.hostname = hostname
        self.open_ports = []

    def scan_result(self):
        return f"{self.hostname} ({self.ip}): {len(self.open_ports)} ports open"

class Server(Host):
    """A host that runs services."""

    def __init__(self, ip, hostname, role):
        super().__init__(ip, hostname)  # Call parent __init__
        self.role = role
        self.services = []

    def add_service(self, name, port):
        self.services.append({"name": name, "port": port})
        self.open_ports.append(port)

    def scan_result(self):
        # Override parent method
        base = super().scan_result()
        return f"{base} [Role: {self.role}]"

# Usage
srv = Server("192.168.1.10", "web01", "web server")
srv.add_service("nginx", 80)
srv.add_service("nginx-ssl", 443)

print(srv.scan_result())
# web01 (192.168.1.10): 2 ports open [Role: web server]

# isinstance checks
print(isinstance(srv, Server))  # True
print(isinstance(srv, Host))    # True (Server IS a Host)

Dunder Methods

Dunder (double-underscore) methods let you define how your objects behave with built-in operations like print(), len(), comparison, and arithmetic.

dunders.py
class IPRange:
    """Represents a range of IP addresses."""

    def __init__(self, base, count):
        self.base = base
        self.count = count

    def __repr__(self):
        """Developer-friendly representation."""
        return f"IPRange('{self.base}', {self.count})"

    def __str__(self):
        """User-friendly string."""
        return f"{self.base}/range({self.count})"

    def __len__(self):
        return self.count

    def __contains__(self, ip):
        """Support 'in' operator."""
        # Simplified: just checks if base matches
        return ip.startswith(self.base.rsplit(".", 1)[0])

    def __eq__(self, other):
        return self.base == other.base and self.count == other.count

subnet = IPRange("192.168.1.0", 254)
print(subnet)                 # 192.168.1.0/range(254)
print(repr(subnet))            # IPRange('192.168.1.0', 254)
print(len(subnet))             # 254
print("192.168.1.50" in subnet) # True

Properties

Properties let you add logic to attribute access without changing the interface. Use them for validation, computed values, or read-only attributes.

properties.py
class Credential:
    def __init__(self, username, password):
        self.username = username
        self._password = password  # Convention: _ means "private"

    @property
    def password(self):
        """Return masked password."""
        return "*" * len(self._password)

    @password.setter
    def password(self, value):
        if len(value) < 8:
            raise ValueError("Password must be at least 8 characters")
        self._password = value

    @property
    def strength(self):
        """Computed property -- not stored, calculated each time."""
        score = len(self._password)
        if any(c.isupper() for c in self._password): score += 5
        if any(c.isdigit() for c in self._password): score += 5
        return "strong" if score >= 20 else "moderate" if score >= 13 else "weak"

cred = Credential("admin", "SuperSecret42")
print(cred.password)   # *************  (masked)
print(cred.strength)   # strong

# cred.password = "short"  # Raises ValueError
cred.password = "NewPassword99"  # Works fine

Key Takeaways

  • Classes are blueprints; objects are instances created from them
  • __init__ initializes new objects; self refers to the current instance
  • Inheritance: class Child(Parent): -- use super() to call parent methods
  • Dunder methods (__str__, __len__, __eq__) customize built-in behavior
  • Properties (@property) add logic to attribute access without changing the API
  • Prefix attributes with _ to signal "private" (convention, not enforced)
