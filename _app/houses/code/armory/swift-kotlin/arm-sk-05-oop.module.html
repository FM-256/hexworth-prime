Classes: Reference Types

Classes are reference types -- multiple variables can point to the same object in memory. Mutations through one reference are visible through all others. Use classes for objects with identity: a network session, a database connection, a UI component.

swift -- class
class NetworkSession {
    var baseURL: String
    private var token: String?

    init(baseURL: String) {
        self.baseURL = baseURL
    }

    func authenticate(token: String) {
        self.token = token
    }

    func isAuthenticated() -> Bool {
        return token != nil
    }
}

let session = NetworkSession(
    baseURL: "https://api.hexworth.io")
session.authenticate(token: "abc")
kotlin -- class
class NetworkSession(
    var baseURL: String
) {
    private var token: String? = null

    fun authenticate(token: String) {
        this.token = token
    }

    fun isAuthenticated(): Boolean {
        return token != null
    }
}

val session = NetworkSession(
    "https://api.hexworth.io")
session.authenticate("abc")

Structs & Data Classes: Value Types

Swift structs and Kotlin data classes are value types -- assignment copies the data. They are ideal for model objects: API responses, configuration records, domain entities that should not share state.

swift -- struct
struct User {
    let id: Int
    var username: String
    var role: String

    // Computed property
    var isAdmin: Bool {
        role == "admin"
    }
}

var u1 = User(id: 1, username: "alex",
               role: "user")
var u2 = u1          // copy
u2.username = "blake"
// u1.username still "alex"
kotlin -- data class
data class User(
    val id: Int,
    var username: String,
    var role: String
) {
    // Computed property
    val isAdmin: Boolean
        get() = role == "admin"
}

var u1 = User(1, "alex", "user")
// copy() with modified fields
val u2 = u1.copy(username = "blake")
// u1.username still "alex"

Protocols & Interfaces

Protocols (Swift) and interfaces (Kotlin) define contracts -- a set of methods a type must implement. They enable polymorphism without inheritance and are the backbone of testable, loosely coupled code.

swift -- protocol
protocol Authenticatable {
    var userId: String { get }
    func authenticate() -> Bool
}

struct BiometricAuth: Authenticatable {
    let userId: String

    func authenticate() -> Bool {
        // Face ID / Touch ID check
        return true
    }
}

func login(_ auth: any Authenticatable) {
    if auth.authenticate() {
        print("Access granted: \(auth.userId)")
    }
}
kotlin -- interface
interface Authenticatable {
    val userId: String
    fun authenticate(): Boolean
}

class BiometricAuth(
    override val userId: String
) : Authenticatable {

    override fun authenticate(): Boolean {
        // Fingerprint / Face check
        return true
    }
}

fun login(auth: Authenticatable) {
    if (auth.authenticate()) {
        println("Access granted: ${auth.userId}")
    }
}

Enums

Both languages have powerful enums that go far beyond simple integer constants. Swift enums can have associated values. Kotlin sealed classes extend this concept into full class hierarchies.

swift -- enum with associated values
enum NetworkError {
    case timeout(Double)   // seconds waited
    case unauthorized
    case serverError(Int) // status code
    case unknown(String)
}

func handle(_ err: NetworkError) {
    switch err {
    case .timeout(let secs):
        print("Timed out after \(secs)s")
    case .unauthorized:
        print("401 -- re-auth required")
    case .serverError(let code):
        print("Server error: \(code)")
    case .unknown(let msg):
        print("Unknown: \(msg)")
    }
}
kotlin -- sealed class
sealed class NetworkError {
    data class Timeout(val secs: Double)
        : NetworkError()
    object Unauthorized : NetworkError()
    data class ServerError(val code: Int)
        : NetworkError()
    data class Unknown(val msg: String)
        : NetworkError()
}

fun handle(err: NetworkError) {
    when (err) {
        is NetworkError.Timeout ->
            println("Timed out: ${err.secs}s")
        NetworkError.Unauthorized ->
            println("401 -- re-auth")
        is NetworkError.ServerError ->
            println("Error: ${err.code}")
        else -> println("Unknown")
    }
}

  4. Make User conform to / implement AccessControlled

Key Takeaways

  • Classes are reference types -- use for objects with identity and shared mutable state
  • Swift structs and Kotlin data classes are value types -- use for model/data objects
  • Protocols/interfaces define contracts enabling polymorphism without inheritance chains
  • Swift enums with associated values and Kotlin sealed classes model sum types safely
  • Kotlin data class auto-generates equals, hashCode, toString, and copy
