Defining Model Types

Before making a network request, define the model types that represent the JSON response. Both languages let the compiler generate all the parsing logic -- you just declare the shape.

swift -- Codable
// Codable = Encodable + Decodable
struct User: Codable {
    let id: Int
    let username: String
    let role: String
    let lastSeen: Date?

    // Map JSON "user_name" to "username"
    enum CodingKeys: String, CodingKey {
        case id
        case username = "user_name"
        case role
        case lastSeen = "last_seen"
    }
}
kotlin -- @Serializable
import kotlinx.serialization.*

// Compiler generates parse logic
@Serializable
data class User(
    val id: Int,
    @SerialName("user_name")
    val username: String,
    val role: String,
    @SerialName("last_seen")
    val lastSeen: String? = null
)

Making HTTP Requests

swift -- URLSession
enum APIError: Error {
    case invalidResponse(Int)
    case decodingFailed
}

func fetchUser(id: Int)
    async throws -> User {
    let url = URL(string:
        "https://api.hexworth.io/users/\(id)")!

    let (data, response) = try await
        URLSession.shared.data(from: url)

    guard let http = response as?
        HTTPURLResponse,
        http.statusCode == 200 else {
        throw APIError.invalidResponse(
            (response as! HTTPURLResponse).statusCode)
    }

    let decoder = JSONDecoder()
    decoder.dateDecodingStrategy = .iso8601
    return try decoder.decode(User.self,
                                from: data)
}
kotlin -- ktor client
import io.ktor.client.*
import io.ktor.client.call.*
import io.ktor.client.request.*

val client = HttpClient {
    install(ContentNegotiation) {
        json()
    }
}

suspend fun fetchUser(id: Int): User {
    val response = client.get(
        "https://api.hexworth.io/users/$id")

    if (!response.status.isSuccess()) {
        throw RuntimeException(
            "HTTP ${response.status.value}")
    }

    return response.body()
}

POST Requests with Body

swift -- POST
struct LoginRequest: Codable {
    let username: String
    let password: String
}

func login(_ req: LoginRequest)
    async throws -> String {
    var request = URLRequest(
        url: URL(string: "https://api/login")!)
    request.httpMethod = "POST"
    request.setValue("application/json",
        forHTTPHeaderField: "Content-Type")
    request.httpBody = try
        JSONEncoder().encode(req)

    let (data, _) = try await
        URLSession.shared.data(for: request)
    return String(data: data, encoding: .utf8)!
}
kotlin -- POST
@Serializable
data class LoginRequest(
    val username: String,
    val password: String
)

suspend fun login(req: LoginRequest): String {
    val response = client.post(
        "https://api/login") {
        contentType(ContentType.Application.Json)
        setBody(req)
    }
    return response.body<String>()
}

Complete API Client Pattern

Wrap all API calls behind a dedicated client class. This isolates network code, makes testing easier (swap in a mock), and provides a single place to inject auth tokens.

swift -- api client class
final class APIClient {
    static let shared = APIClient()
    private let base = "https://api.hexworth.io"
    private var token: String?

    private func makeRequest(_ path: String)
        -> URLRequest {
        var req = URLRequest(
            url: URL(string: base + path)!)
        if let t = token {
            req.setValue("Bearer \(t)",
                forHTTPHeaderField: "Authorization")
        }
        return req
    }

    func getUsers() async throws -> [User] {
        let (data, _) = try await
            URLSession.shared.data(
                for: makeRequest("/users"))
        return try JSONDecoder().decode(
            [User].self, from: data)
    }
}

Security: Never hard-code API tokens in source code. Load from environment variables at build time, or retrieve from the Keychain at runtime. Module 10 covers secure storage in detail.

Try It Yourself

Build a minimal REST client against a public API (jsonplaceholder.typicode.com is good for practice):

  1. Define a Post model with id, userId, title, and body fields using Codable or @Serializable
  2. Fetch the list of posts from https://jsonplaceholder.typicode.com/posts
  3. Filter for posts from userId == 1 using the functional operations from Module 04
  4. Handle network errors and HTTP error status codes explicitly

Key Takeaways

  • Swift Codable and Kotlin @Serializable auto-generate JSON encode/decode logic from type declarations
  • URLSession (Swift) and Ktor (Kotlin) are the idiomatic HTTP clients for each platform
  • Always check HTTP status codes -- a 4xx/5xx response is not a network error at the URL level
  • Wrap all API calls in a dedicated client class for testability and centralized auth injection
  • Never store tokens in source code -- use Keychain (Module 10) or environment config
