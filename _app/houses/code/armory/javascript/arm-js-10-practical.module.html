Project: netaudit -- a Network Audit Tool

We will build a command-line tool that reads a list of hosts from a JSON file, checks their HTTP status, and generates a report. This pulls together everything from the course: variables, functions, async/await, classes, modules, and file I/O.

Reads host lists from JSON File system operations with fs/promises
HTTP health checks Async fetch with timeout and error handling
Colored terminal output Using the chalk npm package
Exports a JSON report Writing structured results to disk

Step 3: The HTTP Checker

lib/checker.js
/** * Checks the HTTP status of a host with a timeout. * Returns a result object with timing and status info. */ const DEFAULT_TIMEOUT = 5000; export async function checkHost(host, timeoutMs = DEFAULT_TIMEOUT) { const startTime = Date.now(); const controller = new AbortController(); const timer = setTimeout(() => controller.abort(), timeoutMs); try { const response = await fetch(host.url, { method: 'HEAD', signal: controller.signal, redirect: 'follow', }); const duration = Date.now() - startTime; return { name: host.name, url: host.url, status: response.status, statusText: response.statusText, ok: response.ok, duration, error: null, }; } catch (err) { const duration = Date.now() - startTime; const isTimeout = err.name === 'AbortError'; return { name: host.name, url: host.url, status: 0, statusText: isTimeout ? 'TIMEOUT' : 'UNREACHABLE', ok: false, duration, error: isTimeout ? `Timeout after ${timeoutMs}ms` : err.message, }; } finally { clearTimeout(timer); } } export async function checkAll(hosts, concurrency = 3) { const results = []; for (let i = 0; i < hosts.length; i += concurrency) { const batch = hosts.slice(i, i + concurrency); const batchResults = await Promise.allSettled( batch.map(h => checkHost(h)) ); results.push(...batchResults.map(r => r.value)); } return results; }

Step 4: The Reporter

lib/reporter.js
import chalk from 'chalk'; import { writeFile } from 'node:fs/promises'; export function printResults(results) { console.log(''); console.log(chalk.bold(' HOST AUDIT RESULTS')); console.log(chalk.dim(' ' + '='.repeat(60))); for (const r of results) { const icon = r.ok ? chalk.green('[OK]') : chalk.red('[FAIL]'); const status = r.ok ? chalk.green(`${r.status}`) : chalk.red(`${r.status || '---'} ${r.statusText}`); const time = r.duration < 200 ? chalk.green(`${r.duration}ms`) : r.duration < 1000 ? chalk.yellow(`${r.duration}ms`) : chalk.red(`${r.duration}ms`); console.log(` ${icon} ${r.name.padEnd(20)} ${status.padEnd(30)} ${time}`); if (r.error) { console.log(chalk.dim(` ${r.error}`)); } } // Summary const passed = results.filter(r => r.ok).length; const failed = results.length - passed; const avgTime = Math.round( results.reduce((sum, r) => sum + r.duration, 0) / results.length ); console.log(chalk.dim(' ' + '='.repeat(60))); console.log(` ${chalk.green(passed + ' passed')} ${chalk.red(failed + ' failed')} Avg: ${avgTime}ms`); console.log(''); } export async function saveReport(results, outputPath) { const report = { generatedAt: new Date().toISOString(), summary: { total: results.length, passed: results.filter(r => r.ok).length, failed: results.filter(r => !r.ok).length, }, results, }; await writeFile(outputPath, JSON.stringify(report, null, 2)); console.log(chalk.dim(` Report saved to ${outputPath}`)); }

Step 5: The Entry Point

index.js
import { readFile } from 'node:fs/promises'; import { resolve } from 'node:path'; import chalk from 'chalk'; import { checkAll } from './lib/checker.js'; import { printResults, saveReport } from './lib/reporter.js'; // Parse CLI arguments const args = process.argv.slice(2); const inputFile = args[0] || 'data/hosts.json'; const outputFile = args.includes('--output') ? args[args.indexOf('--output') + 1] : null; async function main() { console.log(chalk.bold.yellow('\\n netaudit -- Network Health Checker\\n')); // 1. Load the host list let config; try { const raw = await readFile(resolve(inputFile), 'utf-8'); config = JSON.parse(raw); } catch (err) { console.error(chalk.red(` Failed to load ${inputFile}: ${err.message}`)); process.exit(1); } console.log(chalk.dim(` Checking ${config.hosts.length} hosts from "${config.name}"...\\n`)); // 2. Run the checks const results = await checkAll(config.hosts); // 3. Display results printResults(results); // 4. Save report if --output flag given if (outputFile) { await saveReport(results, outputFile); } // 5. Exit with appropriate code const failures = results.filter(r => !r.ok).length; process.exit(failures > 0 ? 1 : 0); } main();

Challenge: Extend the Tool

Build this tool yourself, then add these features:

  1. Add a --timeout flag to customize the request timeout (default 5000ms)
  2. Add a --verbose flag that shows response headers for each host
  3. Add retry logic: if a host fails, retry up to 2 more times before marking it as failed
  4. Add CSV output support with a --format csv flag
  5. Convert the entire project to TypeScript and add type definitions for all functions and data

  3. hashwatch -- monitor files for changes by comparing SHA-256 hashes
  4. envdiff -- compare two .env files and show additions, removals, and changes

Key Takeaways

  • Real Node.js tools use process.argv for CLI arguments, process.exit() for exit codes
  • fs/promises provides async file read/write -- always use the promises version
  • npm packages like chalk, commander, and ora add production polish to CLI tools
  • AbortController provides clean request timeouts for fetch
  • Splitting code into modules (checker, reporter) keeps each file focused and testable
