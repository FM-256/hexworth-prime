Port Scanner

port-scan.sh
Scan common ports on a target host
#!/bin/bash
TARGET="${1:?Usage: $0 <host> [start_port] [end_port]}"
START="${2:-1}"
END="${3:-1024}"
echo "Scanning $TARGET ports $START-$END..."
OPEN=0
for port in $(seq "$START" "$END"); do
if (echo > /dev/tcp/"$TARGET"/"$port") 2>/dev/null; then
SERVICE=$(getent services "$port" 2>/dev/null | awk '{print $1}')
printf " Port %-6d OPEN %s\n" "$port" "$SERVICE"
((OPEN++))
fi
done
echo "---"
echo "$OPEN open port(s) found on $TARGET"

/dev/tcp: Bash has a built-in TCP connection feature. Writing to /dev/tcp/host/port attempts a TCP connection. No external tools needed.

Brute Force Detector

brute-detect.sh
Detect and optionally block brute force attackers
#!/bin/bash
LOGFILE="/var/log/auth.log"
THRESHOLD=5
BLOCK="${1:-false}" # Pass "block" to auto-ban
echo "=== Brute Force Detection ==="
echo "Threshold: $THRESHOLD failures"
echo ""
grep "Failed password" "$LOGFILE" \
| grep -oE '[0-9]+\.[0-9]+\.[0-9]+\.[0-9]+' \
| sort | uniq -c | sort -rn \
| while read -r count ip; do
if [[ "$count" -ge "$THRESHOLD" ]]; then
printf "[THREAT] %-16s %d attempts" "$ip" "$count"
if [[ "$BLOCK" == "block" ]]; then
iptables -A INPUT -s "$ip" -j DROP 2>/dev/null
printf " -> BLOCKED"
fi
printf "\n"
fi
done

SUID/SGID Auditor

audit-suid.sh
Find files with elevated permissions that could be exploited
#!/bin/bash
echo "=== SUID/SGID Binary Audit ==="
echo "Date: $(date)"
echo ""
BASELINE="/var/lib/suid-baseline.txt"
CURRENT=$(mktemp)
# Find all SUID and SGID files
find / -type f \( -perm -4000 -o -perm -2000 \) 2>/dev/null \
| sort > "$CURRENT"
echo "Found $(wc -l < "$CURRENT") SUID/SGID binaries"
if [[ -f "$BASELINE" ]]; then
NEW=$(comm -23 "$CURRENT" "$BASELINE")
REMOVED=$(comm -13 "$CURRENT" "$BASELINE")
if [[ -n "$NEW" ]]; then
echo ""
echo "[ALERT] New SUID/SGID binaries detected:"
echo "$NEW" | while read -r f; do
ls -la "$f" 2>/dev/null
done
fi
else
echo "No baseline found. Creating one now."
cp "$CURRENT" "$BASELINE"
fi
rm -f "$CURRENT"

File Integrity Monitor

integrity-check.sh
Detect unauthorized file modifications using SHA-256 checksums
#!/bin/bash
WATCH_DIRS="/etc /usr/bin /usr/sbin"
DB="/var/lib/integrity.db"
ACTION="${1:-check}" # "init" or "check"
generate_hashes() {
for dir in $WATCH_DIRS; do
find "$dir" -type f -maxdepth 1 2>/dev/null \
| while read -r f; do
sha256sum "$f" 2>/dev/null
done
done | sort
}
case "$ACTION" in
init)
generate_hashes > "$DB"
echo "Baseline created: $(wc -l < "$DB") files"
;;
check)
[[ -f "$DB" ]] || { echo "No baseline. Run: $0 init"; exit 1; }
CURRENT=$(mktemp)
generate_hashes > "$CURRENT"
DIFF=$(diff "$DB" "$CURRENT")
if [[ -n "$DIFF" ]]; then
echo "[ALERT] File integrity changes detected!"
echo "$DIFF"
else
echo "[OK] No changes detected"
fi
rm -f "$CURRENT"
;;
esac

Firewall Rule Generator

firewall-setup.sh
#!/bin/bash
# Basic iptables firewall setup
ALLOWED_PORTS="22 80 443"
TRUSTED_IPS="10.0.0.0/8 192.168.0.0/16"
# Flush existing rules
iptables -F
iptables -X
# Default policies: deny all incoming
iptables -P INPUT DROP
iptables -P FORWARD DROP
iptables -P OUTPUT ACCEPT
# Allow loopback and established connections
iptables -A INPUT -i lo -j ACCEPT
iptables -A INPUT -m state --state ESTABLISHED,RELATED -j ACCEPT
# Open allowed ports
for port in $ALLOWED_PORTS; do
iptables -A INPUT -p tcp --dport "$port" -j ACCEPT
echo "Opened port $port/tcp"
done
# Allow trusted subnets
for net in $TRUSTED_IPS; do
iptables -A INPUT -s "$net" -j ACCEPT
echo "Trusted: $net"
done
echo "Firewall configured."

Key Takeaways

  • Bash can scan ports using /dev/tcp without external tools
  • Log analysis scripts detect brute force attacks by counting failures per IP
  • SUID auditing compares current state against a known baseline
  • File integrity monitoring uses SHA-256 checksums to detect tampering
  • Firewall scripts automate iptables rules for consistent deployments
