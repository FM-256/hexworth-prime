net/http -- Client

Go's HTTP client is production-grade out of the box. Always use a custom http.Client with a timeout -- the default has none, which can cause goroutines to hang indefinitely waiting for a response that never arrives.

http_client.go
package main

import (
    "fmt"
    "io"
    "net/http"
    "time"
)

func main() {
    // Always set a timeout -- never use http.DefaultClient in production
    client := &http.Client{Timeout: 10 * time.Second}

    resp, err := client.Get("https://httpbin.org/json")
    if err != nil {
        fmt.Printf("request failed: %v\n", err)
        return
    }
    defer resp.Body.Close()

    fmt.Printf("status: %s\n", resp.Status)
    fmt.Printf("content-type: %s\n", resp.Header.Get("Content-Type"))

    body, err := io.ReadAll(resp.Body)
    if err != nil {
        fmt.Printf("read failed: %v\n", err)
        return
    }
    fmt.Printf("body (%d bytes): %s\n", len(body), body[:80])
}

// status: 200 OK
// content-type: application/json

net/http -- Server

Building an HTTP server in Go takes fewer lines than most frameworks require just to initialize. The http.Handler interface is the foundation of the entire ecosystem.

http_server.go
package main

import (
    "encoding/json"
    "fmt"
    "net/http"
    "time"
)

type StatusResponse struct {
    Status    string    `json:"status"`
    Timestamp time.Time `json:"timestamp"`
    Version   string    `json:"version"`
}

func statusHandler(w http.ResponseWriter, r *http.Request) {
    if r.Method != http.MethodGet {
        http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
        return
    }

    w.Header().Set("Content-Type", "application/json")
    resp := StatusResponse{
        Status:    "operational",
        Timestamp: time.Now().UTC(),
        Version:   "1.0.0",
    }
    json.NewEncoder(w).Encode(resp)
}

func main() {
    mux := http.NewServeMux()
    mux.HandleFunc("/status", statusHandler)

    srv := &http.Server{
        Addr:         ":8080",
        Handler:      mux,
        ReadTimeout:  5 * time.Second,
        WriteTimeout: 10 * time.Second,
        IdleTimeout:  120 * time.Second,
    }

    fmt.Println("listening on :8080")
    fmt.Println(srv.ListenAndServe())
}
Always set server timeouts

An http.Server without ReadTimeout and WriteTimeout is vulnerable to Slowloris attacks, where a client opens connections and sends data very slowly, eventually exhausting the server's goroutine pool. Always configure all four timeout fields.

encoding/json

JSON encoding uses struct tags to map Go field names to JSON keys. The omitempty option omits zero-value fields. - excludes a field entirely. Always validate user-supplied JSON before processing.

json.go
package main

import (
    "encoding/json"
    "fmt"
)

type Finding struct {
    Host     string   `json:"host"`
    Port     int      `json:"port"`
    CVEs     []string `json:"cves,omitempty"`
    Internal string   `json:"-"` // never serialized
}

func main() {
    // Encode struct to JSON
    f := Finding{
        Host:     "10.0.0.1",
        Port:     22,
        CVEs:     []string{"CVE-2023-38408"},
        Internal: "do not leak",
    }

    data, err := json.MarshalIndent(f, "", "  ")
    if err != nil {
        fmt.Printf("marshal error: %v\n", err)
        return
    }
    fmt.Println(string(data))

    // Decode JSON to struct
    raw := []byte(`{"host":"192.168.1.1","port":443}`)
    var decoded Finding
    if err := json.Unmarshal(raw, &decoded); err != nil {
        fmt.Printf("unmarshal error: %v\n", err)
        return
    }
    fmt.Printf("decoded: %+v\n", decoded)
}

// {
//   "host": "10.0.0.1",
//   "port": 22,
//   "cves": ["CVE-2023-38408"]
// }
// decoded: {Host:192.168.1.1 Port:443 CVEs:[] Internal:}

os, io, and path/filepath

File operations use os for system calls and io for stream manipulation. path/filepath handles OS-specific path separators correctly -- always use it instead of string concatenation.

files.go
package main

import (
    "bufio"
    "fmt"
    "os"
    "path/filepath"
    "strings"
)

// processIPList reads IPs from a file, one per line
func processIPList(path string) ([]string, error) {
    f, err := os.Open(path)
    if err != nil {
        return nil, fmt.Errorf("open %s: %w", path, err)
    }
    defer f.Close()

    var ips []string
    scanner := bufio.NewScanner(f)
    for scanner.Scan() {
        line := strings.TrimSpace(scanner.Text())
        if line == "" || strings.HasPrefix(line, "#") {
            continue
        }
        ips = append(ips, line)
    }
    return ips, scanner.Err()
}

func writeReport(dir, filename string, data []byte) error {
    // filepath.Join handles OS path separators correctly
    path := filepath.Join(dir, filename)
    return os.WriteFile(path, data, 0600) // owner rw, no others
}

func walkLogs(root string) error {
    return filepath.Walk(root, func(path string, info os.FileInfo, err error) error {
        if err != nil { return err }
        if !info.IsDir() && filepath.Ext(path) == ".log" {
            fmt.Printf("found log: %s (%d bytes)\n", path, info.Size())
        }
        return nil
    })
}

func main() {
    fmt.Println(filepath.Join("/var", "log", "auth.log"))
    fmt.Println(filepath.Ext("report.json"))
    fmt.Println(filepath.Base("/home/op/targets.txt"))
}

// /var/log/auth.log
// .json
// targets.txt

strings and time

strings_time.go
package main

import (
    "fmt"
    "strings"
    "time"
)

func main() {
    // strings -- essential operations
    host := "  192.168.1.1  "
    host = strings.TrimSpace(host)

    fmt.Println(strings.Contains(host, "192"))     // true
    fmt.Println(strings.HasPrefix(host, "192"))    // true
    fmt.Println(strings.Split("10.0.0.1:443", ":")) // [10.0.0.1 443]
    fmt.Println(strings.ToUpper("open"))           // OPEN
    fmt.Println(strings.Replace("x-x-x", "x", "O", 2)) // O-O-x

    parts := strings.Fields("  GET  /flag  HTTP/1.1  ")
    fmt.Println(parts) // [GET /flag HTTP/1.1]

    // time -- formatting uses a reference time: Mon Jan 2 15:04:05 2006
    now := time.Now()
    fmt.Println(now.Format(time.RFC3339))
    fmt.Println(now.Format("2006-01-02 15:04:05"))

    // Parse a timestamp
    t, err := time.Parse(time.RFC3339, "2026-03-15T10:00:00Z")
    if err == nil {
        fmt.Printf("parsed: %s\n", t.Format("Jan 02, 2006"))
    }

    // Duration arithmetic
    deadline := now.Add(24 * time.Hour)
    remaining := time.Until(deadline)
    fmt.Printf("deadline in %.0f hours\n", remaining.Hours())
}

Try It Yourself

Build a minimal recon report tool:

  1. Define a Finding struct with JSON tags for host, port, service, and severity
  2. Write an HTTP handler at /findings that accepts POST with JSON body and prints the decoded struct
  3. Write findings to a file named with the current timestamp using time.Now().Format
  4. Use bufio.Scanner to read a file of target IPs and print each non-comment line

Key Takeaways

  • Always create http.Client with a Timeout; never use http.DefaultClient in tools
  • Always set ReadTimeout, WriteTimeout, and IdleTimeout on http.Server to prevent DoS
  • Use struct tags for JSON mapping; use "-" to exclude sensitive fields from serialization
  • Use path/filepath.Join for paths -- never string concatenation
  • Go's time format reference is the specific time: Mon Jan 2 15:04:05 MST 2006 -- not format codes
