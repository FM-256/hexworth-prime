Concurrent TCP Port Scanner

A production-grade port scanner in Go leverages concurrency, rate limiting, and clean result aggregation. This example demonstrates all the patterns from previous modules working together.

scanner.go -- concurrent port scanner
package main

import (
    "fmt"
    "net"
    "sort"
    "sync"
    "time"
)

type Result struct {
    Port  int
    Open  bool
    Error error
}

// scanPort attempts a TCP connection and reports the result
func scanPort(host string, port int, timeout time.Duration) Result {
    addr := fmt.Sprintf("%s:%d", host, port)
    conn, err := net.DialTimeout("tcp", addr, timeout)
    if err != nil {
        return Result{Port: port, Open: false, Error: err}
    }
    conn.Close()
    return Result{Port: port, Open: true}
}

// ScanHost scans a range of ports concurrently with a semaphore
func ScanHost(host string, startPort, endPort, maxConcurrent int, timeout time.Duration) []int {
    sem     := make(chan struct{}, maxConcurrent)
    results := make(chan Result, endPort-startPort+1)
    var wg sync.WaitGroup

    for port := startPort; port <= endPort; port++ {
        wg.Add(1)
        go func(p int) {
            defer wg.Done()
            sem <- struct{}{}
            defer func() { <-sem }()
            results <- scanPort(host, p, timeout)
        }(port)
    }

    wg.Wait()
    close(results)

    var openPorts []int
    for r := range results {
        if r.Open {
            openPorts = append(openPorts, r.Port)
        }
    }
    sort.Ints(openPorts)
    return openPorts
}

func main() {
    host := "scanme.nmap.org"
    fmt.Printf("scanning %s ports 1-1024...\n", host)

    open := ScanHost(host, 1, 1024, 100, 2*time.Second)

    fmt.Printf("\nopen ports: %v\n", open)
}

// scanning scanme.nmap.org ports 1-1024...
// open ports: [22 80]

Banner Grabbing

Banner grabbing reads the initial bytes a service sends when a connection is established. SSH, FTP, SMTP, and many other protocols announce their version in the connection banner -- valuable for fingerprinting and finding outdated software.

banner.go
package main

import (
    "fmt"
    "net"
    "strings"
    "time"
)

// GrabBanner connects to host:port and reads the initial response
func GrabBanner(host string, port int, timeout time.Duration) (string, error) {
    addr := fmt.Sprintf("%s:%d", host, port)
    conn, err := net.DialTimeout("tcp", addr, timeout)
    if err != nil {
        return "", fmt.Errorf("connect to %s: %w", addr, err)
    }
    defer conn.Close()

    // Set deadline for the banner read
    conn.SetDeadline(time.Now().Add(timeout))

    buf := make([]byte, 1024)
    n, err := conn.Read(buf)
    if err != nil && n == 0 {
        return "", fmt.Errorf("read banner: %w", err)
    }

    banner := strings.TrimSpace(string(buf[:n]))
    return banner, nil
}

func main() {
    targets := []struct{ host string; port int }{
        {"scanme.nmap.org", 22},
        {"scanme.nmap.org", 80},
    }

    for _, t := range targets {
        banner, err := GrabBanner(t.host, t.port, 3*time.Second)
        if err != nil {
            fmt.Printf("%s:%d -- no banner\n", t.host, t.port)
            continue
        }
        fmt.Printf("%s:%d -- %s\n", t.host, t.port, banner[:min(len(banner), 80)])
    }
}

// scanme.nmap.org:22 -- SSH-2.0-OpenSSH_6.6.1p1 Ubuntu-2ubuntu2.13

Cryptography -- crypto/rand and crypto/sha256

Go's crypto package implements FIPS-quality cryptography. Always use crypto/rand for random data -- never math/rand. Use the standard library implementations; rolling your own crypto is the most reliable path to a CVE.

crypto.go -- hashing, random, HMAC
package main

import (
    "crypto/hmac"
    "crypto/rand"
    "crypto/sha256"
    "encoding/hex"
    "fmt"
)

// HashFile computes the SHA-256 hash of data
func HashData(data []byte) string {
    h := sha256.Sum256(data)
    return hex.EncodeToString(h[:])
}

// GenerateToken creates a cryptographically random token
func GenerateToken(n int) (string, error) {
    b := make([]byte, n)
    if _, err := rand.Read(b); err != nil {
        return "", fmt.Errorf("generate token: %w", err)
    }
    return hex.EncodeToString(b), nil
}

// SignRequest creates an HMAC-SHA256 signature for API request signing
func SignRequest(payload, secret []byte) string {
    mac := hmac.New(sha256.New, secret)
    mac.Write(payload)
    return hex.EncodeToString(mac.Sum(nil))
}

// VerifySignature uses constant-time comparison to prevent timing attacks
func VerifySignature(payload, secret []byte, sig string) bool {
    expected := SignRequest(payload, secret)
    return hmac.Equal([]byte(expected), []byte(sig))
}

func main() {
    // Hash a payload
    data := []byte("GET /api/users HTTP/1.1\r\nHost: api.example.com")
    fmt.Printf("SHA-256: %s\n", HashData(data))

    // Generate a session token
    token, err := GenerateToken(32)
    if err != nil { panic(err) }
    fmt.Printf("token: %s\n", token)

    // Sign and verify an API request
    secret := []byte("s3cr3t-k3y")
    sig := SignRequest(data, secret)
    fmt.Printf("sig: %s\n", sig)
    fmt.Printf("valid: %v\n", VerifySignature(data, secret, sig))
    fmt.Printf("tampered: %v\n", VerifySignature([]byte("tampered"), secret, sig))
}
hmac.Equal vs string comparison

Never compare HMAC values with == or bytes.Equal. These exit early on the first mismatch, leaking timing information that an attacker can use to forge signatures byte-by-byte. hmac.Equal runs in constant time regardless of where the mismatch occurs.

Go Fuzzing

Go 1.18 introduced native fuzzing. The fuzzer generates random inputs, tracks code coverage, and reports inputs that crash or trigger unexpected behavior in your parsers. This is the automated way to find parsing bugs before attackers do.

parser_fuzz_test.go
package parser

import (
    "testing"
)

// ParseTarget parses "host:port" strings
func ParseTarget(input string) (string, int, error) {
    // ... real implementation ...
    return "", 0, nil
}

// FuzzParseTarget feeds random strings into ParseTarget
// Run: go test -fuzz=FuzzParseTarget -fuzztime=30s
func FuzzParseTarget(f *testing.F) {
    // Seed corpus -- known interesting inputs
    f.Add("192.168.1.1:443")
    f.Add("localhost:80")
    f.Add(":22")
    f.Add("")
    f.Add("[::1]:8080") // IPv6

    f.Fuzz(func(t *testing.T, input string) {
        // The function must not panic on any input
        ParseTarget(input)
    })
}

// Terminal: fuzz for 30 seconds
// $ go test -fuzz=FuzzParseTarget -fuzztime=30s
// fuzz: elapsed: 30s, execs: 1248392 (41597/sec)
// ok  parser 30.045s

Secure Coding Patterns

Beyond the crypto package, security-focused Go code requires discipline in how you handle inputs, outputs, and system interactions.

secure_patterns.go
package main

import (
    "fmt"
    "net"
    "os/exec"
    "regexp"
    "strings"
)

// PATTERN 1: Validate IP addresses before use
func validateIP(input string) (string, error) {
    ip := net.ParseIP(strings.TrimSpace(input))
    if ip == nil {
        return "", fmt.Errorf("invalid IP address: %q", input)
    }
    return ip.String(), nil
}

// PATTERN 2: Validate hostnames to prevent SSRF
var hostnameRE = regexp.MustCompile(`^[a-z0-9]([a-z0-9\-]{0,61}[a-z0-9])?(\.[a-z0-9]([a-z0-9\-]{0,61}[a-z0-9])?)*$`)

func validateHostname(h string) bool {
    return hostnameRE.MatchString(strings.ToLower(h))
}

// PATTERN 3: Use exec.Command with args list, never shell=true style
func runNmap(target string) ([]byte, error) {
    // SAFE: args are separate strings -- no shell injection possible
    if _, err := validateIP(target); err != nil {
        return nil, fmt.Errorf("unsafe target: %w", err)
    }
    cmd := exec.Command("nmap", "-sV", "--open", "-T4", target)
    return cmd.Output()
}

// PATTERN 4: Zero sensitive data after use
func processKey(key []byte) {
    defer func() {
        for i := range key {
            key[i] = 0 // overwrite key material
        }
    }()
    // use key for crypto operations...
    fmt.Println("key processed")
}

func main() {
    ip, err := validateIP("192.168.1.1")
    fmt.Println(ip, err)

    bad, err := validateIP("192.168.1.1; rm -rf /")
    fmt.Println(bad, err)

    fmt.Println(validateHostname("example.com"))
    fmt.Println(validateHostname("evil.com; curl attacker.com"))

    key := []byte{0x01, 0x02, 0x03}
    processKey(key)
    fmt.Printf("key after use: %v\n", key) // [0 0 0]
}

Try It Yourself

Build a complete recon tool that combines all Go skills:

  1. Accept a target IP and port range via flags (--host, --start-port, --end-port, --workers)
  2. Validate the IP with net.ParseIP before scanning -- exit with an error if invalid
  3. Run the concurrent scanner with the semaphore pattern, capped at --workers concurrent goroutines
  4. For each open port, attempt a banner grab with a 2-second timeout
  5. Output results as JSON using encoding/json to a file named after the target and timestamp
  6. Compute and print the SHA-256 hash of the output file as an integrity check

Key Takeaways

  • Go's concurrency primitives make writing fast, correct network scanners natural -- goroutines + semaphore is the pattern
  • Always use crypto/rand for randomness and hmac.Equal for signature comparison
  • Never build command strings for exec -- use separate argument slices to prevent command injection
  • Go's native fuzzer is a first-class testing tool: always fuzz your parsers before shipping
  • Zero out sensitive byte slices with defer after use to minimize key material exposure in memory
