if / else

Go's if is conventional but has one powerful extension: an initialization statement before the condition. This is idiomatic Go for handling function calls that return a value and an error.

if_else.go
package main

import (
    "fmt"
    "strconv"
)

func main() {
    input := "8443"

    // Initialization statement: port is scoped to this if/else block
    if port, err := strconv.Atoi(input); err != nil {
        fmt.Printf("invalid port: %v\n", err)
    } else if port < 1 || port > 65535 {
        fmt.Println("port out of range")
    } else {
        fmt.Printf("scanning port %d\n", port)
    }

    // Standard if without initialization
    privileged := true
    if privileged {
        fmt.Println("elevated access granted")
    }
}

switch

Go's switch does not fall through by default -- unlike C. Each case is independent. This eliminates a common bug class. Use fallthrough explicitly when you actually want it.

switch.go
package main

import "fmt"

func classifyPort(port int) string {
    switch port {
    case 22:
        return "SSH"
    case 80:
        return "HTTP"
    case 443:
        return "HTTPS"
    case 3306, 5432:
        return "database"
    default:
        return "unknown"
    }
}

// Expressionless switch -- equivalent to if/else chain
func riskLevel(port int) string {
    switch {
    case port < 1024:
        return "well-known -- high visibility"
    case port < 49152:
        return "registered"
    default:
        return "ephemeral"
    }
}

func main() {
    fmt.Println(classifyPort(443))   // HTTPS
    fmt.Println(riskLevel(22))      // well-known -- high visibility
}

for -- Go's Only Loop

Go has one loop keyword: for. It covers all patterns that other languages split across for, while, do-while, and foreach.

loops.go
package main

import "fmt"

func main() {
    // Classic C-style for loop
    for i := 0; i < 5; i++ {
        fmt.Printf("scan %d\n", i)
    }

    // While-style: condition only
    attempts := 0
    for attempts < 3 {
        fmt.Println("connecting...")
        attempts++
    }

    // Infinite loop with break
    count := 0
    for {
        count++
        if count >= 3 {
            break
        }
    }

    // Range over a slice
    ports := []int{22, 80, 443, 8080}
    for i, port := range ports {
        fmt.Printf("[%d] port %d\n", i, port)
    }

    // Range over a map
    services := map[int]string{22: "ssh", 443: "https"}
    for port, name := range services {
        fmt.Printf("%d -> %s\n", port, name)
    }

    // Range over a string (yields runes, not bytes)
    for i, ch := range "Go" {
        fmt.Printf("%d: %c\n", i, ch)
    }
}

Functions

Functions in Go are first-class values. A defining feature is multiple return values -- the standard idiom for returning both a result and an error.

functions.go
package main

import (
    "errors"
    "fmt"
)

// Single return value
func square(n int) int {
    return n * n
}

// Multiple return values -- idiomatic error handling
func divide(a, b float64) (float64, error) {
    if b == 0 {
        return 0, errors.New("division by zero")
    }
    return a / b, nil
}

// Named return values (self-documenting)
func parseTarget(input string) (host string, port int, err error) {
    if input == "" {
        err = errors.New("empty target")
        return  // naked return -- returns named values
    }
    host = "192.168.1.1"
    port = 443
    return
}

// Variadic function -- accepts any number of args
func scanPorts(host string, ports ...int) {
    for _, p := range ports {
        fmt.Printf("scanning %s:%d\n", host, p)
    }
}

// Functions as values
func apply(n int, fn func(int) int) int {
    return fn(n)
}

func main() {
    fmt.Println(square(9))

    result, err := divide(10, 3)
    if err != nil {
        fmt.Println("error:", err)
    } else {
        fmt.Printf("%.4f\n", result)
    }

    scanPorts("10.0.0.1", 22, 80, 443)

    // Anonymous function (closure)
    double := func(x int) int { return x * 2 }
    fmt.Println(apply(5, double))
}

defer

defer schedules a function call to run when the surrounding function returns -- no matter how it returns. This is the Go pattern for guaranteed resource cleanup: closing files, releasing locks, closing network connections.

defer.go
package main

import (
    "fmt"
    "os"
)

func readConfig(path string) (string, error) {
    f, err := os.Open(path)
    if err != nil {
        return "", err
    }
    // defer runs when readConfig returns -- even on error paths
    defer f.Close()

    buf := make([]byte, 1024)
    n, _ := f.Read(buf)
    return string(buf[:n]), nil
}

func demoStack() {
    // Multiple defers execute in LIFO order
    defer fmt.Println("third")
    defer fmt.Println("second")
    defer fmt.Println("first")
    fmt.Println("function body")
}

func main() {
    demoStack()
}

// function body
// first
// second
// third
defer in security code

Use defer to zero out sensitive buffers after use: defer func() { for i := range key { key[i] = 0 } }(). The cleanup runs even if the function panics or returns early, preventing key material from lingering in memory.

Key Takeaways

  • if supports an initialization statement: if val, err := f(); err != nil {...} keeps scope tight
  • switch cases do not fall through by default -- eliminates a classic C bug
  • Go has one loop keyword: for covers all patterns including while and foreach
  • Multiple return values are the standard idiom: always return (result, error)
  • defer guarantees cleanup runs on function exit -- essential for file handles, locks, and sensitive data
