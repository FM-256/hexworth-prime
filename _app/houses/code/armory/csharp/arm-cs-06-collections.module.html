Core Generic Collections

collections.cs
// List<T>: ordered, dynamic size, allows duplicates
var targets = new List<string> { "10.0.0.1", "10.0.0.2", "10.0.0.3" };
targets.Add("10.0.0.4");
targets.Remove("10.0.0.2");
targets.Sort();
Console.WriteLine(targets[0]);   // 10.0.0.1

// Dictionary<K,V>: key-value lookup, O(1) average
var portMap = new Dictionary<int, string>
{
    [22]  = "SSH",
    [80]  = "HTTP",
    [443] = "HTTPS"
};

// Safe read with TryGetValue
if (portMap.TryGetValue(443, out var svc))
    Console.WriteLine($"Port 443 = {svc}");

// HashSet<T>: unique elements, O(1) contains check
var seen = new HashSet<string>();
seen.Add("10.0.0.1");
bool isNew = seen.Add("10.0.0.1");  // false: already present

// Queue<T>: FIFO for work queues and task pipelines
var scanQueue = new Queue<string>();
scanQueue.Enqueue("10.0.0.1");
scanQueue.Enqueue("10.0.0.2");
var next = scanQueue.Dequeue();  // "10.0.0.1"

Generics

Generics let you write type-safe code without knowing the concrete type at authoring time. They eliminate casting, reduce runtime errors, and improve performance by avoiding boxing of value types.

generics.cs
// Generic class: T is a placeholder for any type
public class ScanBatch<T>
{
    private readonly List<T> _items = new();

    public void Add(T item) => _items.Add(item);
    public int Count => _items.Count;
    public T Get(int index) => _items[index];

    // Constrain T to types implementing IComparable
    public T Min() where T : IComparable<T>
    {
        if (_items.Count == 0) throw new InvalidOperationException("Empty batch");
        var min = _items[0];
        foreach (var item in _items)
            if (item.CompareTo(min) < 0) min = item;
        return min;
    }
}

// Generic method
static T Clamp<T>(T value, T min, T max) where T : IComparable<T> =>
    value.CompareTo(min) < 0 ? min :
    value.CompareTo(max) > 0 ? max : value;

int port = Clamp(70000, 1, 65535);  // 65535

LINQ -- Query Syntax

LINQ (Language Integrated Query) brings SQL-like query capabilities directly into C#. It operates on any IEnumerable<T> -- lists, arrays, files, databases.

linq_query_syntax.cs
var findings = new List<ScanResult>
{
    new("10.0.0.1", 22,   "Open",   DateTime.UtcNow),
    new("10.0.0.1", 80,   "Closed", DateTime.UtcNow),
    new("10.0.0.2", 443,  "Open",   DateTime.UtcNow),
    new("10.0.0.2", 3389, "Open",   DateTime.UtcNow),
    new("10.0.0.3", 8080, "Open",   DateTime.UtcNow),
};

// Query syntax (SQL-like): deferred until enumerated
var openRdp =
    from f in findings
    where f.Status == "Open" && f.Port == 3389
    orderby f.Host
    select $"{f.Host}:{f.Port}";

foreach (var r in openRdp)
    Console.WriteLine($"RDP exposed: {r}");

LINQ -- Method Syntax & Lambdas

Method syntax is more expressive for chaining. Lambda expressions (x => expression) are the inline functions passed to LINQ operators.

linq_method_syntax.cs
// Method syntax equivalents -- same result as query syntax
var openPorts = findings
    .Where(f => f.Status == "Open")
    .OrderBy(f => f.Port)
    .Select(f => f.Port)
    .Distinct()
    .ToList();   // materialize -- deferred execution ends here

// Aggregation
int openCount = findings.Count(f => f.Status == "Open");
int maxPort   = findings.Max(f => f.Port);
double avgPort = findings.Average(f => f.Port);

// Grouping: hosts with multiple open ports
var byHost = findings
    .Where(f => f.Status == "Open")
    .GroupBy(f => f.Host)
    .Select(g => new { Host = g.Key, OpenPorts = g.Count() })
    .OrderByDescending(x => x.OpenPorts);

foreach (var h in byHost)
    Console.WriteLine($"{h.Host}: {h.OpenPorts} open ports");

// FirstOrDefault: safe single-item retrieval
var first443 = findings.FirstOrDefault(f => f.Port == 443);
if (first443 is not null)
    Console.WriteLine($"HTTPS on {first443.Host}");

// Any and All
bool hasRdp = findings.Any(f => f.Port == 3389 && f.Status == "Open");
bool allScanned = findings.All(f => f.ScannedAt > DateTime.UtcNow.AddHours(-1));

Key Takeaways

  • List<T> for ordered sequences, Dictionary<K,V> for lookups, HashSet<T> for uniqueness
  • Generics provide compile-time type safety without casting overhead
  • LINQ's query and method syntaxes are equivalent -- method syntax chains better
  • LINQ is deferred: queries do not execute until enumerated (ToList(), foreach)
  • Use FirstOrDefault instead of First when the element might not exist
  • Lambda expressions (x => x.Prop) are the backbone of LINQ method syntax
