Classes and Properties

A class is a blueprint for objects. Properties are the preferred way to expose data in C# -- they provide a controlled interface to private backing fields, with optional validation logic.

NetworkTarget.cs
public class NetworkTarget
{
    // Auto-property: compiler generates the backing field
    public string Host { get; set; } = string.Empty;

    // Property with validation logic in the setter
    private int _port;
    public int Port
    {
        get => _port;
        set
        {
            if (value < 1 || value > 65535)
                throw new ArgumentOutOfRangeException(nameof(value), "Port must be 1-65535");
            _port = value;
        }
    }

    // Computed (read-only) property
    public bool IsWellKnown => _port <= 1023;

    // Init-only property (C# 9+): set only during construction
    public string Protocol { get; init; } = "TCP";

    // Constructor
    public NetworkTarget(string host, int port)
    {
        Host = host;
        Port = port;  // goes through the validated setter
    }

    // Override ToString for readable output
    public override string ToString() =>
        $"{Protocol}://{Host}:{Port} (well-known: {IsWellKnown})";
}

// Usage
var target = new NetworkTarget("10.0.0.1", 443) { Protocol = "TLS" };
Console.WriteLine(target);
// TLS://10.0.0.1:443 (well-known: true)

Access Modifiers

access_modifiers.cs
public class Credential
{
    public string Username { get; set; } = "";     // accessible anywhere
    private string _passwordHash = "";             // only within this class
    protected string Salt { get; set; } = "";     // this class + subclasses
    internal bool IsAdmin { get; set; }           // same assembly only

    // Private method: implementation detail, not part of the public API
    private string HashPassword(string plain)
    {
        // In production: use BCrypt or Argon2, not this placeholder
        return Convert.ToBase64String(
            System.Security.Cryptography.SHA256.HashData(
                System.Text.Encoding.UTF8.GetBytes(plain + Salt)));
    }

    public void SetPassword(string plain)
    {
        Salt = Guid.NewGuid().ToString("N");
        _passwordHash = HashPassword(plain);
    }
}

Records

Records (C# 9+) are immutable reference types with value-based equality. They are ideal for data transfer objects (DTOs), configuration models, and any struct where identity is defined by content, not reference.

records.cs
// Record: immutable, with value equality and built-in ToString
public record ScanResult(string Host, int Port, string Status, DateTime ScannedAt);

var r1 = new ScanResult("10.0.0.1", 443, "Open", DateTime.UtcNow);
var r2 = r1 with { Port = 80, Status = "Closed" };  // non-destructive copy

// Value equality: two records with same data are equal
var r3 = new ScanResult("10.0.0.1", 443, "Open", r1.ScannedAt);
Console.WriteLine(r1 == r3);   // true -- same data
Console.WriteLine(r1 == r2);   // false -- different port

// ToString auto-generated: ScanResult { Host = 10.0.0.1, Port = 443, ... }
Console.WriteLine(r1);

Static Members

static_members.cs
public class PortRegistry
{
    // Static field: shared across all instances
    private static readonly Dictionary<int, string> _knownPorts = new()
    {
        { 21,   "FTP"   },
        { 22,   "SSH"   },
        { 80,   "HTTP"  },
        { 443,  "HTTPS" },
        { 3389, "RDP"   }
    };

    // Static method: callable without an instance
    public static string Lookup(int port) =>
        _knownPorts.TryGetValue(port, out var name) ? name : "Unknown";

    // Static property
    public static int Count => _knownPorts.Count;
}

// No instance needed
Console.WriteLine(PortRegistry.Lookup(443));  // HTTPS
Console.WriteLine(PortRegistry.Count);        // 5

Key Takeaways

  • Properties control access to data with optional validation in getters/setters
  • public, private, protected, and internal control visibility scope
  • Init-only properties (init) allow construction-time assignment but prevent mutation
  • Records are immutable, have value equality, and support non-destructive with copies
  • Static members belong to the type itself, not to instances
  • Override ToString() to make objects self-describing in logs and output
