What Are Sprites?

A sprite is any visual game object: the player, enemies, bullets, coins, obstacles. In Pygame, sprites are objects that have an image (what they look like) and a rect (where they are on screen). Pygame's Sprite class provides a standardized way to manage these objects.

Every Pygame sprite needs two things: self.image (a Surface to draw) and self.rect (a Rect defining position and size). The image is what gets drawn, and the rect determines where.

Creating a Sprite Class

player_sprite.py
import pygame

class Player(pygame.sprite.Sprite):
    def __init__(self, x, y):
        super().__init__()

        # Create image (colored rectangle for now)
        self.image = pygame.Surface((40, 40))
        self.image.fill((245, 158, 11))

        # Position rect
        self.rect = self.image.get_rect()
        self.rect.center = (x, y)

        # Velocity
        self.vel_x = 0
        self.vel_y = 0
        self.speed = 5

    def update(self):
        # Handle continuous key input
        keys = pygame.key.get_pressed()
        self.vel_x = 0
        self.vel_y = 0

        if keys[pygame.K_LEFT]:  self.vel_x = -self.speed
        if keys[pygame.K_RIGHT]: self.vel_x = self.speed
        if keys[pygame.K_UP]:    self.vel_y = -self.speed
        if keys[pygame.K_DOWN]:  self.vel_y = self.speed

        # Update position
        self.rect.x += self.vel_x
        self.rect.y += self.vel_y

        # Keep on screen
        self.rect.clamp_ip(pygame.display.get_surface().get_rect())

The update() method is called every frame. It reads the keyboard state, calculates velocity, moves the rect, and clamps the position to stay within the screen. The clamp_ip() method modifies the rect in place to stay within a boundary.

Loading Images

image_sprite.py
class ImageSprite(pygame.sprite.Sprite):
    def __init__(self, image_path, x, y):
        super().__init__()

        # Load image from file
        self.image = pygame.image.load(image_path).convert_alpha()

        # Scale if needed
        self.image = pygame.transform.scale(self.image, (64, 64))

        self.rect = self.image.get_rect(center=(x, y))

# If you do not have image files, create colored surfaces:
class ColorBlock(pygame.sprite.Sprite):
    def __init__(self, color, width, height, x, y):
        super().__init__()
        self.image = pygame.Surface((width, height))
        self.image.fill(color)
        self.rect = self.image.get_rect(topleft=(x, y))

Use convert_alpha() when loading images with transparency (PNG). Use convert() for opaque images. These methods convert the image format for faster rendering -- always use them.

Sprite Groups

Sprite groups manage collections of sprites. Instead of updating and drawing each sprite individually, you call group.update() and group.draw(surface) to handle all sprites at once.

groups.py
# Create groups
all_sprites = pygame.sprite.Group()
enemies = pygame.sprite.Group()
bullets = pygame.sprite.Group()

# Add sprites to groups
player = Player(400, 300)
all_sprites.add(player)

for i in range(5):
    e = Enemy(random.randint(0, 800), random.randint(0, 600))
    all_sprites.add(e)
    enemies.add(e)  # Sprite can be in multiple groups

# In the game loop:
all_sprites.update()           # Calls update() on every sprite
all_sprites.draw(screen)       # Draws every sprite's image at its rect

# Remove a sprite
sprite.kill()  # Removes from ALL groups

Collision Detection

collisions.py
# Single sprite vs group -- returns list of collided sprites
hits = pygame.sprite.spritecollide(player, enemies, False)
# Third arg: True = kill collided sprites, False = just detect

if hits:
    print(f"Hit {len(hits)} enemies!")

# Group vs group -- returns dict {sprite: [collided_sprites]}
collisions = pygame.sprite.groupcollide(bullets, enemies, True, True)
# (group1, group2, dokill1, dokill2)

# Pixel-perfect collision (slower but more accurate)
if pygame.sprite.collide_mask(sprite_a, sprite_b):
    print("Pixel-perfect hit!")

# Circle collision (for round sprites)
if pygame.sprite.collide_circle(sprite_a, sprite_b):
    print("Circle collision!")

Rectangle collision (the default) is the fastest but least accurate. It checks if the bounding rectangles overlap. For round sprites, use collide_circle. For irregular shapes, use collide_mask which checks pixel-by-pixel but is more expensive.

Putting It All Together

sprite_game.py
import pygame, sys, random

pygame.init()
screen = pygame.display.set_mode((800, 600))
clock = pygame.time.Clock()

class Player(pygame.sprite.Sprite):
    def __init__(self):
        super().__init__()
        self.image = pygame.Surface((30, 30))
        self.image.fill((245, 158, 11))
        self.rect = self.image.get_rect(center=(400, 300))
    def update(self):
        keys = pygame.key.get_pressed()
        if keys[pygame.K_LEFT]:  self.rect.x -= 5
        if keys[pygame.K_RIGHT]: self.rect.x += 5
        if keys[pygame.K_UP]:    self.rect.y -= 5
        if keys[pygame.K_DOWN]:  self.rect.y += 5
        self.rect.clamp_ip(screen.get_rect())

class Coin(pygame.sprite.Sprite):
    def __init__(self):
        super().__init__()
        self.image = pygame.Surface((15, 15))
        self.image.fill((74, 222, 128))
        self.rect = self.image.get_rect(
            center=(random.randint(20, 780), random.randint(20, 580)))

all_sprites = pygame.sprite.Group()
coins = pygame.sprite.Group()

player = Player()
all_sprites.add(player)

for _ in range(10):
    c = Coin()
    all_sprites.add(c)
    coins.add(c)

score = 0
font = pygame.font.Font(None, 36)

running = True
while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT: running = False

    all_sprites.update()

    # Check collisions -- True kills the coin
    collected = pygame.sprite.spritecollide(player, coins, True)
    score += len(collected)

    # Respawn coins
    for _ in collected:
        c = Coin()
        all_sprites.add(c)
        coins.add(c)

    screen.fill((9, 9, 16))
    all_sprites.draw(screen)

    text = font.render(f"Score: {score}", True, (255, 255, 255))
    screen.blit(text, (10, 10))

    pygame.display.flip()
    clock.tick(60)

pygame.quit()

Key Takeaways

  • Sprites need self.image (Surface) and self.rect (position/size)
  • The update() method runs every frame for game logic
  • Sprite groups manage rendering and collision for many sprites
  • spritecollide() checks one sprite against a group
  • sprite.kill() removes it from all groups
  • Use convert_alpha() when loading PNG images with transparency
