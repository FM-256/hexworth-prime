Keyboard Input

Turtle's screen object can listen for keyboard events. You register callback functions that run when specific keys are pressed. The screen.listen() method activates the keyboard listener.

keyboard_control.py
import turtle

screen = turtle.Screen()
screen.bgcolor("black")
screen.tracer(0)

player = turtle.Turtle()
player.shape("turtle")
player.color("#f59e0b")
player.penup()
player.speed(0)

STEP = 20

def go_up():
    player.setheading(90)
    player.forward(STEP)

def go_down():
    player.setheading(270)
    player.forward(STEP)

def go_left():
    player.setheading(180)
    player.forward(STEP)

def go_right():
    player.setheading(0)
    player.forward(STEP)

# Register key handlers
screen.onkey(go_up, "Up")
screen.onkey(go_down, "Down")
screen.onkey(go_left, "Left")
screen.onkey(go_right, "Right")

# WASD alternative
screen.onkey(go_up, "w")
screen.onkey(go_down, "s")
screen.onkey(go_left, "a")
screen.onkey(go_right, "d")

screen.listen()  # Start listening for keys

def update():
    screen.update()
    screen.ontimer(update, 16)

update()
screen.mainloop()

Key names for onkey(): arrow keys are "Up", "Down", "Left", "Right". Letters are lowercase: "a", "b", "c". Special keys: "space", "Return", "Escape", "BackSpace", "Tab". Use onkeypress() instead of onkey() for continuous key-held detection.

Mouse Click Events

The screen.onclick() method registers a function that receives the x and y coordinates where the user clicked. You can use this to teleport the turtle, draw at clicked locations, or create point-and-click interactions.

click_drawing.py
import turtle
import random

screen = turtle.Screen()
screen.bgcolor("#090910")
screen.tracer(0)

drawer = turtle.Turtle()
drawer.hideturtle()
drawer.speed(0)
drawer.penup()

colors = ["#f59e0b", "#3b82f6", "#4ade80", "#ef4444", "#a78bfa"]

def draw_at_click(x, y):
    drawer.goto(x, y)
    color = random.choice(colors)
    size = random.randint(10, 40)
    drawer.dot(size, color)
    screen.update()

screen.onclick(draw_at_click)

# Right-click to clear (button 3)
def clear_screen(x, y):
    drawer.clear()
    screen.update()

screen.onclick(clear_screen, btn=3)

screen.update()
screen.mainloop()

An Etch-a-Sketch Program

Combining keyboard input with drawing creates an interactive drawing program. This Etch-a-Sketch lets the user control the turtle with arrow keys, change colors with number keys, and toggle the pen up/down.

etch_a_sketch.py
import turtle

screen = turtle.Screen()
screen.setup(800, 600)
screen.bgcolor("#090910")
screen.title("Etch-a-Sketch | Arrow keys to move | C = clear | P = pen toggle")

pen = turtle.Turtle()
pen.shape("circle")
pen.color("#f59e0b")
pen.pensize(3)
pen.speed(0)

pen_down = True

def move_up():    pen.setheading(90);  pen.forward(10)
def move_down():  pen.setheading(270); pen.forward(10)
def move_left():  pen.setheading(180); pen.forward(10)
def move_right(): pen.setheading(0);   pen.forward(10)

def clear_drawing():
    pen.clear()
    pen.penup()
    pen.home()
    pen.pendown()

def toggle_pen():
    global pen_down
    pen_down = not pen_down
    if pen_down:
        pen.pendown()
        pen.shape("circle")
    else:
        pen.penup()
        pen.shape("arrow")

def set_red():    pen.pencolor("#ef4444")
def set_amber():  pen.pencolor("#f59e0b")
def set_green():  pen.pencolor("#4ade80")
def set_blue():   pen.pencolor("#3b82f6")

screen.onkeypress(move_up, "Up")
screen.onkeypress(move_down, "Down")
screen.onkeypress(move_left, "Left")
screen.onkeypress(move_right, "Right")
screen.onkey(clear_drawing, "c")
screen.onkey(toggle_pen, "p")
screen.onkey(set_red, "1")
screen.onkey(set_amber, "2")
screen.onkey(set_green, "3")
screen.onkey(set_blue, "4")

screen.listen()
screen.mainloop()

Notice the difference between onkey() and onkeypress(): onkey fires once when the key is released, while onkeypress fires repeatedly while the key is held down. For movement, use onkeypress so holding the arrow key continuously moves the turtle.

Building a Chase Game

This simple game has a player turtle that chases food items. Each food collected scores a point. This demonstrates game state management, collision detection, and score display.

chase_game.py
import turtle
import random

screen = turtle.Screen()
screen.setup(800, 600)
screen.bgcolor("#090910")
screen.title("Chase Game -- Collect the dots!")
screen.tracer(0)

# Player
player = turtle.Turtle()
player.shape("turtle")
player.color("#f59e0b")
player.penup()

# Food target
food = turtle.Turtle()
food.shape("circle")
food.color("#4ade80")
food.shapesize(0.5)
food.penup()
food.goto(random.randint(-350, 350), random.randint(-250, 250))

# Scoreboard
score_display = turtle.Turtle()
score_display.hideturtle()
score_display.penup()
score_display.goto(-380, 260)
score_display.color("white")

score = 0
STEP = 15

def update_score():
    score_display.clear()
    score_display.write(f"Score: {score}", font=("Courier", 16, "bold"))

def check_collision():
    global score
    if player.distance(food) < 20:
        score += 1
        food.goto(random.randint(-350, 350), random.randint(-250, 250))
        update_score()

def go_up():    player.setheading(90);  player.forward(STEP); check_collision()
def go_down():  player.setheading(270); player.forward(STEP); check_collision()
def go_left():  player.setheading(180); player.forward(STEP); check_collision()
def go_right(): player.setheading(0);   player.forward(STEP); check_collision()

screen.onkeypress(go_up, "Up")
screen.onkeypress(go_down, "Down")
screen.onkeypress(go_left, "Left")
screen.onkeypress(go_right, "Right")
screen.listen()

update_score()

def game_loop():
    screen.update()
    screen.ontimer(game_loop, 16)

game_loop()
screen.mainloop()

The player.distance(food) method calculates the distance between two turtles. If the distance is less than 20 pixels, we consider it a collision. The food then teleports to a new random location and the score increases.

Collision Detection Patterns

Turtle provides built-in distance checking between any two turtles. For more complex collision (rectangles, boundaries), you check coordinate ranges manually.

collision_patterns.py
# Circle collision (distance-based)
if player.distance(enemy) < 20:
    print("Hit!")

# Boundary collision (screen edges)
if player.xcor() > 380 or player.xcor() < -380:
    print("Hit wall!")

# Rectangle collision (AABB)
def rect_collision(t1, t2, size1, size2):
    return (abs(t1.xcor() - t2.xcor()) < (size1 + size2) / 2 and
            abs(t1.ycor() - t2.ycor()) < (size1 + size2) / 2)

# Zone check (is turtle inside a region?)
def in_zone(t, x1, y1, x2, y2):
    return x1 < t.xcor() < x2 and y1 < t.ycor() < y2

Displaying Text

The write() method draws text on the canvas. It is commonly used for scores, titles, and game over messages. The font parameter takes a tuple of (font_name, size, style).

text_display.py
import turtle

writer = turtle.Turtle()
writer.hideturtle()
writer.penup()
writer.color("white")

# Simple text
writer.goto(0, 100)
writer.write("Hello World", align="center", font=("Arial", 24, "bold"))

# Score display (update by clearing and rewriting)
writer.goto(-350, 250)
writer.write("Score: 0", font=("Courier", 14, "normal"))

# Update text: clear() then write() again
writer.clear()
writer.goto(-350, 250)
writer.write("Score: 10", font=("Courier", 14, "normal"))

# Align options: "left", "center", "right"

Key Takeaways

  • screen.onkey(func, key) fires on key release; onkeypress fires while held
  • screen.onclick(func) passes click coordinates to your callback
  • screen.listen() must be called to activate keyboard handling
  • player.distance(other) measures distance for collision detection
  • write() draws text; clear() then write() updates displayed text
  • Game state (score, lives) can be tracked with global variables
