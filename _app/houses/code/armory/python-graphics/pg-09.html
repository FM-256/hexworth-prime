What Is Game State?

A game is not just one continuous loop -- it has distinct phases. The title screen waits for the player to press Start. The playing state runs the game logic. The pause screen freezes action. The game over screen shows the final score. Each phase has its own event handling, update logic, and rendering.

A state machine manages these transitions. The game loop checks which state is active and runs only that state's code. States can transition to other states based on events (player dies -> game over, press Escape -> pause).

Simple State Machine

state_machine.py
import pygame, sys

# Game states
MENU = "menu"
PLAYING = "playing"
PAUSED = "paused"
GAME_OVER = "game_over"

class Game:
    def __init__(self):
        pygame.init()
        self.screen = pygame.display.set_mode((800, 600))
        self.clock = pygame.time.Clock()
        self.font = pygame.font.Font(None, 48)
        self.small_font = pygame.font.Font(None, 28)
        self.state = MENU
        self.score = 0
        self.high_score = 0
        self.lives = 3

    def run(self):
        while True:
            for event in pygame.event.get():
                if event.type == pygame.QUIT:
                    pygame.quit(); sys.exit()
                self.handle_event(event)

            self.update()
            self.draw()
            pygame.display.flip()
            self.clock.tick(60)

    def handle_event(self, event):
        if self.state == MENU:
            if event.type == pygame.KEYDOWN and event.key == pygame.K_RETURN:
                self.state = PLAYING
                self.score = 0
                self.lives = 3
        elif self.state == PLAYING:
            if event.type == pygame.KEYDOWN and event.key == pygame.K_ESCAPE:
                self.state = PAUSED
        elif self.state == PAUSED:
            if event.type == pygame.KEYDOWN:
                if event.key == pygame.K_ESCAPE:
                    self.state = PLAYING
                if event.key == pygame.K_q:
                    self.state = MENU
        elif self.state == GAME_OVER:
            if event.type == pygame.KEYDOWN and event.key == pygame.K_RETURN:
                self.state = MENU

    def update(self):
        if self.state == PLAYING:
            # Game logic here
            if self.lives <= 0:
                if self.score > self.high_score:
                    self.high_score = self.score
                self.state = GAME_OVER

    def draw(self):
        self.screen.fill((9, 9, 16))

        if self.state == MENU:
            self.draw_text("MY GAME", 400, 200, self.font, (245,158,11))
            self.draw_text("Press ENTER to Start", 400, 350, self.small_font, (200,200,200))
            self.draw_text(f"High Score: {self.high_score}", 400, 420, self.small_font, (140,140,140))
        elif self.state == PLAYING:
            # Draw game objects
            self.draw_text(f"Score: {self.score}  Lives: {self.lives}", 400, 30, self.small_font, (255,255,255))
        elif self.state == PAUSED:
            self.draw_text("PAUSED", 400, 250, self.font, (245,158,11))
            self.draw_text("ESC to Resume | Q to Quit", 400, 330, self.small_font, (160,160,160))
        elif self.state == GAME_OVER:
            self.draw_text("GAME OVER", 400, 220, self.font, (239,68,68))
            self.draw_text(f"Final Score: {self.score}", 400, 300, self.small_font, (255,255,255))
            self.draw_text("Press ENTER for Menu", 400, 380, self.small_font, (160,160,160))

    def draw_text(self, text, x, y, font, color):
        surface = font.render(text, True, color)
        rect = surface.get_rect(center=(x, y))
        self.screen.blit(surface, rect)

Game().run()

Level Progression

Levels add structure. Each level can increase difficulty (faster enemies, more obstacles), change the environment, or introduce new mechanics. Track the current level and adjust parameters accordingly.

levels.py
class LevelManager:
    def __init__(self):
        self.level = 1
        self.enemies_killed = 0
        self.enemies_per_level = 10

    def get_enemy_speed(self):
        return 2 + (self.level * 0.5)

    def get_enemy_count(self):
        return 3 + self.level

    def get_spawn_rate(self):
        return max(500, 2000 - (self.level * 200))  # ms between spawns

    def enemy_killed(self):
        self.enemies_killed += 1
        if self.enemies_killed >= self.enemies_per_level:
            self.enemies_killed = 0
            self.level += 1
            return True  # Level up!
        return False

High Score Persistence

Save high scores to a file so they persist between sessions. A simple approach uses JSON for readability.

highscore.py
import json
import os

class HighScoreManager:
    def __init__(self, filename="highscores.json"):
        self.filename = filename
        self.scores = self.load()

    def load(self):
        if os.path.exists(self.filename):
            with open(self.filename) as f:
                return json.load(f)
        return []

    def save(self):
        with open(self.filename, "w") as f:
            json.dump(self.scores, f)

    def add_score(self, name, score):
        self.scores.append({"name": name, "score": score})
        self.scores.sort(key=lambda x: x["score"], reverse=True)
        self.scores = self.scores[:10]  # Keep top 10
        self.save()

    def is_high_score(self, score):
        if len(self.scores) < 10:
            return True
        return score > self.scores[-1]["score"]

    def get_top(self, n=5):
        return self.scores[:n]

Transition Effects

Smooth transitions between states make the game feel polished. Fade in/out effects are the simplest and most effective.

transitions.py
def fade_out(screen, duration_frames=30):
    """Fade the screen to black over N frames."""
    overlay = pygame.Surface(screen.get_size())
    overlay.fill((0, 0, 0))
    for alpha in range(0, 256, 255 // duration_frames):
        overlay.set_alpha(alpha)
        screen.blit(overlay, (0, 0))
        pygame.display.flip()
        pygame.time.delay(16)

def fade_in(screen, background, duration_frames=30):
    """Fade from black to the background surface."""
    overlay = pygame.Surface(screen.get_size())
    overlay.fill((0, 0, 0))
    for alpha in range(255, -1, -(255 // duration_frames)):
        screen.blit(background, (0, 0))
        overlay.set_alpha(alpha)
        screen.blit(overlay, (0, 0))
        pygame.display.flip()
        pygame.time.delay(16)

# Usage: before changing state
fade_out(screen)
self.state = PLAYING
# New state draws on next frame

Key Takeaways

  • Game states (menu, playing, paused, game over) each have their own logic
  • A state variable controls which code runs in the main loop
  • Level managers scale difficulty by adjusting speed, count, and spawn rates
  • High scores persist using JSON file read/write
  • Fade transitions use Surface alpha blending for polish
  • Encapsulate the game in a class for clean organization
