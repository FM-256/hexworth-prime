Classes and Objects

A class is a blueprint for objects. It bundles data (member variables) and behavior (member functions) together. In security contexts, classes are everywhere -- packet parsers, connection managers, cryptographic primitives all use this pattern.

Encapsulation Private data is hidden from outside code. Public methods are the only interface.
Constructors Automatically called on object creation. Initialize all members to defined states.
Destructors Automatically called on object destruction. Free resources -- no memory leaks.
this Pointer Implicit pointer to the current object. Used to disambiguate and chain calls.
packet.h -- a simple network packet class
#include <iostream>
#include <string>
#include <vector>
using namespace std;

class Packet {
private:
    // Private data -- callers cannot access directly
    string src_ip;
    string dst_ip;
    int src_port;
    int dst_port;
    vector<unsigned char> payload;

public:
    // Constructor: member initializer list is preferred over assignment
    Packet(const string& src, const string& dst, int sp, int dp)
        : src_ip(src), dst_ip(dst), src_port(sp), dst_port(dp) {
        cout << "[+] Packet created: " << src << ":" << sp
             << " -> " << dst << ":" << dp << endl;
    }

    // Destructor: called automatically when object goes out of scope
    ~Packet() {
        cout << "[-] Packet destroyed (src: " << src_ip << ")" << endl;
    }

    // Setter -- validates input before modifying state
    void setPayload(const vector<unsigned char>& data) {
        payload = data;
    }

    // Getter -- returns const reference (no copy, no modification)
    const vector<unsigned char>& getPayload() const {
        return payload;
    }

    // const member function: cannot modify any member variables
    void print() const {
        cout << src_ip << ":" << src_port
             << " -> " << dst_ip << ":" << dst_port
             << " [" << payload.size() << " bytes]" << endl;
    }

    // Returns this* to enable method chaining: pkt.setPort(80).print()
    Packet& setDstPort(int p) {
        dst_port = p;
        return *this;
    }
};
main.cpp -- using the Packet class
int main() {
    // Constructor is called here
    Packet pkt("192.168.1.10", "10.0.0.1", 54321, 80);

    pkt.setPayload({0x47, 0x45, 0x54});  // "GET"
    pkt.print();

    // Method chaining via this pointer
    pkt.setDstPort(443).print();

    // Destructor is called automatically here when pkt leaves scope
    return 0;
}
output
[+] Packet created: 192.168.1.10:54321 -> 10.0.0.1:80
192.168.1.10:54321 -> 10.0.0.1:80 [3 bytes]
192.168.1.10:54321 -> 10.0.0.1:443 [3 bytes]
[-] Packet destroyed (src: 192.168.1.10)

The Rule of Three

If a class manually manages a resource (heap memory, file handles, sockets), you must define all three: destructor, copy constructor, and copy assignment operator. Failing to do so causes double-free bugs, use-after-free vulnerabilities, and data corruption.

rule_of_three.cpp -- manages a raw buffer
#include <iostream>
#include <cstring>
using namespace std;

class Buffer {
private:
    char* data;
    size_t size;

public:
    // Constructor: allocates heap memory
    Buffer(size_t n) : size(n) {
        data = new char[n];
        memset(data, 0, n);
        cout << "Allocated " << n << " bytes" << endl;
    }

    // 1. Destructor: frees heap memory
    ~Buffer() {
        delete[] data;
        cout << "Freed " << size << " bytes" << endl;
    }

    // 2. Copy constructor: deep copy -- own copy of the data
    Buffer(const Buffer& other) : size(other.size) {
        data = new char[size];
        memcpy(data, other.data, size);
        cout << "Copied " << size << " bytes" << endl;
    }

    // 3. Copy assignment operator
    Buffer& operator=(const Buffer& other) {
        if (this == &other) return *this;  // self-assignment guard
        delete[] data;
        size = other.size;
        data = new char[size];
        memcpy(data, other.data, size);
        return *this;
    }

    char* get() { return data; }
    size_t getSize() const { return size; }
};

Static Members

Static member variables are shared across all instances of a class. Static member functions can be called without an object instance. Useful for factory patterns, counters, and shared state.

static_members.cpp
#include <iostream>
using namespace std;

class Connection {
private:
    int id;
    static int count;  // shared across ALL Connection instances

public:
    Connection() : id(++count) {
        cout << "Connection #" << id << " opened" << endl;
    }

    ~Connection() {
        count--;
        cout << "Connection #" << id << " closed" << endl;
    }

    // Static function: can be called as Connection::getCount()
    static int getCount() { return count; }
};

// Static member must be defined outside the class
int Connection::count = 0;

int main() {
    cout << "Active: " << Connection::getCount() << endl;  // 0
    {
        Connection c1, c2, c3;
        cout << "Active: " << Connection::getCount() << endl;  // 3
    } // c1, c2, c3 destroyed here
    cout << "Active: " << Connection::getCount() << endl;  // 0
    return 0;
}

Key Takeaways

  • Classes bundle data and behavior; access specifiers enforce encapsulation
  • Use member initializer lists in constructors for efficiency and correctness
  • Mark member functions const if they do not modify state -- enables use on const objects
  • The Rule of Three: if you define a destructor, also define copy constructor and copy assignment
  • this is a pointer to the current object -- return *this for method chaining
  • Destructors run automatically when objects go out of scope -- no manual cleanup needed
