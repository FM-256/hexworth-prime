Function Templates

Templates let you write code once and use it with multiple types. The compiler generates a concrete version for each type you use -- zero runtime overhead, full type checking.

templates.cpp
#include <iostream>
#include <vector>
using namespace std;

// Generic max: works for any type with operator<
template<typename T>
T myMax(const T& a, const T& b) {
    return (a > b) ? a : b;
}

// Generic print for any container supporting range-for
template<typename Container>
void printAll(const Container& c, const string& label) {
    cout << label << ": ";
    for (const auto& item : c) cout << item << " ";
    cout << endl;
}

int main() {
    cout << myMax(10, 20) << endl;          // int version: 20
    cout << myMax(3.14, 2.71) << endl;    // double version: 3.14
    cout << myMax('z', 'a') << endl;       // char version: z

    vector<int> ports = {22, 80, 443, 8080};
    printAll(ports, "ports");

    return 0;
}

Class Templates

Class templates parameterize an entire class on one or more types. This is how vector<T>, map<K,V>, and smart pointers are implemented in the STL.

ring_buffer.cpp -- fixed-size circular buffer
template<typename T, size_t N>
class RingBuffer {
private:
    T data[N];
    size_t head = 0, tail = 0, count = 0;

public:
    void push(const T& val) {
        data[tail] = val;
        tail = (tail + 1) % N;
        if (count < N) count++;
        else head = (head + 1) % N;  // overwrite oldest
    }

    T pop() {
        T val = data[head];
        head = (head + 1) % N;
        count--;
        return val;
    }

    bool empty() const { return count == 0; }
    size_t size() const { return count; }
};

// Usage: network packet log buffer holding last 8 packets
int main() {
    RingBuffer<int, 4> log;
    log.push(80); log.push(443); log.push(22); log.push(8080);
    log.push(3306);  // 80 is overwritten
    while (!log.empty())
        cout << log.pop() << " ";  // 443 22 8080 3306
    cout << endl;
    return 0;
}

STL Containers

vector<T> Dynamic array. O(1) random access. O(n) insert/erase at front. Most commonly used container.
map<K,V> Sorted key-value store. O(log n) all ops. Keys are unique. Backed by a red-black tree.
unordered_map<K,V> Hash map. O(1) average lookup. No ordering. Faster than map for large datasets.
set<T> Sorted unique values. O(log n). Use unordered_set for O(1) when order doesn't matter.
stl_containers.cpp -- security tooling examples
#include <iostream>
#include <vector>
#include <map>
#include <set>
#include <unordered_map>
#include <algorithm>
using namespace std;

int main() {
    // vector: discovered open ports
    vector<int> open_ports = {22, 80, 443, 8080, 22};
    open_ports.push_back(3306);
    cout << "Ports found: " << open_ports.size() << endl;

    // map: IP to hostname mapping
    map<string, string> dns_cache;
    dns_cache["192.168.1.1"] = "gateway.local";
    dns_cache["10.0.0.1"]    = "dc01.corp";
    dns_cache["10.0.0.2"]    = "dc02.corp";

    // Iterate map in key order (sorted by IP string)
    for (const auto& [ip, host] : dns_cache) {
        cout << ip << " -> " << host << endl;
    }

    // set: unique CVE IDs collected during scan
    set<string> cves;
    cves.insert("CVE-2021-44228");  // Log4Shell
    cves.insert("CVE-2021-44228");  // duplicate -- ignored
    cves.insert("CVE-2022-30190");  // Follina
    cout << "Unique CVEs: " << cves.size() << endl;  // 2

    // unordered_map: fast port-to-service lookup
    unordered_map<int, string> services = {
        {22, "ssh"}, {80, "http"}, {443, "https"}, {3306, "mysql"}
    };

    for (int p : open_ports) {
        auto it = services.find(p);
        if (it != services.end()) {
            cout << "Port " << p << ": " << it->second << endl;
        }
    }

    return 0;
}

STL Algorithms

The <algorithm> header provides a rich set of generic algorithms that work on any container via iterators. These are battle-tested, optimized, and eliminate the need to write common loop patterns manually.

algorithms.cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
using namespace std;

int main() {
    vector<int> ports = {8080, 22, 443, 80, 3306, 22, 8443};

    // sort ascending
    sort(ports.begin(), ports.end());
    // 22 22 80 443 3306 8080 8443

    // find: returns iterator to first match (or end() if not found)
    auto it = find(ports.begin(), ports.end(), 443);
    if (it != ports.end())
        cout << "HTTPS (443) found at index " << (it - ports.begin()) << endl;

    // count: how many SSH (22) ports?
    int ssh_count = count(ports.begin(), ports.end(), 22);
    cout << "SSH instances: " << ssh_count << endl;

    // remove duplicates (unique requires sorted input)
    auto new_end = unique(ports.begin(), ports.end());
    ports.erase(new_end, ports.end());
    cout << "After dedup: " << ports.size() << " ports" << endl;

    // accumulate: sum all port numbers
    int total = accumulate(ports.begin(), ports.end(), 0);
    cout << "Sum of ports: " << total << endl;

    // any_of: does the scan include any well-known ports?
    bool has_common = any_of(ports.begin(), ports.end(),
        [](int p) { return p < 1024; });
    cout << "Has well-known port: " << (has_common ? "yes" : "no") << endl;

    return 0;
}

Key Takeaways

  • Templates generate type-safe code at compile time with zero runtime overhead
  • Use vector as your default container; switch to others when justified
  • map is sorted and O(log n); unordered_map is hash-based and O(1) average
  • set automatically enforces uniqueness -- useful for deduplicating scan results
  • STL algorithms operate via iterators and work on any compliant container
  • The sort + unique + erase pattern is the standard way to deduplicate a vector
