class BankAccount:
    def __init__(self, bal=0):
        self._bal = bal
    def deposit(self, amt):
        self._bal += amt
    def withdraw(self, amt):
        if amt <= self._bal:
            self._bal -= amt
    @property
    def balance(self):
        return self._bal

        bal += amt
