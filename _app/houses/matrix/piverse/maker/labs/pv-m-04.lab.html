Task 3: Chunk Manager

Implement chunk-based streaming. Each chunk is 20 columns wide. Generate chunks ahead of the player and discard old ones.

CHUNK_W = 20 class ChunkManager: def __init__(self, seed): self.seed = seed self.chunks = {} def get_chunk(self, idx): if idx not in self.chunks: cols = [] for c in range(idx * CHUNK_W, (idx + 1) * CHUNK_W): cols.append(generate_perlin_column(c, self.seed)) self.chunks[idx] = cols return self.chunks[idx] def update(self, player_chunk): # Ensure 3 chunks ahead for i in range(player_chunk - 1, player_chunk + 4): self.get_chunk(i) # Discard old old = [k for k in self.chunks if k < player_chunk - 2] for k in old: del self.chunks[k]

import math def get_gap_chance(chunk_idx): # 5% at start, up to 35% by chunk 100 raw = math.log(chunk_idx + 1) / math.log(200) return 0.05 + min(raw, 0.8) * 0.3
