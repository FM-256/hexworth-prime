Task 1: Player with Gravity

Create platformer.py. Define a Player class with position, velocity, and gravity. The player should fall and accelerate downward until hitting the bottom of the screen.

import pygame pygame.init() SCREEN_W, SCREEN_H = 800, 600 TILE_SIZE = 32 GRAVITY = 0.5 MAX_FALL = 12 JUMP_FORCE = -10 class Player: def __init__(self, x, y): self.rect = pygame.Rect(x, y, 24, 36) self.vel_x = 0 self.vel_y = 0 self.on_ground = False self.speed = 4 def update(self, keys): # Horizontal input self.vel_x = 0 if keys[pygame.K_LEFT]: self.vel_x = -self.speed if keys[pygame.K_RIGHT]: self.vel_x = self.speed # Jump if keys[pygame.K_SPACE] and self.on_ground: self.vel_y = JUMP_FORCE self.on_ground = False # Gravity self.vel_y += GRAVITY if self.vel_y > MAX_FALL: self.vel_y = MAX_FALL

Verify the player falls when the program starts. Pressing space while "on_ground" should launch them upward in a parabolic arc.

Task 3: Tile Collision

Implement the separated-axis collision system. Move X, resolve X collisions, then move Y, resolve Y collisions. The player should land on solid tiles and be blocked by walls.

def get_hits(rect, tiles): hits = [] lc = max(0, rect.left // TILE_SIZE) rc = min(len(tiles[0]) - 1, rect.right // TILE_SIZE) tr = max(0, rect.top // TILE_SIZE) br = min(len(tiles) - 1, rect.bottom // TILE_SIZE) for r in range(tr, br + 1): for c in range(lc, rc + 1): if tiles[r][c] != 0: t = pygame.Rect(c * TILE_SIZE, r * TILE_SIZE, TILE_SIZE, TILE_SIZE) if rect.colliderect(t): hits.append(t) return hits def move_and_collide(player, tiles): # Horizontal player.rect.x += player.vel_x for t in get_hits(player.rect, tiles): if player.vel_x > 0: player.rect.right = t.left elif player.vel_x < 0: player.rect.left = t.right # Vertical player.rect.y += player.vel_y player.on_ground = False for t in get_hits(player.rect, tiles): if player.vel_y > 0: player.rect.bottom = t.top player.on_ground = True elif player.vel_y < 0: player.rect.top = t.bottom player.vel_y = 0
