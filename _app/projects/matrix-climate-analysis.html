Mission Brief

Your assignment is to build a complete climate time-series pipeline in Python -- from generating a realistic synthetic temperature record to detecting a measurable warming trend and projecting it forward. The dataset is self-contained: a seeded numpy generator produces 144 years of monthly temperature readings (1880–2023) with a built-in long-term trend, annual seasonality, and random noise. No external download is required.

The analytical pipeline has four stages. First you explore the raw series visually and resample it to monthly and annual averages using pandas. Second you quantify the warming trend with a linear regression via scipy.stats.linregress and report the slope in degrees Celsius per decade. Third you decompose the signal into trend, seasonal cycle, and residual components using statsmodels.tsa.seasonal.seasonal_decompose and plot temperature anomalies relative to a 1951–1980 baseline -- warm years in red, cool years in blue. Fourth you extrapolate the linear trend to 2040 and display an honest caveat that this is a teaching exercise, not a climate model.

By the end you will have produced five plots, a printed warming rate, and a complete Python script you can modify and extend. Every command runs headlessly via matplotlib.use('Agg'), so the script works equally well on a local machine, a Jupyter notebook, or a remote server.

    import numpy as np import pandas as pd # Seed for reproducibility -- every student gets identical numbers. rng = np.random.default_rng(seed=42) # Build a monthly DatetimeIndex: Jan 1880 through Dec 2023 (144 years). dates = pd.date_range(start="1880-01-01", end="2023-12-01", freq="MS") n = len(dates) # 1728 months # Year index starting at 0 (used for trend + noise shaping). year_frac = np.arange(n) / 12.0 # Long-term warming trend: ~+1.5 C over 144 years (~+0.10 C/decade). # Linear-only: avoids the quadratic acceleration that would produce +4.6 C. trend = 0.0105 * year_frac # Annual seasonal cycle: ~14 C peak-to-trough amplitude. # Month 0 = January = coldest; month 6 = July = warmest (Northern Hemisphere). month_of_year = np.array([d.month for d in dates]) seasonal = 7.0 * np.cos(2 * np.pi * (month_of_year - 7) / 12.0) # Gaussian noise with mild autocorrelation (AR(1) phi=0.35). white = rng.standard_normal(n) * 0.45 noise = np.zeros(n) phi = 0.35 noise[0] = white[0] for i in range(1, n): noise[i] = phi * noise[i-1] + white[i] # Baseline temperature: 14.2 C is approximate global mean surface temp (Celsius). baseline = 14.2 temp = baseline + trend + seasonal + noise df = pd.DataFrame({"temperature_c": temp}, index=dates) df.index.name = "date" df.index.freq = "MS" # mark monthly-start frequency explicitly df.to_csv("climate_synthetic.csv") print(f"Saved climate_synthetic.csv shape={df.shape}") print(df.head(6))

Trend Detection and Temperature Anomalies
Use scipy.stats.linregress on the annual series to compute the warming slope in degrees per decade. Then calculate anomalies against the 1951–1980 baseline and render a bar chart where warm years are red and cool years are blue.
  1. Create trend.py and paste this complete script:
    import matplotlib matplotlib.use('Agg') import pandas as pd import numpy as np import matplotlib.pyplot as plt from scipy import stats # ── Load data ──────────────────────────────────────────────────────────────── df = pd.read_csv( "climate_synthetic.csv", parse_dates=["date"], index_col="date" ) df = df.asfreq("MS") # Annual mean temperature. annual = df["temperature_c"].resample("YE").mean() years = annual.index.year.to_numpy(dtype=float) temps = annual.values # ── Linear regression on annual series ────────────────────────────────────── # linregress returns: slope (C/year), intercept, r, p-value, stderr slope, intercept, r_value, p_value, std_err = stats.linregress(years, temps) slope_per_decade = slope * 10 fitted = slope * years + intercept print("=" * 50) print(f" Warming rate : {slope_per_decade:+.3f} C / decade") print(f" R-squared : {r_value**2:.4f}") print(f" p-value : {p_value:.2e}") print(f" Std error : {std_err:.5f} C/yr") print("=" * 50) print("Interpretation: a slope in degrees C per YEAR is multiplied by 10") print("to express the rate per DECADE -- the standard convention in climate science.") # ── Temperature anomalies (deviation from 1951-1980 baseline) ─────────────── # The baseline is the mean of annual temperatures over the reference period. baseline_mask = (years >= 1951) & (years <= 1980) baseline_mean = temps[baseline_mask].mean() anomaly = temps - baseline_mean print(f"\n Baseline mean (1951-1980): {baseline_mean:.3f} C") print(f" Most recent anomaly (2023): {anomaly[-1]:+.3f} C") # ── Plot: trend line over annual means ─────────────────────────────────────── fig, axes = plt.subplots(2, 1, figsize=(13, 9)) fig.patch.set_facecolor("#0a0c10") ax0 = axes[0] ax0.set_facecolor("#0d1410") ax0.scatter(years, temps, color="#00ff41", s=18, alpha=0.7, label="Annual mean") ax0.plot(years, fitted, color="#facc15", linewidth=2.2, label=f"Linear trend: {slope_per_decade:+.3f} C/decade") ax0.set_title("Annual Temperature + Linear Trend", color="#00ff41", fontsize=12, pad=8) ax0.set_ylabel("Temperature (C)", color="white") ax0.tick_params(colors="white") ax0.spines["bottom"].set_color("#00ff4130") ax0.spines["left"].set_color("#00ff4130") ax0.spines["top"].set_visible(False) ax0.spines["right"].set_visible(False) ax0.legend(facecolor="#111a14", labelcolor="white", fontsize=9) ax0.text(0.02, 0.05, f"R² = {r_value**2:.3f} p = {p_value:.1e}", transform=ax0.transAxes, color="#4ade80", fontsize=8) # ── Plot: anomaly bar chart ─────────────────────────────────────────────────── ax1 = axes[1] ax1.set_facecolor("#0d1410") colors = ["#ef4444" if a >= 0 else "#3b82f6" for a in anomaly] ax1.bar(years, anomaly, color=colors, alpha=0.8, width=0.8) ax1.axhline(0, color="white", linewidth=0.8, linestyle="--", alpha=0.4) ax1.axvspan(1951, 1980, alpha=0.06, color="#4ade80", label="Baseline period (1951-1980)") ax1.set_title("Temperature Anomaly vs. 1951-1980 Baseline (red = warm / blue = cool)", color="#00ff41", fontsize=11, pad=8) ax1.set_ylabel("Anomaly (C)", color="white") ax1.set_xlabel("Year", color="white") ax1.tick_params(colors="white") ax1.spines["bottom"].set_color("#00ff4130") ax1.spines["left"].set_color("#00ff4130") ax1.spines["top"].set_visible(False) ax1.spines["right"].set_visible(False) ax1.legend(facecolor="#111a14", labelcolor="white", fontsize=9) plt.tight_layout() plt.savefig("plot_02_trend_anomaly.png", dpi=130, bbox_inches="tight", facecolor=fig.get_facecolor()) print("\nSaved plot_02_trend_anomaly.png")
  2. Run it:
    python trend.py
  3. Read the terminal output carefully. The warming rate should print near +0.10 C / decade. If you see a negative value, the index order is reversed -- confirm the CSV is sorted chronologically.

Seasonal Decomposition
Use statsmodels.tsa.seasonal.seasonal_decompose to separate the monthly series into three components: trend, seasonal cycle, and residual noise. Then extract the monthly climatology to show which months are warmest and coolest on average.
  1. Create decompose.py and paste this complete script:
    import matplotlib matplotlib.use('Agg') import pandas as pd import matplotlib.pyplot as plt from statsmodels.tsa.seasonal import seasonal_decompose # ── Load data ──────────────────────────────────────────────────────────────── df = pd.read_csv( "climate_synthetic.csv", parse_dates=["date"], index_col="date" ) df = df.asfreq("MS") # ── Seasonal decomposition ─────────────────────────────────────────────────── # model="additive": assumes temperature = trend + seasonal + residual. # period=12: one cycle = 12 months. # seasonal_decompose requires the DatetimeIndex to have a set frequency; # that is why asfreq("MS") is mandatory before calling this function. result = seasonal_decompose(df["temperature_c"], model="additive", period=12) print("Decomposition complete.") print(f" Trend range : {result.trend.dropna().min():.2f} to {result.trend.dropna().max():.2f} C") print(f" Seasonal range : {result.seasonal.min():.2f} to {result.seasonal.max():.2f} C") print(f" Residual std : {result.resid.dropna().std():.4f} C") # ── Monthly climatology (average temp by calendar month) ──────────────────── climatology = df["temperature_c"].groupby(df.index.month).mean() month_names = ["Jan","Feb","Mar","Apr","May","Jun", "Jul","Aug","Sep","Oct","Nov","Dec"] print("\n Monthly climatology (all-year average):") for m, v in zip(month_names, climatology.values): bar = "#" * int((v - 7) * 1.5) print(f" {m}: {v:5.2f} C {bar}") # ── Plot: 4-panel decomposition ─────────────────────────────────────────────── fig, axes = plt.subplots(4, 1, figsize=(13, 13), sharex=True) fig.patch.set_facecolor("#0a0c10") panels = [ (df["temperature_c"], "Observed", "#00ff41", 0.4), (result.trend, "Trend", "#facc15", 1.8), (result.seasonal, "Seasonal", "#4ade80", 1.2), (result.resid, "Residual", "#94a3b8", 0.6), ] for ax, (series, label, color, lw) in zip(axes, panels): ax.set_facecolor("#0d1410") ax.plot(series.index, series, color=color, linewidth=lw, alpha=0.85) ax.set_ylabel(label, color="white", fontsize=9) ax.tick_params(colors="white", labelsize=8) ax.spines["bottom"].set_color("#00ff4130") ax.spines["left"].set_color("#00ff4130") ax.spines["top"].set_visible(False) ax.spines["right"].set_visible(False) axes[0].set_title("Seasonal Decomposition of Monthly Temperature (additive model)", color="#00ff41", fontsize=12, pad=8) axes[-1].set_xlabel("Year", color="white", fontsize=9) plt.tight_layout() plt.savefig("plot_03_decomposition.png", dpi=130, bbox_inches="tight", facecolor=fig.get_facecolor()) print("\nSaved plot_03_decomposition.png") # ── Plot: monthly climatology bar chart ────────────────────────────────────── fig2, ax2 = plt.subplots(figsize=(10, 5)) fig2.patch.set_facecolor("#0a0c10") ax2.set_facecolor("#0d1410") ax2.bar(month_names, climatology.values, color="#00ff41", alpha=0.7, edgecolor="#4ade80", linewidth=0.8) ax2.set_title("Monthly Climatology (all-year average by calendar month)", color="#00ff41", fontsize=12, pad=8) ax2.set_ylabel("Mean Temperature (C)", color="white") ax2.tick_params(colors="white") ax2.spines["bottom"].set_color("#00ff4130") ax2.spines["left"].set_color("#00ff4130") ax2.spines["top"].set_visible(False) ax2.spines["right"].set_visible(False) plt.tight_layout() plt.savefig("plot_04_climatology.png", dpi=130, bbox_inches="tight", facecolor=fig2.get_facecolor()) print("Saved plot_04_climatology.png")
  2. Run it:
    python decompose.py
  3. Open plot_03_decomposition.png. You should see four panels stacked vertically: the raw jagged signal, the smooth rising trend curve, a repeating annual wave, and a small noisy residual centered near zero.

Linear Trend Forecast to 2040
Extrapolate the linear trend from Phase 3 forward to 2040. This is a teaching forecast that assumes the warming rate stays constant -- which is explicitly not how climate science works. The script prints a disclaimer and the plot labels the projection clearly as an extrapolation.
  1. Create forecast.py and paste this complete script:
    import matplotlib matplotlib.use('Agg') import pandas as pd import numpy as np import matplotlib.pyplot as plt from scipy import stats # ── Load and resample ──────────────────────────────────────────────────────── df = pd.read_csv( "climate_synthetic.csv", parse_dates=["date"], index_col="date" ) df = df.asfreq("MS") annual = df["temperature_c"].resample("YE").mean() years = annual.index.year.to_numpy(dtype=float) temps = annual.values # ── Re-fit the linear regression (same as Phase 3) ────────────────────────── slope, intercept, r_value, p_value, std_err = stats.linregress(years, temps) slope_per_decade = slope * 10 # ── Generate forecast years: 2024 through 2040 ─────────────────────────────── forecast_years = np.arange(2024, 2041, dtype=float) forecast_vals = slope * forecast_years + intercept # Uncertainty band: +/- 2 * std_err * sqrt(n) gives a rough extrapolation spread. # This is NOT a rigorous confidence interval for a climate projection -- # it is illustrative only, widening with distance from the training data. n = len(years) x_mean = years.mean() se_factor = std_err * np.sqrt(1 + 1/n + (forecast_years - x_mean)**2 / np.sum((years - x_mean)**2)) upper = forecast_vals + 2 * se_factor lower = forecast_vals - 2 * se_factor print("=" * 56) print(" TEACHING FORECAST -- NOT A CLIMATE MODEL") print(" This projection assumes the observed linear warming") print(" rate continues unchanged. Real climate projections") print(" require general circulation models (GCMs), emissions") print(" scenarios, and coupled ocean-atmosphere physics.") print("=" * 56) print(f"\n Historical slope : {slope_per_decade:+.3f} C / decade") print(f" Forecast 2030 : {slope * 2030 + intercept:.2f} C (annual mean)") print(f" Forecast 2040 : {slope * 2040 + intercept:.2f} C (annual mean)") delta = (slope * 2040 + intercept) - temps[-1] print(f" Delta vs 2023 : {delta:+.2f} C") # ── Historical fitted line ──────────────────────────────────────────────────── hist_fitted = slope * years + intercept # ── Plot ────────────────────────────────────────────────────────────────────── fig, ax = plt.subplots(figsize=(13, 6)) fig.patch.set_facecolor("#0a0c10") ax.set_facecolor("#0d1410") # Historical observations ax.scatter(years, temps, color="#00ff41", s=16, alpha=0.6, zorder=3, label="Annual observed") # Historical trend line ax.plot(years, hist_fitted, color="#facc15", linewidth=1.8, label="Historical trend (linear fit)") # Forecast line (dashed) ax.plot(forecast_years, forecast_vals, color="#f97316", linewidth=2.2, linestyle="--", label="Forecast 2024-2040 (linear extrapolation)") # Uncertainty band ax.fill_between(forecast_years, lower, upper, color="#f97316", alpha=0.12, label="Extrapolation uncertainty (+/- 2 SE)") # Divider at 2024 ax.axvline(x=2023.5, color="white", linewidth=0.8, linestyle=":", alpha=0.4) ax.text(2024.2, temps.min() + 0.3, "FORECAST ->", color="#f97316", fontsize=8, alpha=0.7) ax.set_title("Annual Temperature: Historical Trend + Linear Forecast to 2040\n" "(Teaching projection only -- assumes constant rate; not a climate model)", color="#00ff41", fontsize=11, pad=10) ax.set_ylabel("Mean Annual Temperature (C)", color="white") ax.set_xlabel("Year", color="white") ax.tick_params(colors="white") ax.spines["bottom"].set_color("#00ff4130") ax.spines["left"].set_color("#00ff4130") ax.spines["top"].set_visible(False) ax.spines["right"].set_visible(False) ax.legend(facecolor="#111a14", labelcolor="white", fontsize=9) plt.tight_layout() plt.savefig("plot_05_forecast.png", dpi=130, bbox_inches="tight", facecolor=fig.get_facecolor()) print("\nSaved plot_05_forecast.png")
